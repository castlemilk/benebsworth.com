  <script>
    try {
      // Setup Canvas
      const canvas = document.getElementById('canvas');
      const ctx = canvas.getContext('2d');
      const errorDisplay = document.getElementById('error-display');

      // State variables
      let numPendulums = 18;
      let gravity = 9.81;
      let cyclePeriod = 60.0; // Seconds for full alignment repeat cycle
      let baseOscillations = 24; // Longest pendulum does this many swings in cyclePeriod T
      let isPlaying = true;
      let showTrails = true;
      let simTime = 0.0;

      // Render & Camera controls
      let dpr = window.devicePixelRatio || 1;
      let width = 0, height = 0;
      
      // Camera angle (spherical coordinates)
      let camYaw = 0.65;
      let camPitch = 0.35;
      let camDist = 950;
      let targetYaw = camYaw;
      let targetPitch = camPitch;
      let targetDist = camDist;

      // Mouse drag state
      let isDragging = false;
      let lastMouseX = 0;
      let lastMouseY = 0;

      // Pendulum structures
      let pendulums = [];
      const MAX_TRAIL_POINTS = 35;
      const MAX_AMPLITUDE_RAD = 0.42; // ~24 degrees max swing angle

      // UI Elements
      const sliderN = document.getElementById('slider-n');
      const valN = document.getElementById('val-n');
      const sliderG = document.getElementById('slider-g');
      const valG = document.getElementById('val-g');
      const sliderT = document.getElementById('slider-t');
      const valT = document.getElementById('val-t');
      
      const btnPlay = document.getElementById('btn-play');
      const btnReset = document.getElementById('btn-reset');
      const btnTrail = document.getElementById('btn-trail');
      const cycleProgress = document.getElementById('cycle-progress');
      const timeDisplay = document.getElementById('time-display');
      const fpsCounter = document.getElementById('fps-counter');

      // Initialize Pendulums
      function initPendulums() {
        pendulums = [];
        const spacingZ = 22; // Distance along rail between pendulums
        const totalZLength = (numPendulums - 1) * spacingZ;
        const startZ = -totalZLength / 2;

        for (let i = 0; i < numPendulums; i++) {
          // Number of swings for pendulum i in full wave cycle T
          const oscillations = baseOscillations + i;
          // Frequency f = oscillations / T
          const freq = oscillations / cyclePeriod;
          const omega = 2 * Math.PI * freq; // Angular frequency
          
          // L = g / omega^2
          const length = gravity / (omega * omega);

          // Hue gradient across the spectrum
          const hue = Math.floor((i / Math.max(1, numPendulums - 1)) * 300);
          
          pendulums.push({
            index: i,
            oscillations: oscillations,
            omega: omega,
            length: length,
            z: startZ + i * spacingZ,
            hue: hue,
            angle: 0,
            bobPos: { x: 0, y: 0, z: 0 },
            trail: []
          });
        }
      }

      // Convert length scale for visual fitting
      // Standard calculated physics length in meters needs scaling to pixel-ish world space
      function getWorldLength(lengthInMeters) {
        // Find min/max real lengths to scale nicely into space [220..420]
        const minOsc = baseOscillations + numPendulums - 1;
        const maxOsc = baseOscillations;
        const minL = gravity / Math.pow((2 * Math.PI * minOsc) / cyclePeriod, 2);
        const maxL = gravity / Math.pow((2 * Math.PI * maxOsc) / cyclePeriod, 2);
        
        const norm = (lengthInMeters - minL) / (maxL - minL || 1);
        return 220 + norm * 200; // Visual length range in world units
      }

      // 3D Projection Engine
      function project(x, y, z) {
        // Rotate around Y axis (Yaw)
        const cosY = Math.cos(camYaw);
        const sinY = Math.sin(camYaw);
        const x1 = x * cosY + z * sinY;
        const z1 = -x * sinY + z * cosY;

        // Rotate around X axis (Pitch)
        const cosP = Math.cos(camPitch);
        const sinP = Math.sin(camPitch);
        const y2 = y * cosP - z1 * sinP;
        const z2 = y * sinP + z1 * cosP;

        // Perspective transform
        const depth = z2 + camDist;
        const focalLength = 800;
        const scale = focalLength / Math.max(depth, 1);

        return {
          sx: width / 2 + x1 * scale,
          sy: height / 2 - y2 * scale,
          scale: scale,
          depth: depth
        };
      }

      // Resizing handling
      function resize() {
        dpr = window.devicePixelRatio || 1;
        width = window.innerWidth;
        height = window.innerHeight;
        canvas.width = Math.floor(width * dpr);
        canvas.height = Math.floor(height * dpr);
        ctx.scale(dpr, dpr);
      }
      window.addEventListener('resize', resize);
      resize();

      // UI Listeners
      sliderN.addEventListener('input', (e) => {
        numPendulums = parseInt(e.target.value, 10);
        valN.textContent = numPendulums;
        initPendulums();
      });

      sliderG.addEventListener('input', (e) => {
        gravity = parseFloat(e.target.value);
        valG.textContent = gravity.toFixed(2) + ' m/s²';
        initPendulums();
      });

      sliderT.addEventListener('input', (e) => {
        cyclePeriod = parseFloat(e.target.value);
        valT.textContent = cyclePeriod + ' s';
        initPendulums();
      });

      btnPlay.addEventListener('click', () => {
        isPlaying = !isPlaying;
        btnPlay.textContent = isPlaying ? 'Pause' : 'Play';
        btnPlay.classList.toggle('btn-active', isPlaying);
      });

      btnReset.addEventListener('click', () => {
        simTime = 0.0;
        pendulums.forEach(p => p.trail = []);
      });

      btnTrail.addEventListener('click', () => {
        showTrails = !showTrails;
        btnTrail.textContent = 'Trails: ' + (showTrails ? 'ON' : 'OFF');
        btnTrail.classList.toggle('btn-active', showTrails);
        if (!showTrails) {
          pendulums.forEach(p => p.trail = []);
        }
      });

      // View presets
      document.getElementById('btn-view-iso').addEventListener('click', () => { targetYaw = 0.65; targetPitch = 0.35; targetDist = 950; });
      document.getElementById('btn-view-front').addEventListener('click', () => { targetYaw = 0; targetPitch = 0; targetDist = 900; });
      document.getElementById('btn-view-side').addEventListener('click', () => { targetYaw = Math.PI / 2; targetPitch = 0; targetDist = 950; });
      document.getElementById('btn-view-top').addEventListener('click', () => { targetYaw = 0; targetPitch = Math.PI / 2 - 0.01; targetDist = 950; });

      // Interactive Mouse / Touch Orbit Controls
      function onPointerDown(e) {
        if (e.target !== canvas) return;
        isDragging = true;
        lastMouseX = e.clientX || (e.touches && e.touches[0].clientX);
        lastMouseY = e.clientY || (e.touches && e.touches[0].clientY);
      }

      function onPointerMove(e) {
        if (!isDragging) return;
        const clientX = e.clientX || (e.touches && e.touches[0].clientX);
        const clientY = e.clientY || (e.touches && e.touches[0].clientY);
        
        const dx = clientX - lastMouseX;
        const dy = clientY - lastMouseY;

        targetYaw += dx * 0.005;
        targetPitch += dy * 0.005;

        // Clamp pitch to prevent camera inversion
        targetPitch = Math.max(-Math.PI / 2 + 0.05, Math.min(Math.PI / 2 - 0.05, targetPitch));

        lastMouseX = clientX;
        lastMouseY = clientY;
      }

      function onPointerUp() {
        isDragging = false;
      }

      window.addEventListener('mousedown', onPointerDown);
      window.addEventListener('mousemove', onPointerMove);
      window.addEventListener('mouseup', onPointerUp);
      
      window.addEventListener('touchstart', onPointerDown, { passive: true });
      window.addEventListener('touchmove', onPointerMove, { passive: true });
      window.addEventListener('touchend', onPointerUp);

      window.addEventListener('wheel', (e) => {
        targetDist += e.deltaY * 0.8;
        targetDist = Math.max(300, Math.min(2000, targetDist));
      }, { passive: true });

      // Physics & State Update
      function updatePhysics(dt) {
        if (isPlaying) {
          simTime += dt;
        }

        const railY = 180; // Top mounting rail Y offset in world units

        for (let i = 0; i < pendulums.length; i++) {
          const p = pendulums[i];
          
          // Angle from Simple Harmonic Motion equation for pendulum wave
          p.angle = MAX_AMPLITUDE_RAD * Math.cos(p.omega * simTime);

          const visLength = getWorldLength(p.length);
          
          // Position relative to pivot point (0, railY, p.z)
          const x = visLength * Math.sin(p.angle);
          const y = railY - visLength * Math.cos(p.angle);
          const z = p.z;

          p.bobPos = { x, y, z };

          if (isPlaying && showTrails) {
            p.trail.unshift({ x, y, z });
            if (p.trail.length > MAX_TRAIL_POINTS) {
              p.trail.pop();
            }
          }
        }
      }

      // Render Loop
      let lastTime = performance.now();
      let frameCount = 0;
      let fpsTime = 0;

      function render(currentTime) {
        const dt = Math.min((currentTime - lastTime) / 1000, 0.1); // Clamp delta time
        lastTime = currentTime;

        // FPS counter
        frameCount++;
        fpsTime += dt;
        if (fpsTime >= 0.5) {
          fpsCounter.textContent = Math.round(frameCount / fpsTime) + ' FPS';
          frameCount = 0;
          fpsTime = 0;
        }

        // Camera smoothing
        camYaw += (targetYaw - camYaw) * 0.1;
        camPitch += (targetPitch - camPitch) * 0.1;
        camDist += (targetDist - camDist) * 0.1;

        updatePhysics(dt);

        // UI Progress updates
        const cycleProgressVal = (simTime % cyclePeriod) / cyclePeriod;
        cycleProgress.style.width = (cycleProgressVal * 100).toFixed(1) + '%';
        timeDisplay.textContent = `${(simTime % cyclePeriod).toFixed(1)}s / ${cyclePeriod.toFixed(1)}s`;

        // Clear Background
        ctx.fillStyle = '#07070a';
        ctx.fillRect(0, 0, width, height);

        // Draw background floor grid for perspective depth
        drawGrid();

        // Draw top rail beam
        drawRail();

        // Collect all items for depth sorting
        // We need to depth-sort strings, bobs, and trails to render transparent elements correctly!
        const renderList = [];

        for (let i = 0; i < pendulums.length; i++) {
          const p = pendulums[i];
          const visLength = getWorldLength(p.length);
          const pivotPos = { x: 0, y: 180, z: p.z };
          
          const projBob = project(p.bobPos.x, p.bobPos.y, p.bobPos.z);
          const projPivot = project(pivotPos.x, pivotPos.y, pivotPos.z);

          renderList.push({
            type: 'pendulum',
            depth: projBob.depth,
            pendulum: p,
            projBob: projBob,
            projPivot: projPivot
          });
        }

        // Sort back-to-front (largest depth rendered first)
        renderList.sort((a, b) => b.depth - a.depth);

        // Render each pendulum (Trail -> String -> Bob)
        for (let item of renderList) {
          const p = item.pendulum;

          // 1. Render Trail
          if (showTrails && p.trail.length > 1) {
            ctx.beginPath();
            let first = true;
            for (let t = 0; t < p.trail.length; t++) {
              const pt = project(p.trail[t].x, p.trail[t].y, p.trail[t].z);
              if (first) {
                ctx.moveTo(pt.sx, pt.sy);
                first = false;
              } else {
                ctx.lineTo(pt.sx, pt.sy);
              }
            }
            const alpha = Math.max(0.15, 0.6 * (1 - item.depth / 2000));
            ctx.strokeStyle = `hsla(${p.hue}, 85%, 60%, ${alpha})`;
            ctx.lineWidth = 2 * item.projBob.scale;
            ctx.stroke();
          }

          // 2. Render String
          ctx.beginPath();
          ctx.moveTo(item.projPivot.sx, item.projPivot.sy);
          ctx.lineTo(item.projBob.sx, item.projBob.sy);
          ctx.strokeStyle = `rgba(200, 210, 240, ${Math.min(0.6, 250 / item.depth)})`;
          ctx.lineWidth = Math.max(0.8, 1.2 * item.projBob.scale);
          ctx.stroke();

          // 3. Render Bob (Sphere with realistic specular highlight)
          const radius = Math.max(3, 11 * item.projBob.scale);
          const grad = ctx.createRadialGradient(
            item.projBob.sx - radius * 0.3,
            item.projBob.sy - radius * 0.3,
            radius * 0.1,
            item.projBob.sx,
            item.projBob.sy,
            radius
          );

          grad.addColorStop(0, '#ffffff');
          grad.addColorStop(0.3, `hsl(${p.hue}, 90%, 65%)`);
          grad.addColorStop(0.8, `hsl(${p.hue}, 80%, 35%)`);
          grad.addColorStop(1, `hsl(${p.hue}, 90%, 15%)`);

          ctx.beginPath();
          ctx.arc(item.projBob.sx, item.projBob.sy, radius, 0, Math.PI * 2);
          ctx.fillStyle = grad;
          ctx.fill();

          // Outer ambient glow ring
          ctx.beginPath();
          ctx.arc(item.projBob.sx, item.projBob.sy, radius * 1.3, 0, Math.PI * 2);
          ctx.fillStyle = `hsla(${p.hue}, 100%, 60%, 0.12)`;
          ctx.fill();
        }

        requestAnimationFrame(render);
      }

      // Draw background perspective floor grid
      function drawGrid() {
        const gridSize = 800;
        const gridStep = 100;
        const floorY = -250;

        ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
        ctx.lineWidth = 1;

        // Lines parallel to X axis
        for (let z = -gridSize; z <= gridSize; z += gridStep) {
          const p1 = project(-gridSize, floorY, z);
          const p2 = project(gridSize, floorY, z);
          ctx.beginPath();
          ctx.moveTo(p1.sx, p1.sy);
          ctx.lineTo(p2.sx, p2.sy);
          ctx.stroke();
        }

        // Lines parallel to Z axis
        for (let x = -gridSize; x <= gridSize; x += gridStep) {
          const p1 = project(x, floorY, -gridSize);
          const p2 = project(x, floorY, gridSize);
          ctx.beginPath();
          ctx.moveTo(p1.sx, p1.sy);
          ctx.lineTo(p2.sx, p2.sy);
          ctx.stroke();
        }
      }

      // Draw structure holding pendulums
      function drawRail() {
        if (pendulums.length === 0) return;

        const railY = 180;
        const minZ = pendulums[0].z - 30;
        const maxZ = pendulums[pendulums.length - 1].z + 30;

        const p1 = project(0, railY, minZ);
        const p2 = project(0, railY, maxZ);

        // Rail bar gradient
        const railGrad = ctx.createLinearGradient(p1.sx, p1.sy, p2.sx, p2.sy);
        railGrad.addColorStop(0, 'rgba(120, 140, 180, 0.2)');
        railGrad.addColorStop(0.5, 'rgba(200, 220, 255, 0.8)');
        railGrad.addColorStop(1, 'rgba(120, 140, 180, 0.2)');

        ctx.beginPath();
        ctx.moveTo(p1.sx, p1.sy);
        ctx.lineTo(p2.sx, p2.sy);
        ctx.strokeStyle = railGrad;
        ctx.lineWidth = Math.max(2, 4 * p1.scale);
        ctx.lineCap = 'round';
        ctx.stroke();
      }

      // Boot application
      initPendulums();
      requestAnimationFrame(render);

    } catch (err) {
      const errDiv = document.getElementById('error-display');
      if (errDiv) {
        errDiv.style.display = 'block';
        errDiv.textContent = 'Simulation Error: ' + err.message;
      }
      console.error(err);
    }
  </script>
