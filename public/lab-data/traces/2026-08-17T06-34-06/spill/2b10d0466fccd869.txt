    try {
      // Safe LocalStorage Helper
      const safeStorage = {
        get(key, def) {
          try { return localStorage.getItem(key) || def; } catch(e) { return def; }
        },
        set(key, val) {
          try { localStorage.setItem(key, val); } catch(e) {}
        }
      };

      // Toast Notification

      function showToast(msg) {
        toastEl.textContent = msg;
        toastEl.classList.add('show');
        setTimeout(() => toastEl.classList.remove('show'), 3000);
      }

      // Theme Handler
      const themeToggleBtn = document.getElementById('theme-toggle');
      let currentTheme = safeStorage.get('aether_theme', 'dark');
      
      function applyTheme(theme) {
        document.documentElement.setAttribute('data-theme', theme);
        currentTheme = theme;
        safeStorage.set('aether_theme', theme);
      }
      applyTheme(currentTheme);

      themeToggleBtn.addEventListener('click', () => {
        applyTheme(currentTheme === 'dark' ? 'light' : 'dark');
      });

      // Interactive Demo Telemetry Simulator
      const nodeSlider = document.getElementById('node-slider');
      const nodeCountLabel = document.getElementById('node-count-label');
      const valOps = document.getElementById('val-ops');
      const valLatency = document.getElementById('val-latency');
      const clusterBars = document.getElementById('cluster-bars');

      function initBars() {
        clusterBars.innerHTML = '';
        for (let i = 0; i < 32; i++) {
          const bar = document.createElement('div');
          bar.className = 'node-bar';
          clusterBars.appendChild(bar);
        }
      }
      initBars();

      function updateDemo(nodeCount) {
        nodeCountLabel.textContent = `${nodeCount} Node${nodeCount > 1 ? 's' : ''}`;
        
        // Dynamic calculations
        const ops = (nodeCount * 3.0).toFixed(1);
        const latency = (2.4 / Math.sqrt(nodeCount)).toFixed(2);
        
        valOps.textContent = `${ops}M`;
        valLatency.textContent = `${latency}ms`;

        const bars = clusterBars.children;
        for (let i = 0; i < 32; i++) {
          if (i < nodeCount) {
            bars[i].classList.add('active');
            bars[i].style.height = `${Math.floor(60 + Math.random() * 40)}%`;
          } else {
            bars[i].classList.remove('active');
            bars[i].style.height = '20%';
          }
        }
      }

      nodeSlider.addEventListener('input', (e) => updateDemo(parseInt(e.target.value, 10)));

        const count = parseInt(nodeSlider.value, 10);

      const canvas = document.getElementById('bg-canvas');
      const ctx = canvas.getContext('2d');
      let width, height;

      function resizeCanvas() {
        width = window.innerWidth;
        height = window.innerHeight;
        const dpr = window.devicePixelRatio || 1;
        canvas.width = width * dpr;
        canvas.height = height * dpr;
        ctx.scale(dpr, dpr);
      }

      window.addEventListener('resize', resizeCanvas);
      resizeCanvas();

      class Particle {
        constructor() {
          this.reset();
        }

        reset() {
          this.x = Math.random() * width;
          this.y = Math.random() * height;
          this.vx = (Math.random() - 0.5) * 0.4;
          this.vy = (Math.random() - 0.5) * 0.4;
          this.radius = Math.random() * 1.8 + 0.5;
          this.alpha = Math.random() * 0.5 + 0.2;
        }

        update() {
          this.x += this.vx;
          this.y += this.vy;

          if (this.x < 0 || this.x > width || this.y < 0 || this.y > height) {
            this.reset();
          }
        }

        draw() {
          ctx.beginPath();
          ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
          const color = currentTheme === 'dark' ? '99, 102, 241' : '79, 70, 229';
          ctx.fillStyle = `rgba(${color}, ${this.alpha})`;
          ctx.fill();
        }
      }

      for (let i = 0; i < particleCount; i++) {
        particles.push(new Particle());
      }

      let animationFrameId;
      function animate() {
        ctx.clearRect(0, 0, width, height);

        // Draw ambient connecting web lines
        for (let i = 0; i < particles.length; i++) {
          particles[i].update();
          particles[i].draw();

          for (let j = i + 1; j < particles.length; j++) {
            const dx = particles[i].x - particles[j].x;
            const dy = particles[i].y - particles[j].y;
            const dist = Math.sqrt(dx * dx + dy * dy);

            if (dist < 120) {
              ctx.beginPath();
              ctx.moveTo(particles[i].x, particles[i].y);
              ctx.lineTo(particles[j].x, particles[j].y);
              const color = currentTheme === 'dark' ? '99, 102, 241' : '99, 102, 241';
              ctx.strokeStyle = `rgba(${color}, ${0.12 * (1 - dist / 120)})`;
              ctx.lineWidth = 0.8;
              ctx.stroke();
            }
          }
        }

        animationFrameId = requestAnimationFrame(animate);
      }

      animate();

      window.addEventListener('unload', () => {
        cancelAnimationFrame(animationFrameId);
