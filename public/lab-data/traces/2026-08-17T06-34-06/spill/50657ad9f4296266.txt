      // Hero Canvas Animation Engine
      let animId = null;
      function initCanvas() {
        const canvas = document.getElementById('heroCanvas');
        if (!canvas) return;
        const ctx = canvas.getContext('2d');

        function resizeCanvas() {
          const rect = canvas.getBoundingClientRect();
          canvas.width = rect.width * (window.devicePixelRatio || 1);
          canvas.height = rect.height * (window.devicePixelRatio || 1);
          ctx.scale(window.devicePixelRatio || 1, window.devicePixelRatio || 1);
        }

        resizeCanvas();
        window.addEventListener('resize', resizeCanvas);

        // Nodes setup
        const nodes = [];
        const nodeCount = 18;
        
        for (let i = 0; i < nodeCount; i++) {
          nodes.push({
            x: Math.random() * canvas.clientWidth,
            y: Math.random() * canvas.clientHeight,
            vx: (Math.random() - 0.5) * 1.2,
            vy: (Math.random() - 0.5) * 1.2,
            radius: Math.random() * 3 + 2,
            pulse: Math.random() * Math.PI
          });
        }

        function draw() {
          const width = canvas.clientWidth;
          const height = canvas.clientHeight;
          ctx.clearRect(0, 0, width, height);

          // Update & Draw Nodes
          for (let i = 0; i < nodes.length; i++) {
            const n = nodes[i];
            n.x += n.vx;
            n.y += n.vy;

            if (n.x < 0 || n.x > width) n.vx *= -1;
            if (n.y < 0 || n.y > height) n.vy *= -1;

            n.pulse += 0.03;
            const currentRadius = n.radius + Math.sin(n.pulse) * 1;

            // Connections
            for (let j = i + 1; j < nodes.length; j++) {
              const n2 = nodes[j];
              const dist = Math.hypot(n2.x - n.x, n2.y - n.y);
              if (dist < 110) {
                ctx.beginPath();
                ctx.moveTo(n.x, n.y);
                ctx.lineTo(n2.x, n2.y);
                const alpha = (1 - dist / 110) * 0.35;
                ctx.strokeStyle = state.theme === 'dark' 
                  ? `rgba(99, 102, 241, ${alpha})` 
                  : `rgba(79, 70, 229, ${alpha})`;
                ctx.lineWidth = 1;
                ctx.stroke();
              }
            }

            // Draw Node Dot
            ctx.beginPath();
            ctx.arc(n.x, n.y, Math.max(0, currentRadius), 0, Math.PI * 2);
            ctx.fillStyle = state.theme === 'dark' ? '#818cf8' : '#4f46e5';
            ctx.fill();
          }

          animId = requestAnimationFrame(draw);
        }

        if (animId) cancelAnimationFrame(animId);
        draw();
      }

      // Initial Mount
      renderApp();

      // Clean up animation frame on page unload
      window.addEventListener('unload', () => {
        if (animId) cancelAnimationFrame(animId);
      });

    } catch (err) {
      // Visible fallback error display if top-level script fails
      const alertDiv = document.createElement('div');
      alertDiv.role = 'alert';
      alertDiv.style.cssText = 'margin: 20px; padding: 20px; background: #ef4444; color: white; font-family: sans-serif; border-radius: 8px; font-weight: bold;';
      alertDiv.textContent = 'Application failed to initialize: ' + err.message;
      document.body.appendChild(alertDiv);
    }
  </script>
