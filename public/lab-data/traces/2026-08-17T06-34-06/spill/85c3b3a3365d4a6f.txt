  // Canvas Particle Animation Component
  let animationId = null;
  function initCanvas() {
    const canvas = document.getElementById('hero-canvas');
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    
    function resize() {
      const rect = canvas.parentElement.getBoundingClientRect();
      const dpr = window.devicePixelRatio || 1;
      canvas.width = rect.width * dpr;
      canvas.height = rect.height * dpr;
      ctx.scale(dpr, dpr);
    }
    resize();
    window.addEventListener('resize', resize);

    const particles = Array.from({ length: 45 }, () => ({
      x: Math.random() * canvas.offsetWidth,
      y: Math.random() * canvas.offsetHeight,
      vx: (Math.random() - 0.5) * 0.8,
      vy: (Math.random() - 0.5) * 0.8,
      radius: Math.random() * 2 + 1.5
    }));

    function draw() {
      const width = canvas.offsetWidth;
      const height = canvas.offsetHeight;
      const isDark = state.theme === 'dark';
      
      ctx.clearRect(0, 0, width, height);

      // Draw Connection Lines
      for (let i = 0; i < particles.length; i++) {
        for (let j = i + 1; j < particles.length; j++) {
          const dx = particles[i].x - particles[j].x;
          const dy = particles[i].y - particles[j].y;
          const dist = Math.sqrt(dx * dx + dy * dy);

          if (dist < 100) {
            ctx.beginPath();
            ctx.moveTo(particles[i].x, particles[i].y);
            ctx.lineTo(particles[j].x, particles[j].y);
            const opacity = (1 - dist / 100) * (isDark ? 0.35 : 0.2);
            ctx.strokeStyle = isDark ? `rgba(99, 102, 241, ${opacity})` : `rgba(79, 70, 229, ${opacity})`;
            ctx.lineWidth = 1;
            ctx.stroke();
          }
        }
      }

      // Draw Particles
      particles.forEach(p => {
        p.x += p.vx;
        p.y += p.vy;

        if (p.x < 0 || p.x > width) p.vx *= -1;
        if (p.y < 0 || p.y > height) p.vy *= -1;

        ctx.beginPath();
        ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
        ctx.fillStyle = isDark ? '#818cf8' : '#4f46e5';
        ctx.fill();
      });

      animationId = requestAnimationFrame(draw);
    }

    if (animationId) cancelAnimationFrame(animationId);
    draw();
  }

          <span style="font-weight: 600; color: ${state.billingCycle === 'monthly' ? 'var(--text-main)' : 'var(--text-muted)'};">Monthly</span>
          <label class="switch">
            <input type="checkbox" id="billing-toggle" ${state.billingCycle === 'yearly' ? 'checked' : ''}>
            <span class="slider"></span>
          </label>
          <span style="font-weight: 600; color: ${state.billingCycle === 'yearly' ? 'var(--text-main)' : 'var(--text-muted)'};">
            Yearly <span style="font-size: 0.75rem; background: var(--accent-light); color: var(--accent); padding: 0.15rem 0.5rem; border-radius: 1rem;">Save 20%</span>
          </span>
        </div>

        <div class="grid grid-cols-1 lg:grid-cols-3 gap-8">
          ${pricingPlans.map(plan => {
            const price = state.billingCycle === 'yearly' ? plan.priceYearly : plan.priceMonthly;
            return `
              <div class="pricing-card ${plan.popular ? 'popular' : ''}">
                ${plan.popular ? `<div class="pricing-badge">Most Popular</div>` : ''}
                <h3 style="font-size: 1.25rem; font-weight: 700; margin-bottom: 0.5rem;">${plan.name}</h3>
                <p style="color: var(--text-muted); font-size: 0.875rem; min-height: 2.5rem;">${plan.desc}</p>
                <div style="margin: 1.5rem 0;">
                  <span style="font-size: 2.5rem; font-weight: 800;">$${price}</span>
                  <span style="color: var(--text-muted);">/month</span>
                </div>
                <button class="btn ${plan.popular ? 'btn-primary' : 'btn-secondary'} plan-btn" data-plan="${plan.name}" style="width: 100%; margin-bottom: 1.5rem;">
                  Choose ${plan.name}
                </button>
                <div class="flex flex-col gap-4">
                  ${plan.features.map(f => `
                    <div class="flex items-center gap-4" style="font-size: 0.875rem;">
                      ${getIcon('check')}
                      <span>${f}</span>
                    </div>
                  `).join('')}
                </div>
              </div>
            `;
          }).join('')}
        </div>
      </section>

      <!-- Footer -->
      <footer style="background: var(--bg-surface); border-top: 1px solid var(--border-color); padding: 2.5rem 0; color: var(--text-muted); font-size: 0.875rem;">
        <div class="container flex flex-col sm:flex-row justify-between items-center gap-4">
          <div class="flex items-center gap-2">
            <span style="font-weight: 700; color: var(--text-main);">NexusAI Inc.</span>
            <span>© ${new Date().getFullYear()} All rights reserved.</span>
          </div>
          <div class="flex gap-6">
            <a href="#" style="color: var(--text-muted); text-decoration: none;">Privacy Policy</a>
            <a href="#" style="color: var(--text-muted); text-decoration: none;">Terms of Service</a>
            <a href="#" style="color: var(--text-muted); text-decoration: none;">System Status</a>
          </div>
        </div>
      </footer>

      <!-- Modal Container -->
      ${state.modal.open ? `
        <div class="modal-overlay" id="modal-overlay">
          <div class="modal-content">
            <h3 style="font-size: 1.25rem; font-weight: 700; margin-bottom: 0.75rem;">${state.modal.title}</h3>
            <p style="color: var(--text-muted); line-height: 1.5; margin-bottom: 1.5rem;">${state.modal.message}</p>
            <button class="btn btn-primary" id="modal-close" style="width: 100%;">Got it!</button>
          </div>
        </div>
      ` : ''}
    `;

    // Attach Event Handlers
    document.getElementById('theme-btn').onclick = toggleTheme;

    document.getElementById('hero-cta-primary').onclick = () => openModal('Deploy Free Cluster', 'Your sandbox environment has been provisioned. Cluster ID: #nx-88392.');
    document.getElementById('hero-cta-secondary').onclick = () => openModal('Documentation', 'Documentation portal loaded. Check back for full API reference.');
    document.getElementById('get-started-nav').onclick = () => openModal('Get Started', 'Welcome to NexusAI! Start by connecting your cloud provider.');

    const billingToggle = document.getElementById('billing-toggle');
    if (billingToggle) {
      billingToggle.onchange = (e) => {
        state.billingCycle = e.target.checked ? 'yearly' : 'monthly';
        render();
      };
    }

    document.querySelectorAll('.tab-btn').forEach(btn => {
      btn.onclick = () => {
        state.activeTab = parseInt(btn.getAttribute('data-tab'));
        render();
      };
    });

    document.querySelectorAll('.plan-btn').forEach(btn => {
      btn.onclick = () => {
        const planName = btn.getAttribute('data-plan');
        openModal(`Selected ${planName}`, `Thank you for selecting the ${planName} plan! Redirecting to secure checkout...`);
      };
    });

    if (state.modal.open) {
      document.getElementById('modal-close').onclick = closeModal;
      document.getElementById('modal-overlay').onclick = (e) => {
        if (e.target.id === 'modal-overlay') closeModal();
      };
    }

    // Initialize interactive visual canvas
    initCanvas();
  }

  // Initial Render
  render();

  // Cancel canvas loops on window unload to prevent memory leak
  window.addEventListener('unload', () => {
    if (animationId) cancelAnimationFrame(animationId);
  });

} catch (err) {

      <h3 style="font-size: 1.25rem; font-weight: bold; margin-bottom: 0.5rem;">Runtime Exception</h3>
      <p style="white-space: pre-wrap;">${err.stack || err.message || err}</p>
    </div>
  `;
}
