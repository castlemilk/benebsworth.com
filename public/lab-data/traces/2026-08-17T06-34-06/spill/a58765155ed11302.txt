<script>
try {
  const canvas = document.getElementById('simCanvas');
  const ctx = canvas.getContext('2d');

  // DOM elements
  const sliderR = document.getElementById('sliderR');
  const sliderL = document.getElementById('sliderL');
  const sliderC = document.getElementById('sliderC');
  const sliderVin = document.getElementById('sliderVin');

  const valR = document.getElementById('valR');
  const valL = document.getElementById('valL');
  const valC = document.getElementById('valC');
  const valVin = document.getElementById('valVin');

  const statZeta = document.getElementById('statZeta');
  const statW0 = document.getElementById('statW0');
  const statWd = document.getElementById('statWd');
  const statOvershoot = document.getElementById('statOvershoot');
  const statSettling = document.getElementById('statSettling');
  const dampingBadge = document.getElementById('dampingBadge');

  const btnUnder = document.getElementById('btnUnder');
  const btnCritical = document.getElementById('btnCritical');
  const btnOver = document.getElementById('btnOver');

  // Interactive mouse/touch cursor
  let mousePos = { x: -1, y: -1, active: false };

  // Canvas Sizing
  let dpr = window.devicePixelRatio || 1;
  let width = 0;
  let height = 0;

  function resize() {
    dpr = window.devicePixelRatio || 1;
    width = window.innerWidth;
    height = window.innerHeight;
    canvas.width = width * dpr;
    canvas.height = height * dpr;
    ctx.scale(dpr, dpr);
  }
  window.addEventListener('resize', resize);
  resize();

  // Mouse / Touch events for plot inspection
  canvas.addEventListener('pointermove', (e) => {
    const rect = canvas.getBoundingClientRect();
    mousePos.x = e.clientX - rect.left;
    mousePos.y = e.clientY - rect.top;
    mousePos.active = true;
  });
  canvas.addEventListener('pointerleave', () => {
    mousePos.active = false;
  });

  // Circuit Parameters
  let params = {
    R: parseFloat(sliderR.value),       // Ohms
    L: parseFloat(sliderL.value) * 1e-3,// Henries
    C: parseFloat(sliderC.value) * 1e-6,// Farads
    Vin: parseFloat(sliderVin.value)    // Volts
  };

  function updateInputs() {
    params.R = parseFloat(sliderR.value);
    params.L = parseFloat(sliderL.value) * 1e-3;
    params.C = parseFloat(sliderC.value) * 1e-6;
    params.Vin = parseFloat(sliderVin.value);

    valR.textContent = `${params.R.toFixed(1)} Ω`;
    valL.textContent = `${(params.L * 1e3).toFixed(1)} mH`;
    valC.textContent = `${(params.C * 1e6).toFixed(1)} µF`;
    valVin.textContent = `${params.Vin.toFixed(1)} V`;

    updateActivePresetButtons();
  }

  sliderR.addEventListener('input', updateInputs);
  sliderL.addEventListener('input', updateInputs);
  sliderC.addEventListener('input', updateInputs);
  sliderVin.addEventListener('input', updateInputs);

  // Presets
  btnUnder.addEventListener('click', () => {
    sliderL.value = 10;
    sliderC.value = 10;
    sliderR.value = 15;
    updateInputs();
  });

  btnCritical.addEventListener('click', () => {
    // R = 2 * sqrt(L/C) => L=10mH, C=10uF => L/C = 1000 => sqrt = 31.62 => R = 63.25
    sliderL.value = 10;
    sliderC.value = 10;
    sliderR.value = (2 * Math.sqrt(10e-3 / 10e-6)).toFixed(1);
    updateInputs();
  });

  btnOver.addEventListener('click', () => {
    sliderL.value = 10;
    sliderC.value = 10;
    sliderR.value = 180;
    updateInputs();
  });

  function updateActivePresetButtons() {
    btnUnder.classList.remove('active');
    btnCritical.classList.remove('active');
    btnOver.classList.remove('active');

    const alpha = params.R / (2 * params.L);
    const w0 = 1 / Math.sqrt(params.L * params.C);
    const zeta = alpha / w0;

    if (Math.abs(zeta - 1.0) < 0.03) {
      btnCritical.classList.add('active');
    } else if (zeta < 1.0) {
      btnUnder.classList.add('active');
    } else {
      btnOver.classList.add('active');
    }
  }

  // Formatting helpers
  function formatFreq(hz) {
    if (hz >= 1e6) return (hz / 1e6).toFixed(2) + ' MHz';
    if (hz >= 1e3) return (hz / 1e3).toFixed(2) + ' kHz';
    return hz.toFixed(1) + ' Hz';
  }

  function formatTime(sec) {
    if (sec < 1e-6) return (sec * 1e9).toFixed(1) + ' ns';
    if (sec < 1e-3) return (sec * 1e6).toFixed(1) + ' µs';
    if (sec < 1) return (sec * 1e3).toFixed(2) + ' ms';
    return sec.toFixed(2) + ' s';
  }

  // Time animated marker
  let animTimeOffset = 0;
  let lastTimestamp = performance.now();

  function draw() {
    const now = performance.now();
    const dt = (now - lastTimestamp) / 1000;
    lastTimestamp = now;

    // Clear background
    ctx.fillStyle = '#090a0f';
    ctx.fillRect(0, 0, width, height);

    // Calculate RLC dynamics
    const R = params.R;
    const L = params.L;
    const C = params.C;
    const Vin = params.Vin;

    const alpha = R / (2 * L);
    const w0 = 1 / Math.sqrt(L * C);
    const zeta = alpha / w0;
    const f0 = w0 / (2 * Math.PI);

    let wd = 0;
    let fd = 0;
    let type = 'under';

    if (zeta < 0.9995) {
      type = 'under';
      wd = Math.sqrt(w0 * w0 - alpha * alpha);
      fd = wd / (2 * Math.PI);
    } else if (zeta > 1.0005) {
      type = 'over';
    } else {
      type = 'critical';
    }

    // Settling time ~ 4 / alpha for under/critical, or dominant pole for overdamped
    let settlingTime = 4 / alpha;
    if (type === 'over') {
      const s1 = -alpha + Math.sqrt(alpha * alpha - w0 * w0);
      settlingTime = Math.abs(4 / s1);
    }

    // Adaptive time window (show ~2-3 settling times or at least 4 natural periods)
    let tMax = Math.max(settlingTime * 1.8, (1 / f0) * 4);
    if (!isFinite(tMax) || tMax <= 0) tMax = 0.01;

    // Advance dynamic scan marker
    animTimeOffset = (animTimeOffset + dt * (tMax * 0.4)) % tMax;

    // Calculate maximum voltage for overshoot calculation & vertical auto-scale
    let maxVc = Vin;
    let peakTime = 0;
    if (type === 'under') {
      const tp = Math.PI / wd;
      if (tp < tMax) {
        maxVc = Vin * (1 + Math.exp(-alpha * tp));
        peakTime = tp;
      }
    }
    const overshootPct = (type === 'under' && maxVc > Vin) ? ((maxVc - Vin) / Vin) * 100 : 0;

    // Update HUD Stats DOM
    statZeta.textContent = zeta.toFixed(3);
    statW0.textContent = formatFreq(f0);
    statWd.textContent = type === 'under' ? formatFreq(fd) : 'N/A';
    statOvershoot.textContent = overshootPct > 0 ? `${overshootPct.toFixed(1)}%` : '0%';
    statSettling.textContent = formatTime(settlingTime);

    if (type === 'under') {
      dampingBadge.textContent = 'UNDERDAMPED (OSCILLATORY)';
      dampingBadge.className = 'badge badge-under';
    } else if (type === 'critical') {
      dampingBadge.textContent = 'CRITICALLY DAMPED';
      dampingBadge.className = 'badge badge-critical';
    } else {
      dampingBadge.textContent = 'OVERDAMPED';
      dampingBadge.className = 'badge badge-over';
    }

    // Compute waveform points
    const samples = 600;
    const points = [];
    let maxCurrent = 0.001;

    for (let i = 0; i <= samples; i++) {
      const t = (i / samples) * tMax;
      let vc = 0;
      let iCap = 0;

      if (type === 'under') {
        const cosTerm = Math.cos(wd * t);
        const sinTerm = Math.sin(wd * t);
        vc = Vin * (1 - Math.exp(-alpha * t) * (cosTerm + (alpha / wd) * sinTerm));
        iCap = (Vin / (L * wd)) * Math.exp(-alpha * t) * sinTerm;
      } else if (type === 'critical') {
        vc = Vin * (1 - Math.exp(-alpha * t) * (1 + alpha * t));
        iCap = (Vin / L) * t * Math.exp(-alpha * t);
      } else {
        const gamma = Math.sqrt(alpha * alpha - w0 * w0);
        const s1 = -alpha + gamma;
        const s2 = -alpha - gamma;
        vc = Vin * (1 + (s2 * Math.exp(s1 * t) - s1 * Math.exp(s2 * t)) / (s1 - s2));
        iCap = (Vin / (L * (s1 - s2))) * (Math.exp(s1 * t) - Math.exp(s2 * t));
      }

      if (Math.abs(iCap) > maxCurrent) maxCurrent = Math.abs(iCap);
      points.push({ t, vc, iCap });
    }

    // Determine Plot Area layout (responsive HUD padding)
    const isMobile = width < 650;
    const hudWidth = isMobile ? 0 : 340;
    const padding = {
      left: hudWidth + (isMobile ? 40 : 60),
      right: isMobile ? 20 : 220, // space on right for pole-zero inset plot on desktop
      top: 50,
      bottom: isMobile ? 60 : 70
    };

    if (width - padding.left - padding.right < 150) {
      padding.right = 20; // fallback if screen narrow
    }

    const plotW = Math.max(100, width - padding.left - padding.right);
    const plotH = Math.max(100, height - padding.top - padding.bottom);

    // Dynamic Vertical Scales
    const maxVScale = Math.max(Vin * 1.35, maxVc * 1.15);
    const maxIScale = maxCurrent * 1.3;

    // Helper functions for mapping to screen coords
    const getX = (t) => padding.left + (t / tMax) * plotW;
    const getYV = (v) => padding.top + plotH - (v / maxVScale) * plotH;
    const getYI = (iVal) => padding.top + plotH - (iVal / maxIScale) * plotH * 0.8; // scaled current plot

    // --- DRAW GRID & AXES ---
    ctx.lineWidth = 1;
    ctx.strokeStyle = '#181e2e';
    ctx.fillStyle = '#64748b';
    ctx.font = '11px ui-monospace, monospace';

    // Horizontal Grid Lines (Voltage)
    const vSteps = 5;
    for (let i = 0; i <= vSteps; i++) {
      const vVal = (maxVScale / vSteps) * i;
      const y = getYV(vVal);

      ctx.beginPath();
      ctx.strokeStyle = i === 0 ? '#334155' : 'rgba(255, 255, 255, 0.05)';
      ctx.moveTo(padding.left, y);
      ctx.lineTo(padding.left + plotW, y);
      ctx.stroke();

      ctx.textAlign = 'right';
      ctx.textBaseline = 'middle';
      ctx.fillStyle = '#38bdf8';
      ctx.fillText(`${vVal.toFixed(1)}V`, padding.left - 8, y);
    }

    // Vertical Grid Lines (Time)
    const tSteps = 6;
    for (let i = 0; i <= tSteps; i++) {
      const tVal = (tMax / tSteps) * i;
      const x = getX(tVal);

      ctx.beginPath();
      ctx.strokeStyle = i === 0 ? '#334155' : 'rgba(255, 255, 255, 0.05)';
      ctx.moveTo(x, padding.top);
      ctx.lineTo(x, padding.top + plotH);
      ctx.stroke();

      ctx.textAlign = 'center';
      ctx.textBaseline = 'top';
      ctx.fillStyle = '#64748b';
      ctx.fillText(formatTime(tVal), x, padding.top + plotH + 8);
    }

    // Axes Titles
    ctx.save();
    ctx.fillStyle = '#94a3b8';
    ctx.font = '12px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('Time', padding.left + plotW / 2, padding.top + plotH + 32);

    ctx.rotate(-Math.PI / 2);
    ctx.fillText('Capacitor Voltage (V)', -(padding.top + plotH / 2), padding.left - 42);
    ctx.restore();

    // --- DRAW STEP INPUT TARGET LINE ---
    const yVin = getYV(Vin);
    ctx.beginPath();
    ctx.setLineDash([4, 4]);
    ctx.strokeStyle = 'rgba(245, 158, 11, 0.6)';
    ctx.lineWidth = 1.5;
    ctx.moveTo(padding.left, yVin);
    ctx.lineTo(padding.left + plotW, yVin);
    ctx.stroke();
    ctx.setLineDash([]);

    ctx.fillStyle = '#f59e0b';
    ctx.font = '11px system-ui, sans-serif';
    ctx.textAlign = 'left';
    ctx.fillText(`Step Target (${Vin}V)`, padding.left + 8, yVin - 8);

    // --- DRAW CURRENT I(t) WAVEFORM ---
    ctx.beginPath();
    ctx.strokeStyle = '#f43f5e';
    ctx.lineWidth = 2;
    for (let i = 0; i < points.length; i++) {
      const x = getX(points[i].t);
      const y = getYI(points[i].iCap);
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }
    ctx.stroke();

    // Glow effect for Voltage Waveform
    ctx.save();
    ctx.shadowColor = '#38bdf8';
    ctx.shadowBlur = 10;
    ctx.beginPath();
    ctx.strokeStyle = '#38bdf8';
    ctx.lineWidth = 2.5;
    for (let i = 0; i < points.length; i++) {
      const x = getX(points[i].t);
      const y = getYV(points[i].vc);
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }
    ctx.stroke();
    ctx.restore();

    // Fill under Voltage Curve
    ctx.beginPath();
    ctx.moveTo(getX(0), getYV(0));
    for (let i = 0; i < points.length; i++) {
      ctx.lineTo(getX(points[i].t), getYV(points[i].vc));
    }
    ctx.lineTo(getX(tMax), getYV(0));
    ctx.closePath();
    const grad = ctx.createLinearGradient(0, padding.top, 0, padding.top + plotH);
    grad.addColorStop(0, 'rgba(56, 189, 248, 0.25)');
    grad.addColorStop(1, 'rgba(56, 189, 248, 0.0)');
    ctx.fillStyle = grad;
    ctx.fill();

    // --- ANIMATED TIME MARKER ---
    const animX = getX(animTimeOffset);
    let animVc = 0, animI = 0;

    // Interpolate animated value
    const index = Math.floor((animTimeOffset / tMax) * samples);
    if (points[index]) {
      animVc = points[index].vc;
      animI = points[index].iCap;
    }

    const animYVc = getYV(animVc);

    ctx.beginPath();
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
    ctx.setLineDash([2, 2]);
    ctx.moveTo(animX, padding.top);
    ctx.lineTo(animX, padding.top + plotH);
    ctx.stroke();
    ctx.setLineDash([]);

    // Glowing dot at dynamic pulse
    ctx.beginPath();
    ctx.arc(animX, animYVc, 5, 0, Math.PI * 2);
    ctx.fillStyle = '#ffffff';
    ctx.shadowColor = '#38bdf8';
    ctx.shadowBlur = 12;
    ctx.fill();
    ctx.shadowBlur = 0;

    // --- MOUSE INSPECTION CROSSHAIR ---
    if (mousePos.active &&
        mousePos.x >= padding.left && mousePos.x <= padding.left + plotW &&
        mousePos.y >= padding.top && mousePos.y <= padding.top + plotH) {

      const hoverT = ((mousePos.x - padding.left) / plotW) * tMax;
      const hoverIndex = Math.min(samples, Math.max(0, Math.floor((hoverT / tMax) * samples)));
      const p = points[hoverIndex];

      if (p) {
        const hX = getX(p.t);
        const hY = getYV(p.vc);

        // Vertical guide line
        ctx.beginPath();
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
        ctx.lineWidth = 1;
        ctx.moveTo(hX, padding.top);
        ctx.lineTo(hX, padding.top + plotH);
        ctx.stroke();

        // Highlight marker circle
        ctx.beginPath();
        ctx.arc(hX, hY, 6, 0, Math.PI * 2);
        ctx.fillStyle = '#38bdf8';
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 2;
        ctx.fill();
        ctx.stroke();

        // Tooltip box
        const tooltipText1 = `t: ${formatTime(p.t)}`;
        const tooltipText2 = `Vc: ${p.vc.toFixed(2)} V`;
        const tooltipText3 = `I: ${(p.iCap * 1000).toFixed(1)} mA`;

        ctx.font = '11px ui-monospace, monospace';
        const boxW = 110;
        const boxH = 54;
        let boxX = hX + 12;
        let boxY = hY - 20;

        if (boxX + boxW > padding.left + plotW) boxX = hX - boxW - 12;
        if (boxY + boxH > padding.top + plotH) boxY = padding.top + plotH - boxH;

        ctx.fillStyle = 'rgba(15, 23, 42, 0.9)';
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.roundRect(boxX, boxY, boxW, boxH, 6);
        ctx.fill();
        ctx.stroke();

        ctx.fillStyle = '#e2e8f0';
        ctx.textAlign = 'left';
        ctx.fillText(tooltipText1, boxX + 8, boxY + 16);
        ctx.fillStyle = '#38bdf8';
        ctx.fillText(tooltipText2, boxX + 8, boxY + 32);
        ctx.fillStyle = '#f43f5e';
        ctx.fillText(tooltipText3, boxX + 8, boxY + 46);
      }
    }

    // --- LEGEND ---
    const legendX = padding.left + 12;
    const legendY = padding.top + 16;

    ctx.fillStyle = 'rgba(15, 23, 42, 0.7)';
    ctx.beginPath();
    ctx.roundRect(legendX, legendY, 190, 42, 6);
    ctx.fill();

    // Legend line Vc
    ctx.strokeStyle = '#38bdf8';
    ctx.lineWidth = 2.5;
    ctx.beginPath();
    ctx.moveTo(legendX + 10, legendY + 14);
    ctx.lineTo(legendX + 30, legendY + 14);
    ctx.stroke();

    ctx.fillStyle = '#e2e8f0';
    ctx.font = '11px system-ui, sans-serif';
    ctx.textAlign = 'left';
    ctx.fillText('Capacitor Voltage v_C(t)', legendX + 36, legendY + 17);

    // Legend line Current
    ctx.strokeStyle = '#f43f5e';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(legendX + 10, legendY + 30);
    ctx.lineTo(legendX + 30, legendY + 30);
    ctx.stroke();

    ctx.fillStyle = '#e2e8f0';
    ctx.fillText(`Current i(t) [Peak: ${(maxCurrent * 1000).toFixed(1)}mA]`, legendX + 36, legendY + 33);

    // --- S-PLANE / POLE PLOT INSET (On right side if space permits) ---
    if (!isMobile && padding.right >= 180) {
      const insetW = 160;
      const insetH = 160;
      const insetX = width - insetW - 20;
      const insetY = padding.top;

      ctx.fillStyle = 'rgba(15, 23, 42, 0.85)';
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.1)';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.roundRect(insetX, insetY, insetW, insetH, 8);
      ctx.fill();
      ctx.stroke();

      ctx.fillStyle = '#94a3b8';
      ctx.font = '10px system-ui, sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText('S-Plane (Poles)', insetX + insetW / 2, insetY + 14);

      const cX = insetX + insetW * 0.7; // origin moved slightly right since real poles are negative
      const cY = insetY + insetH / 2 + 8;

      // Axes
      ctx.strokeStyle = '#334155';
      ctx.beginPath();
      ctx.moveTo(insetX + 10, cY); ctx.lineTo(insetX + insetW - 10, cY); // Real axis
      ctx.moveTo(cX, insetY + 24); ctx.lineTo(cX, insetY + insetH - 10); // Imag axis
      ctx.stroke();

      ctx.fillStyle = '#475569';
      ctx.font = '9px monospace';
      ctx.fillText('Re', insetX + 20, cY - 4);
      ctx.fillText('Im', cX + 10, insetY + 32);

      // Pole calculations
      let scaleS = (insetW * 0.35) / (w0 * 1.5);
      if (!isFinite(scaleS) || scaleS === 0) scaleS = 1;

      let pole1 = { re: 0, im: 0 };
      let pole2 = { re: 0, im: 0 };

      if (type === 'under') {
        pole1 = { re: -alpha, im: wd };
        pole2 = { re: -alpha, im: -wd };
      } else if (type === 'over') {
        const gamma = Math.sqrt(alpha * alpha - w0 * w0);
        pole1 = { re: -alpha + gamma, im: 0 };
        pole2 = { re: -alpha - gamma, im: 0 };
      } else {
        pole1 = { re: -alpha, im: 0 };
        pole2 = { re: -alpha, im: 0 };
      }

      // Draw Poles (X marks)
      const drawPole = (p) => {
        const px = cX + p.re * scaleS;
        const py = cY - p.im * scaleS;

        // Clamp inside inset boundary
        const clampedX = Math.max(insetX + 10, Math.min(insetX + insetW - 10, px));
        const clampedY = Math.max(insetY + 24, Math.min(insetY + insetH - 10, py));

        ctx.strokeStyle = '#f59e0b';
        ctx.lineWidth = 2;
        const r = 5;
        ctx.beginPath();
        ctx.moveTo(clampedX - r, clampedY - r); ctx.lineTo(clampedX + r, clampedY + r);
        ctx.moveTo(clampedX + r, clampedY - r); ctx.lineTo(clampedX - r, clampedY + r);
        ctx.stroke();
      };

      drawPole(pole1);
      drawPole(pole2);
    }

    requestAnimationFrame(draw);
  }

  // Initial inputs sync & start animation loop
  updateInputs();
  requestAnimationFrame(draw);

} catch (err) {
  const errDiv = document.getElementById('error-display');
  if (errDiv) {
    errDiv.style.display = 'block';
    errDiv.textContent = 'Application Error: ' + err.message;
  }
}
