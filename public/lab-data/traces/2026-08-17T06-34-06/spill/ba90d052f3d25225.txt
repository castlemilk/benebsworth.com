        create: () => new Float32Array(16),

          return out;

          out.fill(0);
          out[0] = f / aspect;
          out[5] = f;

        lookAt: (out, eye, center, up) => {
          let x0, x1, x2, y0, y1, y2, z0, z1, z2, len;
          let eyex = eye[0], eyey = eye[1], eyez = eye[2];
          let upx = up[0], upy = up[1], upz = up[2];

          x0 = upy * z2 - upz * z1;
          x1 = upz * z0 - upx * z2;
          x2 = upx * z1 - upy * z0;
          len = 1 / (Math.hypot(x0, x1, x2) || 1);
          x0 *= len; x1 *= len; x2 *= len;

          y0 = z1 * x2 - z2 * x1;
          y1 = z2 * x0 - z0 * x2;
          y2 = z0 * x1 - z1 * x0;
          len = 1 / (Math.hypot(y0, y1, y2) || 1);
          y0 *= len; y1 *= len; y2 *= len;

          out[0] = x0; out[1] = y0; out[2] = z0; out[3] = 0;

      // Physics N-Body Gravitational Physics Step
      let maxVelocityRecorded = 1.0;
      let totalEkin = 0, totalEpot = 0;

      function updatePhysics() {
        if (isPaused) return;

        // Accelerations
        const ax = new Float32Array(NUM_PARTICLES);
        const ay = new Float32Array(NUM_PARTICLES);
        const az = new Float32Array(NUM_PARTICLES);

        totalEpot = 0;

        // Mutual Gravitational Force Calculation O(N^2)
        for (let i = 0; i < NUM_PARTICLES; i++) {
          const xi = x[i], yi = y[i], zi = z[i];
          let axi = 0, ayi = 0, azi = 0;

          for (let j = i + 1; j < NUM_PARTICLES; j++) {
            const dx = x[j] - xi;
            const dy = y[j] - yi;
            const dz = z[j] - zi;

            const r2 = dx * dx + dy * dy + dz * dz;
            const rSoft2 = r2 + softeningSq;
            const invRsoft = 1.0 / Math.sqrt(rSoft2);
            const invRsoft3 = invRsoft * invRsoft * invRsoft;

            const f = G * invRsoft3;

            // Acceleration components
            const f_mj = f * mass[j];
            axi += dx * f_mj;
            ayi += dy * f_mj;
            azi += dz * f_mj;

            const f_mi = f * mass[i];
            ax[j] -= dx * f_mi;
            ay[j] -= dy * f_mi;
            az[j] -= dz * f_mi;

            // Potential energy calculation
            totalEpot -= (G * mass[i] * mass[j]) * invRsoft;
          }

          ax[i] += axi;
          ay[i] += ayi;
          az[i] += azi;

          vy[i] += ay[i] * dt;
          vz[i] += az[i] * dt;

          y[i] += vy[i] * dt;
          z[i] += vz[i] * dt;

          const v = Math.sqrt(v2);
          if (v > currentMaxV) currentMaxV = v;

          totalEkin += 0.5 * mass[i] * v2;

          // Record trail position
          if (MAX_TRAIL_LEN > 0) {
            const tList = trailHistory[i];
            tList.push(x[i], y[i], z[i]);
            if (tList.length > MAX_TRAIL_LEN * 3) {
              tList.splice(0, 3);
            }
          }
        }

        // Smooth max velocity scaling for color mapping
        maxVelocityRecorded = maxVelocityRecorded * 0.95 + currentMaxV * 0.05;
        if (maxVelocityRecorded < currentMaxV) maxVelocityRecorded = currentMaxV;
      }

      // -------------------------------------------------------------
      // 4. Camera & Interactive Mouse Controls
      // -------------------------------------------------------------
      let cameraAngleX = 0.3;
      let cameraAngleY = 0.5;
      let cameraDistance = 45;
      let isDragging = false;
      let lastMouseX = 0, lastMouseY = 0;

      canvas.addEventListener('mousedown', (e) => {
        isDragging = true;
        lastMouseX = e.clientX;
        lastMouseY = e.clientY;
      });

      window.addEventListener('mouseup', () => { isDragging = false; });

      canvas.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        const dx = e.clientX - lastMouseX;
        const dy = e.clientY - lastMouseY;
        cameraAngleY += dx * 0.008;
        cameraAngleX += dy * 0.008;
        // Clamp pitch angle
        cameraAngleX = Math.max(-Math.PI / 2 + 0.01, Math.min(Math.PI / 2 - 0.01, cameraAngleX));
        lastMouseX = e.clientX;
        lastMouseY = e.clientY;
      });

      canvas.addEventListener('wheel', (e) => {
        cameraDistance += e.deltaY * 0.03;
        cameraDistance = Math.max(5, Math.min(180, cameraDistance));
        e.preventDefault();
      }, { passive: false });

      // Touch controls
      canvas.addEventListener('touchstart', (e) => {
        if (e.touches.length === 1) {
          isDragging = true;
          lastMouseX = e.touches[0].clientX;
          lastMouseY = e.touches[0].clientY;
        }
      });
      canvas.addEventListener('touchmove', (e) => {
        if (isDragging && e.touches.length === 1) {
          const dx = e.touches[0].clientX - lastMouseX;
          const dy = e.touches[0].clientY - lastMouseY;
          cameraAngleY += dx * 0.008;
          cameraAngleX += dy * 0.008;
          cameraAngleX = Math.max(-Math.PI / 2 + 0.01, Math.min(Math.PI / 2 - 0.01, cameraAngleX));
          lastMouseX = e.touches[0].clientX;
          lastMouseY = e.touches[0].clientY;
        }
      });
      canvas.addEventListener('touchend', () => { isDragging = false; });

      // -------------------------------------------------------------
      // 5. Render Loop & Dynamic Buffer Updates
      // -------------------------------------------------------------
      const projMatrix = Mat4.create();
      const viewMatrix = Mat4.create();

      function resize() {
        const width = window.innerWidth;
        const height = window.innerHeight;
        const dpr = Math.min(window.devicePixelRatio || 1, 2);

        canvas.width = width * dpr;
        canvas.height = height * dpr;
        gl.viewport(0, 0, canvas.width, canvas.height);

        Mat4.perspective(projMatrix, Math.PI / 4, width / height, 0.1, 500.0);
      }

      window.addEventListener('resize', resize);
      resize();

      let frameCount = 0;
      let lastFpsUpdate = performance.now();
      let currentFPS = 60;

      function render() {
        updatePhysics();

        // Clear WebGL Framebuffer
        gl.clearColor(0.03, 0.03, 0.05, 1.0);
        gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

        // Enable Additive Blending for glowing particles
        gl.enable(gl.BLEND);
        gl.blendFunc(gl.SRC_ALPHA, gl.ONE);
        gl.disable(gl.DEPTH_TEST);

        // Calculate Camera Position
        const eyeX = cameraDistance * Math.cos(cameraAngleX) * Math.sin(cameraAngleY);
        const eyeY = cameraDistance * Math.sin(cameraAngleX);
        const eyeZ = cameraDistance * Math.cos(cameraAngleX) * Math.cos(cameraAngleY);
        Mat4.lookAt(viewMatrix, [eyeX, eyeY, eyeZ], [0, 0, 0], [0, 1, 0]);

        // ----------------- Render Particles -----------------
        for (let i = 0; i < NUM_PARTICLES; i++) {
          posData[i * 3]     = x[i];
          posData[i * 3 + 1] = y[i];
          posData[i * 3 + 2] = z[i];

          const v = Math.hypot(vx[i], vy[i], vz[i]);
          const normV = v / (maxVelocityRecorded || 1.0);
          const [r, g, b] = getVelocityColor(normV);

          colData[i * 3]     = r;
          colData[i * 3 + 1] = g;
          colData[i * 3 + 2] = b;
        }

        gl.useProgram(pointProgram);
        gl.uniformMatrix4fv(pointUniforms.pM, false, projMatrix);
        gl.uniformMatrix4fv(pointUniforms.mVM, false, viewMatrix);

        // Position Attribute
        gl.bindBuffer(gl.ARRAY_BUFFER, pointPosBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, posData, gl.DYNAMIC_DRAW);
        gl.enableVertexAttribArray(pointAttrs.position);
        gl.vertexAttribPointer(pointAttrs.position, 3, gl.FLOAT, false, 0, 0);

        // Color Attribute
        gl.bindBuffer(gl.ARRAY_BUFFER, pointColBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, colData, gl.DYNAMIC_DRAW);
        gl.enableVertexAttribArray(pointAttrs.color);
        gl.vertexAttribPointer(pointAttrs.color, 3, gl.FLOAT, false, 0, 0);

        // Size Attribute
        gl.bindBuffer(gl.ARRAY_BUFFER, pointSizeBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, pSize, gl.STATIC_DRAW);
        gl.enableVertexAttribArray(pointAttrs.size);
        gl.vertexAttribPointer(pointAttrs.size, 1, gl.FLOAT, false, 0, 0);

        gl.drawArrays(gl.POINTS, 0, NUM_PARTICLES);

        // ----------------- Render Particle Trails -----------------
        if (MAX_TRAIL_LEN > 0) {
          const linePositions = [];
          const lineColors = [];

          for (let i = 0; i < NUM_PARTICLES; i++) {
            const history = trailHistory[i];
            const hLen = history.length / 3;
            if (hLen < 2) continue;

            const v = Math.hypot(vx[i], vy[i], vz[i]);
            const normV = v / (maxVelocityRecorded || 1.0);
            const [r, g, b] = getVelocityColor(normV);

            for (let j = 0; j < hLen - 1; j++) {
              linePositions.push(
                history[j * 3], history[j * 3 + 1], history[j * 3 + 2],
                history[(j + 1) * 3], history[(j + 1) * 3 + 1], history[(j + 1) * 3 + 2]
              );

              // Fade alpha along history length
              const alpha1 = (j / hLen) * 0.4;
              const alpha2 = ((j + 1) / hLen) * 0.4;

              lineColors.push(
                r, g, b, alpha1,
                r, g, b, alpha2
              );
            }
          }

          if (linePositions.length > 0) {
            gl.useProgram(lineProgram);
            gl.uniformMatrix4fv(lineUniforms.pM, false, projMatrix);
            gl.uniformMatrix4fv(lineUniforms.mVM, false, viewMatrix);

            gl.bindBuffer(gl.ARRAY_BUFFER, linePosBuffer);
            gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(linePositions), gl.DYNAMIC_DRAW);
            gl.enableVertexAttribArray(lineAttrs.position);
            gl.vertexAttribPointer(lineAttrs.position, 3, gl.FLOAT, false, 0, 0);

            gl.bindBuffer(gl.ARRAY_BUFFER, lineColBuffer);
            gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(lineColors), gl.DYNAMIC_DRAW);
            gl.enableVertexAttribArray(lineAttrs.color);
            gl.vertexAttribPointer(lineAttrs.color, 4, gl.FLOAT, false, 0, 0);

            gl.drawArrays(gl.LINES, 0, linePositions.length / 3);
          }
        }

        // ----------------- FPS & UI Update -----------------
        frameCount++;
        const now = performance.now();
        if (now - lastFpsUpdate >= 250) {
          currentFPS = Math.round((frameCount * 1000) / (now - lastFpsUpdate));
          frameCount = 0;
          lastFpsUpdate = now;

          document.getElementById('statFPS').textContent = currentFPS;
          document.getElementById('statMaxV').textContent = maxVelocityRecorded.toFixed(1);
          document.getElementById('statEkin').textContent = totalEkin.toFixed(0);
          document.getElementById('statEpot').textContent = totalEpot.toFixed(0);
        }

        requestAnimationFrame(render);
      }

      // -------------------------------------------------------------
      // 6. HUD Controls & Listeners Setup
      // -------------------------------------------------------------
      const presetSelect = document.getElementById('presetSelect');
      const sliderG = document.getElementById('sliderG');
      const sliderEps = document.getElementById('sliderEps');
      const sliderDT = document.getElementById('sliderDT');
      const sliderTrail = document.getElementById('sliderTrail');
      const btnPause = document.getElementById('btnPause');
      const btnReset = document.getElementById('btnReset');

      sliderG.addEventListener('input', (e) => {
        G = parseFloat(e.target.value);
        document.getElementById('valG').textContent = G.toFixed(1);
      });

      sliderEps.addEventListener('input', (e) => {
        const eps = parseFloat(e.target.value);
        softeningSq = eps * eps;
        document.getElementById('valEps').textContent = eps.toFixed(2);
      });

      sliderDT.addEventListener('input', (e) => {
        dt = parseFloat(e.target.value);
        document.getElementById('valDT').textContent = dt.toFixed(3);
      });

      sliderTrail.addEventListener('input', (e) => {
        MAX_TRAIL_LEN = parseInt(e.target.value, 10);
        document.getElementById('valTrail').textContent = MAX_TRAIL_LEN;
        // Trim trail history
        for (let i = 0; i < NUM_PARTICLES; i++) {
          if (MAX_TRAIL_LEN === 0) trailHistory[i] = [];
          else if (trailHistory[i].length > MAX_TRAIL_LEN * 3) {
            trailHistory[i].splice(0, trailHistory[i].length - MAX_TRAIL_LEN * 3);
          }
        }
      });

      presetSelect.addEventListener('change', (e) => {
        initPreset(e.target.value);
      });

      btnReset.addEventListener('click', () => {
        initPreset(presetSelect.value);
      });

      btnPause.addEventListener('click', () => {
        isPaused = !isPaused;
        btnPause.textContent = isPaused ? 'Resume' : 'Pause';
      });

      // Start initial simulation state
      initPreset('galaxy');
      requestAnimationFrame(render);

    } catch (err) {
      const errBox = document.getElementById('error-overlay');
      if (errBox) {
        errBox.style.display = 'block';
        errBox.textContent = 'Simulation Error:\n' + (err.stack || err.message || err);
      }
      console.error(err);
    }
  </script>
</body>
</html>
