      function perspectiveMat4(out, fovy, aspect, near, far) {
        const f = 1.0 / Math.tan(fovy / 2);
        const nf = 1.0 / (near - far);
        out.fill(0);
        out[0] = f / aspect;
        out[5] = f;
        out[10] = (far + near) * nf;
        out[11] = -1;
        out[14] = (2 * far * near) * nf;

      function lookAtMat4(out, eye, center, up) {
        let x0, x1, x2, y0, y1, y2, z0, z1, z2, len;
        let eyex = eye[0], eyey = eye[1], eyez = eye[2];
        let upx = up[0], upy = up[1], upz = up[2];
        let centerx = center[0], centery = center[1], centerz = center[2];

        z0 = eyex - centerx; z1 = eyey - centery; z2 = eyez - centerz;
        len = 1 / Math.hypot(z0, z1, z2);
        z0 *= len; z1 *= len; z2 *= len;

        x0 = upy * z2 - upz * z1;
        x1 = upz * z0 - upx * z2;
        x2 = upx * z1 - upy * z0;
        len = Math.hypot(x0, x1, x2);
        if (!len) { x0 = 0; x1 = 0; x2 = 0; } else { len = 1 / len; x0 *= len; x1 *= len; x2 *= len; }

        y0 = z1 * x2 - z2 * x1;
        y1 = z2 * x0 - z0 * x2;
        y2 = z0 * x1 - z1 * x0;
        len = Math.hypot(y0, y1, y2);
        if (!len) { y0 = 0; y1 = 0; y2 = 0; } else { len = 1 / len; y0 *= len; y1 *= len; y2 *= len; }

        out[0] = x0; out[1] = y0; out[2] = z0; out[3] = 0;
        out[4] = x1; out[5] = y1; out[6] = z1; out[7] = 0;
        out[8] = x2; out[9] = y2; out[10] = z2; out[11] = 0;
        out[12] = -(x0 * eyex + x1 * eyey + x2 * eyez);
        out[13] = -(y0 * eyex + y1 * eyey + y2 * eyez);
        out[14] = -(z0 * eyex + z1 * eyey + z2 * eyez);
        out[15] = 1;
      }

      // WebGL Shader Sources

        attribute vec3 aPosition;

      // Helper function to create WebGL shader program
      function createProgram(vsSource, fsSource) {
        const vs = gl.createShader(gl.VERTEX_SHADER);
        gl.shaderSource(vs, vsSource);
        gl.compileShader(vs);
        if (!gl.getShaderParameter(vs, gl.COMPILE_STATUS)) {
          throw new Error('Vertex shader compile error: ' + gl.getShaderInfoLog(vs));
        }

        const fs = gl.createShader(gl.FRAGMENT_SHADER);
        gl.shaderSource(fs, fsSource);
        gl.compileShader(fs);
        if (!gl.getShaderParameter(fs, gl.COMPILE_STATUS)) {
          throw new Error('Fragment shader compile error: ' + gl.getShaderInfoLog(fs));
        }

        const prog = gl.createProgram();
        gl.attachShader(prog, vs);
        gl.attachShader(prog, fs);
        gl.linkProgram(prog);
        if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) {
          throw new Error('Program link error: ' + gl.getProgramInfoLog(prog));
        }
        return prog;
      }

      const pointProgram = createProgram(pointVS, pointFS);
      const trailProgram = createProgram(trailVS, trailFS);

      // Preset Initializers
      function initParticles(type) {
        historyHead = 0;
        history.fill(0);

        for (let i = 0; i < N; i++) {
          const i3 = i * 3;
          
          if (type === 'galaxy') {
            // Spiral disk with heavy core
            if (i === 0) {
              pos[i3] = 0; pos[i3+1] = 0; pos[i3+2] = 0;
              vel[i3] = 0; vel[i3+1] = 0; vel[i3+2] = 0;
              mass[i] = 250.0;
            } else {
              const r = 4.0 + Math.random() * 28.0;
              const angle = Math.random() * Math.PI * 2;
              const height = (Math.random() - 0.5) * (r * 0.15);
              pos[i3] = r * Math.cos(angle);
              pos[i3+1] = height;
              pos[i3+2] = r * Math.sin(angle);

              // Circular orbital velocity v = sqrt(G*M/r)
              const vOrb = Math.sqrt((1.0 * 250.0) / r) * (0.9 + Math.random() * 0.2);
              vel[i3] = -vOrb * Math.sin(angle);
              vel[i3+1] = (Math.random() - 0.5) * 0.5;
              vel[i3+2] = vOrb * Math.cos(angle);
              mass[i] = 0.5 + Math.random() * 1.5;
            }
          } else if (type === 'clusters') {
            // Two colliding star clusters
            const cluster = i < N / 2 ? 0 : 1;
            const cx = cluster === 0 ? -16.0 : 16.0;
            const cy = cluster === 0 ? -5.0 : 5.0;
            const cz = 0;

            const r = Math.pow(Math.random(), 1.5) * 10.0;
            const theta = Math.random() * Math.PI * 2;
            const phi = Math.acos(2 * Math.random() - 1);

            pos[i3] = cx + r * Math.sin(phi) * Math.cos(theta);
            pos[i3+1] = cy + r * Math.sin(phi) * Math.sin(theta);
            pos[i3+2] = cz + r * Math.cos(phi);

            const vx = cluster === 0 ? 1.8 : -1.8;
            const vy = cluster === 0 ? 0.6 : -0.6;
            vel[i3] = vx + (Math.random() - 0.5) * 0.8;
            vel[i3+1] = vy + (Math.random() - 0.5) * 0.8;
            vel[i3+2] = (Math.random() - 0.5) * 0.8;
            mass[i] = 1.0 + Math.random() * 2.0;
          } else if (type === 'globular') {
            // Spherical Plummer-like distribution
            const r = Math.tan(Math.random() * Math.PI * 0.45) * 6.0;
            const theta = Math.random() * Math.PI * 2;
            const phi = Math.acos(2 * Math.random() - 1);

            pos[i3] = r * Math.sin(phi) * Math.cos(theta);
            pos[i3+1] = r * Math.sin(phi) * Math.sin(theta);
            pos[i3+2] = r * Math.cos(phi);

            const v = Math.sqrt((G * 100.0) / (r + 4.0)) * (Math.random() * 0.8);
            const vAngle = Math.random() * Math.PI * 2;
            vel[i3] = v * Math.cos(vAngle);
            vel[i3+1] = v * Math.sin(vAngle);
            vel[i3+2] = (Math.random() - 0.5) * v;
            mass[i] = 1.0 + Math.random() * 3.0;
          } else if (type === 'binary') {
            // Heavy binary pair orbiting each other with surrounding debris
            if (i === 0 || i === 1) {
              const sign = i === 0 ? 1 : -1;
              pos[i3] = sign * 6.0; pos[i3+1] = 0; pos[i3+2] = 0;
              vel[i3] = 0; vel[i3+1] = sign * 2.2; vel[i3+2] = 0;
              mass[i] = 120.0;
            } else {

              const angle = Math.random() * Math.PI * 2;
              pos[i3] = r * Math.cos(angle);
              pos[i3+1] = (Math.random() - 0.5) * 4.0;
              pos[i3+2] = r * Math.sin(angle);

              const v = Math.sqrt((G * 240.0) / r);
              vel[i3] = -v * Math.sin(angle);
              vel[i3+1] = (Math.random() - 0.5) * 0.2;
              vel[i3+2] = v * Math.cos(angle);
              mass[i] = 0.8 + Math.random() * 1.5;
            }
          } else {
            // Random Chaos Cloud
            pos[i3] = (Math.random() - 0.5) * 35.0;
            pos[i3+1] = (Math.random() - 0.5) * 35.0;
            pos[i3+2] = (Math.random() - 0.5) * 35.0;
            vel[i3] = (Math.random() - 0.5) * 1.0;
            vel[i3+1] = (Math.random() - 0.5) * 1.0;
            vel[i3+2] = (Math.random() - 0.5) * 1.0;
            mass[i] = 0.5 + Math.random() * 3.5;
          }

          // Prime history
          for (let h = 0; h < maxHistory; h++) {
            const hIdx = (h * N + i) * 3;
            history[hIdx] = pos[i3];
            history[hIdx+1] = pos[i3+1];
            history[hIdx+2] = pos[i3+2];
          }
        }
      }

      // Physics Integration (Velocity-Verlet / Softened Gravity)
      function computeAccelerations() {
        acc.fill(0);
        const eps2 = softening * softening;

        for (let i = 0; i < N; i++) {
          const i3 = i * 3;
          const px = pos[i3], py = pos[i3+1], pz = pos[i3+2];
          let ax = 0, ay = 0, az = 0;

          for (let j = 0; j < N; j++) {
            if (i === j) continue;
            const j3 = j * 3;
            const dx = pos[j3] - px;
            const dy = pos[j3+1] - py;
            const dz = pos[j3+2] - pz;

            const distSq = dx * dx + dy * dy + dz * dz + eps2;
            const invDist = 1.0 / Math.sqrt(distSq);
            const invDist3 = invDist * invDist * invDist;

            const f = G * mass[j] * invDist3;
            ax += dx * f;
            ay += dy * f;
            az += dz * f;
          }

          acc[i3] = ax;
          acc[i3+1] = ay;
          acc[i3+2] = az;
        }
      }

      let dynamicMaxSpeed = 15.0;
      let avgSpeedStat = 0.0;

      function updatePhysics(dt) {
        if (isPaused) return;

        // Sub-stepping for numerical stability
        const subSteps = 4;
        const subDt = (dt * simSpeed) / subSteps;

        for (let step = 0; step < subSteps; step++) {
          // Half-step velocity & full-step position
          for (let i = 0; i < N * 3; i++) {
            vel[i] += acc[i] * 0.5 * subDt;
            pos[i] += vel[i] * subDt;
          }

          computeAccelerations();

          // Second half-step velocity
          for (let i = 0; i < N * 3; i++) {
            vel[i] += acc[i] * 0.5 * subDt;
          }
        }

        // Compute Center of Mass and adjust drift
        let cx = 0, cy = 0, cz = 0, totalM = 0;
        let sumSpeed = 0;
        let currentMaxV = 0.1;

        for (let i = 0; i < N; i++) {
          const i3 = i * 3;
          const m = mass[i];
          cx += pos[i3] * m;
          cy += pos[i3+1] * m;
          cz += pos[i3+2] * m;
          totalM += m;

          const vx = vel[i3], vy = vel[i3+1], vz = vel[i3+2];
          const spd = Math.hypot(vx, vy, vz);
          speed[i] = spd;
          sumSpeed += spd;
          if (spd > currentMaxV) currentMaxV = spd;
        }

        cx /= totalM; cy /= totalM; cz /= totalM;
        // Gently drift center back to origin
        for (let i = 0; i < N; i++) {
          const i3 = i * 3;
          pos[i3] -= cx * 0.02;
          pos[i3+1] -= cy * 0.02;
          pos[i3+2] -= cz * 0.02;
        }

        avgSpeedStat = sumSpeed / N;
        // Smoothly adapt color map scale
        dynamicMaxSpeed += (currentMaxV - dynamicMaxSpeed) * 0.05;

        // Record history frame for trails
        historyHead = (historyHead + 1) % maxHistory;
        for (let i = 0; i < N; i++) {
          const i3 = i * 3;
          const hIdx = (historyHead * N + i) * 3;
          history[hIdx] = pos[i3];
          history[hIdx+1] = pos[i3+1];
          history[hIdx+2] = pos[i3+2];
        }
      }

      // Convert speed to RGB vector (for trail colors)
      function getSpeedColor(vNorm) {
        const t = Math.max(0, Math.min(1, vNorm));
        if (t < 0.2) return [0.11 + t*5.0*0.09, 0.31 + t*5.0*0.40, 0.85 - t*5.0*0.02];
        if (t < 0.4) return [0.02 + (t-0.2)*5.0*0.04, 0.71 + (t-0.2)*5.0*0.02, 0.83 - (t-0.2)*5.0*0.32];
        if (t < 0.6) return [0.06 + (t-0.4)*5.0*0.92, 0.73 + (t-0.4)*5.0*0.07, 0.51 - (t-0.4)*5.0*0.43];
        if (t < 0.8) return [0.98 - (t-0.6)*5.0*0.04, 0.80 - (t-0.6)*5.0*0.53, 0.08 + (t-0.6)*5.0*0.19];
        return [0.94 + (t-0.8)*5.0*0.06, 0.27 + (t-0.8)*5.0*0.73, 0.27 + (t-0.8)*5.0*0.73];
      }

      // Prepare Orbital Trail Geometry Data
      function buildTrailGeometry() {
        if (trailLength <= 0) return 0;

        let vIdx = 0;
        let aIdx = 0;
        let cIdx = 0;
        let vertCount = 0;

        for (let i = 0; i < N; i++) {
          const spdNorm = speed[i] / Math.max(dynamicMaxSpeed, 0.001);
          const col = getSpeedColor(spdNorm);

          for (let h = 0; h < trailLength - 1; h++) {
            const h1 = (historyHead - h + maxHistory) % maxHistory;
            const h2 = (historyHead - h - 1 + maxHistory) % maxHistory;

            const idx1 = (h1 * N + i) * 3;
            const idx2 = (h2 * N + i) * 3;

            trailPosBufData[vIdx++] = history[idx1];
            trailPosBufData[vIdx++] = history[idx1+1];
            trailPosBufData[vIdx++] = history[idx1+2];

            trailPosBufData[vIdx++] = history[idx2];
            trailPosBufData[vIdx++] = history[idx2+1];
            trailPosBufData[vIdx++] = history[idx2+2];

            const alpha1 = 1.0 - (h / trailLength);
            const alpha2 = 1.0 - ((h + 1) / trailLength);

            trailAlphaBufData[aIdx++] = alpha1;
            trailAlphaBufData[aIdx++] = alpha2;

            trailColorBufData[cIdx++] = col[0]; trailColorBufData[cIdx++] = col[1]; trailColorBufData[cIdx++] = col[2];
            trailColorBufData[cIdx++] = col[0]; trailColorBufData[cIdx++] = col[1]; trailColorBufData[cIdx++] = col[2];

            vertCount += 2;
          }
        }

        gl.bindBuffer(gl.ARRAY_BUFFER, trailPosBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, trailPosBufData.subarray(0, vIdx), gl.DYNAMIC_DRAW);

        gl.bindBuffer(gl.ARRAY_BUFFER, trailAlphaBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, trailAlphaBufData.subarray(0, aIdx), gl.DYNAMIC_DRAW);

        gl.bindBuffer(gl.ARRAY_BUFFER, trailColorBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, trailColorBufData.subarray(0, cIdx), gl.DYNAMIC_DRAW);

        return vertCount;
      }

      // Render Loop
      function render() {
        // Update matrices
        if (autoRotate && !isDragging) {
          cameraTheta += 0.002;
        }

        const eyeX = cameraDist * Math.sin(cameraPhi) * Math.cos(cameraTheta);
        const eyeY = cameraDist * Math.cos(cameraPhi);
        const eyeZ = cameraDist * Math.sin(cameraPhi) * Math.sin(cameraTheta);

        const aspect = gl.canvas.clientWidth / gl.canvas.clientHeight;
        perspectiveMat4(projMatrix, Math.PI / 4, aspect, 0.1, 1000.0);
        lookAtMat4(viewMatrix, [eyeX, eyeY, eyeZ], [0, 0, 0], [0, 1, 0]);

        // Clear Screen
        gl.viewport(0, 0, gl.canvas.width, gl.canvas.height);
        gl.clearColor(0.02, 0.02, 0.05, 1.0);
        gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

        gl.enable(gl.BLEND);
        gl.blendFunc(gl.SRC_ALPHA, gl.ONE); // Additive glowing blend
        gl.disable(gl.DEPTH_TEST);

        // 1. Draw Particle Trails
        const trailVerts = buildTrailGeometry();
        if (trailVerts > 0) {
          gl.useProgram(trailProgram);
          gl.uniformMatrix4fv(trailUniforms.projection, false, projMatrix);
          gl.uniformMatrix4fv(trailUniforms.view, false, viewMatrix);

          gl.bindBuffer(gl.ARRAY_BUFFER, trailPosBuffer);
          gl.enableVertexAttribArray(trailAttribs.position);
          gl.vertexAttribPointer(trailAttribs.position, 3, gl.FLOAT, false, 0, 0);

          gl.bindBuffer(gl.ARRAY_BUFFER, trailAlphaBuffer);
          gl.enableVertexAttribArray(trailAttribs.alpha);
          gl.vertexAttribPointer(trailAttribs.alpha, 1, gl.FLOAT, false, 0, 0);

          gl.bindBuffer(gl.ARRAY_BUFFER, trailColorBuffer);
          gl.enableVertexAttribArray(trailAttribs.color);
          gl.vertexAttribPointer(trailAttribs.color, 3, gl.FLOAT, false, 0, 0);

          gl.drawArrays(gl.LINES, 0, trailVerts);
        }

        // 2. Draw Gravitating Point Particles
        gl.useProgram(pointProgram);
        gl.uniformMatrix4fv(pointUniforms.projection, false, projMatrix);
        gl.uniformMatrix4fv(pointUniforms.view, false, viewMatrix);
        gl.uniform1f(pointUniforms.maxSpeed, dynamicMaxSpeed);
        gl.uniform1f(pointUniforms.pointSize, window.innerHeight < 500 ? 12.0 : 18.0);

        gl.bindBuffer(gl.ARRAY_BUFFER, posBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, pos, gl.DYNAMIC_DRAW);
        gl.enableVertexAttribArray(pointAttribs.position);
        gl.vertexAttribPointer(pointAttribs.position, 3, gl.FLOAT, false, 0, 0);

        gl.bindBuffer(gl.ARRAY_BUFFER, speedBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, speed, gl.DYNAMIC_DRAW);
        gl.enableVertexAttribArray(pointAttribs.speed);
        gl.vertexAttribPointer(pointAttribs.speed, 1, gl.FLOAT, false, 0, 0);

        gl.drawArrays(gl.POINTS, 0, N);
      }

      // Performance & Stats Tracking
      let lastTime = performance.now();
      let frameCount = 0;
      let fps = 60;

      function frame(now) {
        const dt = Math.min((now - lastTime) / 1000, 0.1);
        lastTime = now;

        frameCount++;
        if (frameCount % 15 === 0) {
          fps = Math.round(1 / dt);
          document.getElementById('stat-fps').textContent = fps;
          document.getElementById('stat-vavg').textContent = avgSpeedStat.toFixed(1);
          document.getElementById('stat-vmax').textContent = dynamicMaxSpeed.toFixed(1);
          document.getElementById('val-vmax').textContent = dynamicMaxSpeed.toFixed(1);
        }

        updatePhysics(0.016);
        render();
        requestAnimationFrame(frame);
      }

      // Handle Resize
      function onResize() {
        const dpr = Math.min(window.devicePixelRatio || 1, 2);
        const displayWidth = Math.floor(canvas.clientWidth * dpr);
        const displayHeight = Math.floor(canvas.clientHeight * dpr);

        if (canvas.width !== displayWidth || canvas.height !== displayHeight) {
          canvas.width = displayWidth;
          canvas.height = displayHeight;
        }
      }
      window.addEventListener('resize', onResize);
      onResize();

      // Camera Interaction Handlers
      canvas.addEventListener('mousedown', (e) => {
        isDragging = true;
        autoRotate = false;
        prevMouseX = e.clientX;
        prevMouseY = e.clientY;
      });

      window.addEventListener('mouseup', () => { isDragging = false; });

      window.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        const dx = e.clientX - prevMouseX;
        const dy = e.clientY - prevMouseY;
        prevMouseX = e.clientX;
        prevMouseY = e.clientY;

        cameraTheta -= dx * 0.005;
        cameraPhi = Math.max(0.05, Math.min(Math.PI - 0.05, cameraPhi - dy * 0.005));
      });

      canvas.addEventListener('wheel', (e) => {
        e.preventDefault();
        cameraDist = Math.max(10.0, Math.min(250.0, cameraDist + e.deltaY * 0.05));
      }, { passive: false });

      // Touch Controls for Mobile
      let touchStartDist = 0;
      canvas.addEventListener('touchstart', (e) => {
        if (e.touches.length === 1) {
          isDragging = true;
          autoRotate = false;
          prevMouseX = e.touches[0].clientX;
          prevMouseY = e.touches[0].clientY;
        } else if (e.touches.length === 2) {
          touchStartDist = Math.hypot(
            e.touches[0].clientX - e.touches[1].clientX,
            e.touches[0].clientY - e.touches[1].clientY
          );
        }
      });

      canvas.addEventListener('touchmove', (e) => {
        if (e.touches.length === 1 && isDragging) {
          const dx = e.touches[0].clientX - prevMouseX;
          const dy = e.touches[0].clientY - prevMouseY;
          prevMouseX = e.touches[0].clientX;
          prevMouseY = e.touches[0].clientY;

          cameraTheta -= dx * 0.005;
          cameraPhi = Math.max(0.05, Math.min(Math.PI - 0.05, cameraPhi - dy * 0.005));
        } else if (e.touches.length === 2) {
          const dist = Math.hypot(
            e.touches[0].clientX - e.touches[1].clientX,
            e.touches[0].clientY - e.touches[1].clientY
          );
          const factor = touchStartDist / dist;
          cameraDist = Math.max(10.0, Math.min(250.0, cameraDist * (1 + (factor - 1) * 0.1)));
          touchStartDist = dist;
        }
      });

      canvas.addEventListener('touchend', () => { isDragging = false; });

      // UI Controls Binding
      const uiContainer = document.getElementById('ui-container');
      const uiHeader = document.getElementById('ui-header');
      const collapseBtn = document.getElementById('collapse-btn');

      uiHeader.addEventListener('click', () => {
        uiContainer.classList.toggle('collapsed');
        collapseBtn.textContent = uiContainer.classList.contains('collapsed') ? '▲' : '▼';
      });

      document.getElementById('slider-softening').addEventListener('input', (e) => {
        softening = parseFloat(e.target.value);
        document.getElementById('val-softening').textContent = softening.toFixed(2);
      });

      document.getElementById('slider-gravity').addEventListener('input', (e) => {
        G = parseFloat(e.target.value);
        document.getElementById('val-gravity').textContent = G.toFixed(2);
      });

      document.getElementById('slider-speed').addEventListener('input', (e) => {
        simSpeed = parseFloat(e.target.value);
        document.getElementById('val-speed').textContent = simSpeed.toFixed(1) + 'x';
      });

      document.getElementById('slider-trails').addEventListener('input', (e) => {
        trailLength = parseInt(e.target.value, 10);
        document.getElementById('val-trails').textContent = trailLength;
      });

      document.getElementById('preset-select').addEventListener('change', (e) => {
        currentPreset = e.target.value;
        initParticles(currentPreset);
        computeAccelerations();
      });

      const pauseBtn = document.getElementById('btn-pause');
      pauseBtn.addEventListener('click', () => {
        isPaused = !isPaused;
        pauseBtn.textContent = isPaused ? 'Resume' : 'Pause';
        pauseBtn.style.background = isPaused ? '#0284c7' : '#1e293b';
      });

      document.getElementById('btn-reset').addEventListener('click', () => {
        initParticles(currentPreset);
        computeAccelerations();
      });

      // Initialization
      initParticles(currentPreset);
      computeAccelerations();
      requestAnimationFrame(frame);

    } catch (err) {
      const errorBanner = document.getElementById('error-banner');
      errorBanner.style.display = 'block';
      errorBanner.textContent = 'Application Error: ' + err.message + '\n' + (err.stack || '');
      console.error(err);
    }
  </script>
