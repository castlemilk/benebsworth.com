  // --- Matrix Math Helpers ---
  function mat4Create() {
    return new Float32Array(16);
  }

  function mat4Perspective(out, fovy, aspect, near, far) {
    const f = 1.0 / Math.tan(fovy / 2);
    out.fill(0);
    out[0] = f / aspect;
    out[5] = f;
    out[10] = (far + near) / (near - far);
    out[11] = -1;
    out[14] = (2 * far * near) / (near - far);
  }

  function mat4LookAt(out, eye, center, up) {
    let x0, x1, x2, y0, y1, y2, z0, z1, z2, len;
    let eyex = eye[0], eyey = eye[1], eyez = eye[2];
    let upx = up[0], upy = up[1], upz = up[2];
    let centerx = center[0], centery = center[1], centerz = center[2];

    z0 = eyex - centerx; z1 = eyey - centery; z2 = eyez - centerz;
    len = 1 / Math.hypot(z0, z1, z2);
    z0 *= len; z1 *= len; z2 *= len;

    x0 = upy * z2 - upz * z1;
    x1 = upz * z0 - upx * z2;
    x2 = upx * z1 - upy * z0;
    len = Math.hypot(x0, x1, x2);
    if (!len) { x0 = 0; x1 = 0; x2 = 0; }
    else { len = 1 / len; x0 *= len; x1 *= len; x2 *= len; }

    y0 = z1 * x2 - z2 * x1;
    y1 = z2 * x0 - z0 * x2;
    y2 = z0 * x1 - z1 * x0;
    len = Math.hypot(y0, y1, y2);
    if (!len) { y0 = 0; y1 = 0; y2 = 0; }
    else { len = 1 / len; y0 *= len; y1 *= len; y2 *= len; }

    out[0] = x0; out[1] = y0; out[2] = z0; out[3] = 0;
    out[4] = x1; out[5] = y1; out[6] = z1; out[7] = 0;
    out[8] = x2; out[9] = y2; out[10] = z2; out[11] = 0;
    out[12] = -(x0 * eyex + x1 * eyey + x2 * eyez);
    out[13] = -(y0 * eyex + y1 * eyey + y2 * eyez);
    out[14] = -(z0 * eyex + z1 * eyey + z2 * eyez);
    out[15] = 1;
  }

  function mat4Multiply(out, a, b) {
    let a00 = a[0], a01 = a[1], a02 = a[2], a03 = a[3];
    let a10 = a[4], a11 = a[5], a12 = a[6], a13 = a[7];

    out[1] = b0*a01 + b1*a11 + b2*a21 + b3*a31;
    out[2] = b0*a02 + b1*a12 + b2*a22 + b3*a23;

  }

      vec3 c5 = vec3(1.00, 1.00, 1.00); // White

      if (t < 0.4) return mix(c1, c2, (t - 0.2) * 5.0);

  // --- Physics & Simulation Setup ---

  function initPreset(preset) {
    // Reset arrays
    px.fill(0); py.fill(0); pz.fill(0);
    vx.fill(0); vy.fill(0); vz.fill(0);
    mass.fill(1.0);

    if (preset === 'galaxy') {
      // Central supermassive body
      mass[0] = 300.0;
      px[0] = 0; py[0] = 0; pz[0] = 0;
      vx[0] = 0; vy[0] = 0; vz[0] = 0;

      for (let i = 1; i < N; i++) {
        const r = 4.0 + Math.random() * 28.0;
        const theta = Math.random() * Math.PI * 2;
        const phi = (Math.random() - 0.5) * 0.2; // slightly flattened disk

        px[i] = r * Math.cos(theta);
        py[i] = r * Math.sin(phi);
        pz[i] = r * Math.sin(theta);

        // Circular orbit velocity around center: v = sqrt(G * M / r)
        const vOrbit = Math.sqrt((G * mass[0]) / r) * (0.9 + Math.random() * 0.2);
        vx[i] = -vOrbit * Math.sin(theta);
        vy[i] = (Math.random() - 0.5) * 0.5;
        vz[i] = vOrbit * Math.cos(theta);

        mass[i] = 0.5 + Math.random() * 1.5;
      }
    } else if (preset === 'binary') {
      // Two central heavy bodies orbiting each other
      mass[0] = 150.0; mass[1] = 150.0;
      const d = 10.0;
      px[0] = -d; py[0] = 0; pz[0] = 0;
      px[1] =  d; py[1] = 0; pz[1] = 0;

      const vBin = Math.sqrt((G * 150.0) / (4 * d));
      vy[0] =  vBin;
      vy[1] = -vBin;

        px[i] = r * Math.cos(a);
        py[i] = (Math.random() - 0.5) * 6.0;
        pz[i] = r * Math.sin(a);

        const v = Math.sqrt((G * 300.0) / r);
        vx[i] = -v * Math.sin(a);
        vy[i] += (Math.random() - 0.5) * 1.0;
        vz[i] = v * Math.cos(a);
        mass[i] = 0.5 + Math.random();
      }
    } else if (preset === 'collapse') {

        const phi = Math.acos(2 * Math.random() - 1);

        px[i] = r * Math.sin(phi) * Math.cos(theta);
        py[i] = r * Math.sin(phi) * Math.sin(theta);
        pz[i] = r * Math.cos(phi);

        vx[i] = (Math.random() - 0.5) * 0.2;
        vy[i] = (Math.random() - 0.5) * 0.2;

    } else if (preset === 'collision') {

        vy[i] = -group * 0.6 + (Math.random() - 0.5) * 0.5;
        vz[i] = (Math.random() - 0.5) * 0.5;
        mass[i] = 1.0 + Math.random() * 1.5;
      }
    }

  function stepPhysics() {
    const eps2 = softening * softening;
    const ax = new Float32Array(N);
    const ay = new Float32Array(N);
    const az = new Float32Array(N);

    // Pairwise gravity computation - N^2 = 40,000 operations (very fast in JS)
    for (let i = 0; i < N; i++) {
      const xi = px[i], yi = py[i], zi = pz[i];
      let axi = 0, ayi = 0, azi = 0;

      for (let j = 0; j < N; j++) {
        if (i === j) continue;
        const dx = px[j] - xi;
        const dy = py[j] - yi;
        const dz = pz[j] - zi;

        const distSq = dx * dx + dy * dy + dz * dz + eps2;
        const invDist3 = 1.0 / (distSq * Math.sqrt(distSq));
        const f = G * mass[j] * invDist3;

        axi += dx * f;
        ayi += dy * f;
        azi += dz * f;
      }

      ax[i] = axi;
      ay[i] = ayi;
      az[i] = azi;
    }

    // Symplectic Euler Integration
    let currentVMax = 0.1;
    for (let i = 0; i < N; i++) {
      vx[i] += ax[i] * dt;
      vy[i] += ay[i] * dt;
      vz[i] += az[i] * dt;

      px[i] += vx[i] * dt;
      py[i] += vy[i] * dt;
      pz[i] += vz[i] * dt;

      // Pack buffers
      posBufferData[i * 3]     = px[i];
      posBufferData[i * 3 + 1] = py[i];
      posBufferData[i * 3 + 2] = pz[i];

      velBufferData[i * 3]     = vx[i];
      velBufferData[i * 3 + 1] = vy[i];
      velBufferData[i * 3 + 2] = vz[i];

      const speed = Math.hypot(vx[i], vy[i], vz[i]);
      if (speed > currentVMax) currentVMax = speed;
    }

    // Smooth maximum speed auto-scaling for color mapping
    maxObservedSpeed = maxObservedSpeed * 0.95 + currentVMax * 0.05;
    return currentVMax;
  }

  sliderG.addEventListener('input', (e) => { G = parseFloat(e.target.value); valG.textContent = G.toFixed(1); });
  sliderEps.addEventListener('input', (e) => { softening = parseFloat(e.target.value); valEps.textContent = softening.toFixed(1); });
  sliderDt.addEventListener('input', (e) => { dt = parseFloat(e.target.value); valDt.textContent = dt.toFixed(3); });

  presetSelect.addEventListener('change', (e) => {
    initPreset(e.target.value);
  });

  btnReset.addEventListener('click', () => {
    initPreset(presetSelect.value);
  });

  btnPause.addEventListener('click', () => {
    isPaused = !isPaused;
    btnPause.textContent = isPaused ? 'Resume' : 'Pause';
    btnPause.style.background = isPaused ? '#10b981' : '#1e2230';
  });

  let isHudCollapsed = false;
  document.getElementById('hud-header').addEventListener('click', () => {
    isHudCollapsed = !isHudCollapsed;
    hud.classList.toggle('collapsed', isHudCollapsed);
    toggleHudBtn.textContent = isHudCollapsed ? 'Expand' : 'Collapse';
  });

  // --- Mouse & Touch Orbit Controls ---
  let isDragging = false;
  let lastMouseX = 0, lastMouseY = 0;

  function onPointerDown(x, y) {
    isDragging = true;
    lastMouseX = x;
    lastMouseY = y;
  }

  function onPointerMove(x, y) {
    if (!isDragging) return;
    const dx = x - lastMouseX;
    const dy = y - lastMouseY;

    camAzimuth -= dx * 0.008;
    camElevation += dy * 0.008;

    // Clamp elevation to avoid gimbal flipping
    camElevation = Math.max(-Math.PI / 2 + 0.05, Math.min(Math.PI / 2 - 0.05, camElevation));

    lastMouseX = x;
    lastMouseY = y;
  }

  function onPointerUp() {
    isDragging = false;
  }

  canvas.addEventListener('mousedown', (e) => onPointerDown(e.clientX, e.clientY));
  window.addEventListener('mousemove', (e) => onPointerMove(e.clientX, e.clientY));
  window.addEventListener('mouseup', onPointerUp);

  canvas.addEventListener('touchstart', (e) => {
    if (e.touches.length === 1) {
      onPointerDown(e.touches[0].clientX, e.touches[0].clientY);
    }
  }, { passive: true });

  canvas.addEventListener('touchmove', (e) => {
    if (e.touches.length === 1) {
      onPointerMove(e.touches[0].clientX, e.touches[0].clientY);
    }
  }, { passive: true });

  canvas.addEventListener('touchend', onPointerUp);

  canvas.addEventListener('wheel', (e) => {
    e.preventDefault();
    camDistance += e.deltaY * 0.05;
    camDistance = Math.max(10.0, Math.min(250.0, camDistance));
  }, { passive: false });

  // --- Window Resize & Canvas Store Handling ---
  function resizeCanvas() {
    const width = window.innerWidth;
    const height = window.innerHeight;
    const dpr = Math.min(window.devicePixelRatio || 1, 2);

    canvas.width = Math.floor(width * dpr);
    canvas.height = Math.floor(height * dpr);
    gl.viewport(0, 0, canvas.width, canvas.height);
  }

  window.addEventListener('resize', resizeCanvas);
  resizeCanvas();

  // Initialize particles
  initPreset('galaxy');

  // --- Render Loop ---
  let lastTime = performance.now();
  let frameCount = 0;
  let lastFpsUpdate = performance.now();

  function animate(now) {
    // FPS Calculation
    frameCount++;
    if (now - lastFpsUpdate >= 500) {
      const fps = Math.round((frameCount * 1000) / (now - lastFpsUpdate));
      statFps.textContent = `FPS: ${fps}`;
      frameCount = 0;
      lastFpsUpdate = now;
    }

    // Step physics if active
    let vMax = 0;
    if (!isPaused) {
      // Substep for integration accuracy
      vMax = stepPhysics();
      stepPhysics();
    }
    statVmax.textContent = `Vmax: ${vMax.toFixed(1)}`;

    // Automatic smooth orbit rotation when idle
    if (!isDragging && !isPaused) {
      camAzimuth += 0.001;
    }

    // Clear Screen
    gl.clearColor(0.03, 0.03, 0.05, 1.0);
    gl.clear(gl.COLOR_BUFFER_BIT);

    // Update Camera Matrices
    const eyeX = camTarget[0] + camDistance * Math.cos(camElevation) * Math.sin(camAzimuth);
    const eyeY = camTarget[1] + camDistance * Math.sin(camElevation);
    const eyeZ = camTarget[2] + camDistance * Math.cos(camElevation) * Math.cos(camAzimuth);

    const aspect = canvas.width / canvas.height;
    mat4Perspective(projMat, Math.PI / 4, aspect, 0.5, 500.0);
    mat4LookAt(viewMat, [eyeX, eyeY, eyeZ], camTarget, [0, 1, 0]);
    mat4Multiply(mvpMat, projMat, viewMat);

    // Upload Uniforms
    gl.uniformMatrix4fv(uMvpLoc, false, mvpMat);
    gl.uniform1f(uPtScaleLoc, canvas.height * 0.6);
    gl.uniform1f(uMaxSpeedLoc, Math.max(maxObservedSpeed, 2.0));

    // Upload Position Buffer Data
    gl.bindBuffer(gl.ARRAY_BUFFER, posBuffer);
    gl.bufferData(gl.ARRAY_BUFFER, posBufferData, gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(aPosLoc);
    gl.vertexAttribPointer(aPosLoc, 3, gl.FLOAT, false, 0, 0);

    // Upload Velocity Buffer Data
    gl.bindBuffer(gl.ARRAY_BUFFER, velBuffer);
    gl.bufferData(gl.ARRAY_BUFFER, velBufferData, gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(aVelLoc);
    gl.vertexAttribPointer(aVelLoc, 3, gl.FLOAT, false, 0, 0);

    // Draw Particles
    gl.drawArrays(gl.POINTS, 0, N);

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);

} catch (err) {
  const errDiv = document.getElementById('error-display');
  if (errDiv) {
    errDiv.style.display = 'block';
    errDiv.textContent = "Simulation Error: " + err.message;
  }
  console.error(err);
}
