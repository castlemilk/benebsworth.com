<script>
let animFrameId = null;

window.addEventListener('unload', () => {
  if (animFrameId) cancelAnimationFrame(animFrameId);
});

try {
  // Initialize WebGL Context
  const canvas = document.getElementById('canvas');
  const gl = canvas.getContext('webgl', { antialias: true, alpha: false, preserveDrawingBuffer: false }) ||
             canvas.getContext('experimental-webgl');

  if (!gl) {
    throw new Error('WebGL is not supported by your browser or environment.');
  }

  // Simulation Parameters
  const params = {
    N: 200,
    G: 2.5,
    eps: 0.8,
    dt: 0.008,
    substeps: 4,
    colorMode: 0,
    pointSize: 14.0,
    preset: 'galaxy',
    paused: false,
    recenterCoM: true
  };

  // State Arrays (Max capacity 500)
  const MAX_N = 500;
  const px = new Float32Array(MAX_N);
  const py = new Float32Array(MAX_N);
  const pz = new Float32Array(MAX_N);
  const vx = new Float32Array(MAX_N);
  const vy = new Float32Array(MAX_N);
  const vz = new Float32Array(MAX_N);
  const ax = new Float32Array(MAX_N);
  const ay = new Float32Array(MAX_N);
  const az = new Float32Array(MAX_N);
  const mass = new Float32Array(MAX_N);
  const speed = new Float32Array(MAX_N);

  // Buffer arrays for GPU
  const posBufferData = new Float32Array(MAX_N * 3);

  // Camera State
  const camera = {
    theta: Math.PI * 0.25,
    phi: Math.PI * 0.15,
    radius: 75.0,
    target: [0, 0, 0]
  };

  // Matrix math helpers
  function mat4Create() { return new Float32Array(16); }
  
  function mat4Perspective(out, fovy, aspect, near, far) {
    const f = 1.0 / Math.tan(fovy / 2);
    const nf = 1 / (near - far);
    out.fill(0);
    out[0] = f / aspect;
    out[5] = f;
    out[10] = (far + near) * nf;
    out[11] = -1;
    out[14] = (2 * far * near) * nf;
    return out;
  }

  function mat4LookAt(out, eye, center, up) {
    let z0 = eye[0] - center[0], z1 = eye[1] - center[1], z2 = eye[2] - center[2];
    let len = 1 / (Math.hypot(z0, z1, z2) || 1);
    z0 *= len; z1 *= len; z2 *= len;

    let x0 = up[1] * z2 - up[2] * z1;
    let x1 = up[2] * z0 - up[0] * z2;
    let x2 = up[0] * z1 - up[1] * z0;
    len = 1 / (Math.hypot(x0, x1, x2) || 1);
    x0 *= len; x1 *= len; x2 *= len;

    let y0 = z1 * x2 - z2 * x1;
    let y1 = z2 * x0 - z0 * x2;
    let y2 = z0 * x1 - z1 * x0;

    out[0] = x0; out[1] = y0; out[2] = z0; out[3] = 0;
    out[4] = x1; out[5] = y1; out[6] = z1; out[7] = 0;
    out[8] = x2; out[9] = y2; out[10] = z2; out[11] = 0;
    out[12] = -(x0 * eye[0] + x1 * eye[1] + x2 * eye[2]);
    out[13] = -(y0 * eye[0] + y1 * eye[1] + y2 * eye[2]);
    out[14] = -(z0 * eye[0] + z1 * eye[1] + z2 * eye[2]);
    out[15] = 1;
    return out;
  }

  // Shaders
  const vsSource = `
    attribute vec3 aPosition;
    attribute float aVelocity;
    attribute float aMass;

    uniform mat4 uProjectionMatrix;
    uniform mat4 uViewMatrix;
    uniform float uMaxSpeed;
    uniform float uPointSize;
    uniform int uColorMode;

    varying vec4 vColor;

    vec3 colorThermal(float t) {
      if (t < 0.25) return mix(vec3(0.0, 0.1, 0.5), vec3(0.0, 0.7, 1.0), t * 4.0);
      if (t < 0.50) return mix(vec3(0.0, 0.7, 1.0), vec3(0.2, 0.9, 0.2), (t - 0.25) * 4.0);
      if (t < 0.75) return mix(vec3(0.2, 0.9, 0.2), vec3(1.0, 0.8, 0.0), (t - 0.50) * 4.0);
      return mix(vec3(1.0, 0.8, 0.0), vec3(1.0, 0.1, 0.1), (t - 0.75) * 4.0);
    }

    vec3 colorCosmic(float t) {
      if (t < 0.33) return mix(vec3(0.1, 0.2, 0.8), vec3(0.9, 0.1, 0.6), t * 3.0);
      if (t < 0.66) return mix(vec3(0.9, 0.1, 0.6), vec3(0.2, 1.0, 0.9), (t - 0.33) * 3.0);
      return mix(vec3(0.2, 1.0, 0.9), vec3(1.0, 1.0, 1.0), (t - 0.66) * 3.0);
    }

    vec3 colorFire(float t) {
      if (t < 0.33) return mix(vec3(0.3, 0.0, 0.1), vec3(0.9, 0.2, 0.0), t * 3.0);
      if (t < 0.66) return mix(vec3(0.9, 0.2, 0.0), vec3(1.0, 0.7, 0.0), (t - 0.33) * 3.0);
      return mix(vec3(1.0, 0.7, 0.0), vec3(1.0, 1.0, 0.95), (t - 0.66) * 3.0);
    }

    vec3 colorPlasma(float t) {
      if (t < 0.5) return mix(vec3(0.2, 0.0, 0.4), vec3(0.0, 0.5, 1.0), t * 2.0);
      return mix(vec3(0.0, 0.5, 1.0), vec3(0.9, 1.0, 1.0), (t - 0.5) * 2.0);
    }

    void main() {
      vec4 mvPosition = uViewMatrix * vec4(aPosition, 1.0);
      gl_Position = uProjectionMatrix * mvPosition;
      
      float sizeMult = pow(aMass, 0.33);
      gl_PointSize = uPointSize * sizeMult * (140.0 / -mvPosition.z);
      gl_PointSize = clamp(gl_PointSize, 2.0, 128.0);
      
      float normSpeed = clamp(aVelocity / max(uMaxSpeed, 0.001), 0.0, 1.0);
      
      vec3 col = vec3(1.0);
      if (uColorMode == 0) col = colorThermal(normSpeed);
      else if (uColorMode == 1) col = colorCosmic(normSpeed);
      else if (uColorMode == 2) col = colorFire(normSpeed);
      else col = colorPlasma(normSpeed);
      
      vColor = vec4(col, 1.0);
    }
  `;

  const fsSource = `
    precision mediump float;
    varying vec4 vColor;

    void main() {
      vec2 coord = gl_PointCoord - vec2(0.5);
      float distSq = dot(coord, coord);
      if (distSq > 0.25) discard;
      
      // Smooth core and soft glowing edge
      float alpha = exp(-distSq * 10.0) + 0.35 * exp(-distSq * 3.0);
      alpha = clamp(alpha, 0.0, 1.0);
      
      gl_FragColor = vec4(vColor.rgb, vColor.a * alpha);
    }
  `;

  function createShader(gl, type, source) {
    const shader = gl.createShader(type);
    gl.shaderSource(shader, source);
    gl.compileShader(shader);
    if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
      const err = gl.getShaderInfoLog(shader);
      gl.deleteShader(shader);
      throw new Error("Shader error: " + err);
    }
    return shader;
  }

  const program = gl.createProgram();
  const vs = createShader(gl, gl.VERTEX_SHADER, vsSource);
  const fs = createShader(gl, gl.FRAGMENT_SHADER, fsSource);
  gl.attachShader(program, vs);
  gl.attachShader(program, fs);
  gl.linkProgram(program);

  if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
    throw new Error("Program link error: " + gl.getProgramInfoLog(program));
  }

  // Location lookups
  const locs = {
    aPosition: gl.getAttribLocation(program, 'aPosition'),
    aVelocity: gl.getAttribLocation(program, 'aVelocity'),
    aMass: gl.getAttribLocation(program, 'aMass'),
    uProjectionMatrix: gl.getUniformLocation(program, 'uProjectionMatrix'),
    uViewMatrix: gl.getUniformLocation(program, 'uViewMatrix'),
    uMaxSpeed: gl.getUniformLocation(program, 'uMaxSpeed'),
    uPointSize: gl.getUniformLocation(program, 'uPointSize'),
    uColorMode: gl.getUniformLocation(program, 'uColorMode')
  };

  // VBOs
  const posVBO = gl.createBuffer();
  const velVBO = gl.createBuffer();
  const massVBO = gl.createBuffer();

  // Preset Generators
  function initPreset(presetType) {
    const N = params.N;
    px.fill(0); py.fill(0); pz.fill(0);
    vx.fill(0); vy.fill(0); vz.fill(0);
    mass.fill(1.0);

    if (presetType === 'galaxy') {
      // Rotating Galaxy Disk with Central Massive BH
      mass[0] = 350.0;
      px[0] = 0; py[0] = 0; pz[0] = 0;
      vx[0] = 0; vy[0] = 0; vz[0] = 0;

      for (let i = 1; i < N; i++) {
        const r = 5.0 + Math.pow(Math.random(), 0.75) * 35.0;
        const theta = Math.random() * Math.PI * 2.0;
        const z = (Math.random() - 0.5) * (1.2 + r * 0.04);
        
        px[i] = r * Math.cos(theta);
        py[i] = z;
        pz[i] = r * Math.sin(theta);
        
        const vOrb = Math.sqrt((params.G * mass[0]) / Math.sqrt(r * r + params.eps * params.eps));
        const noise = vOrb * 0.04;
        vx[i] = -vOrb * Math.sin(theta) + (Math.random() - 0.5) * noise;
        vy[i] = (Math.random() - 0.5) * noise;
        vz[i] = vOrb * Math.cos(theta) + (Math.random() - 0.5) * noise;
        mass[i] = 0.8 + Math.random() * 1.5;
      }
    } else if (presetType === 'cluster') {
      // Plummer Star Cluster Model
      for (let i = 0; i < N; i++) {
        const u = Math.random() * 0.95 + 0.01;
        const r = 16.0 / Math.sqrt(Math.pow(u, -2.0 / 3.0) - 1.0);
        const costheta = Math.random() * 2.0 - 1.0;
        const sintheta = Math.sqrt(Math.max(0, 1.0 - costheta * costheta));
        const phi = Math.random() * Math.PI * 2.0;
        
        px[i] = r * sintheta * Math.cos(phi);
        py[i] = r * sintheta * Math.sin(phi);
        pz[i] = r * costheta;
        mass[i] = 1.0 + Math.random() * 1.2;
        
        const vScale = 0.65 * Math.sqrt((params.G * N * 1.0) / Math.hypot(r, 16.0));
        vx[i] = (Math.random() - 0.5) * 2.0 * vScale;
        vy[i] = (Math.random() - 0.5) * 2.0 * vScale;
        vz[i] = (Math.random() - 0.5) * 2.0 * vScale;
      }
    } else if (presetType === 'collision') {
      // Colliding Binary Clusters
      const n1 = Math.floor(N / 2);
      for (let i = 0; i < N; i++) {
        const isGroup1 = i < n1;
        const center = isGroup1 ? [-22, 6, 0] : [22, -6, 0];
        const vGroup = isGroup1 ? [0.65, -0.35, 0] : [-0.65, 0.35, 0];
        
        if (i === 0 || i === n1) {
          mass[i] = 180.0;
          px[i] = center[0]; py[i] = center[1]; pz[i] = center[2];
          vx[i] = vGroup[0]; vy[i] = vGroup[1]; vz[i] = vGroup[2];
          continue;
        }

        const r = 3.0 + Math.random() * 14.0;
        const a = Math.random() * Math.PI * 2.0;
        px[i] = center[0] + r * Math.cos(a);
        py[i] = center[1] + (Math.random() - 0.5) * 1.5;
        pz[i] = center[2] + r * Math.sin(a);
        
        const vo = Math.sqrt((params.G * 180.0) / (r + 0.5));
        const dir = isGroup1 ? 1 : -1;
        vx[i] = vGroup[0] - dir * vo * Math.sin(a);
        vy[i] = vGroup[1] + (Math.random() - 0.5) * 0.1;
        vz[i] = vGroup[2] + dir * vo * Math.cos(a);
        mass[i] = 0.8 + Math.random() * 0.6;
      }
    } else if (presetType === 'collapse') {
      // Cold Spherical Collapse (Zero initial velocity)
      for (let i = 0; i < N; i++) {
        const r = Math.pow(Math.random(), 0.333) * 32.0;
        const costheta = Math.random() * 2.0 - 1.0;
        const sintheta = Math.sqrt(Math.max(0, 1.0 - costheta * costheta));
        const phi = Math.random() * Math.PI * 2.0;
        
        px[i] = r * sintheta * Math.cos(phi);
        py[i] = r * sintheta * Math.sin(phi);
        pz[i] = r * costheta;
        vx[i] = 0; vy[i] = 0; vz[i] = 0;
        mass[i] = 1.0 + Math.random() * 1.0;
      }
    } else if (presetType === 'solar') {
      // Keplerian Solar System with Inner/Outer Orbiters
      px[0] = 0; py[0] = 0; pz[0] = 0;
      vx[0] = 0; vy[0] = 0; vz[0] = 0;
      mass[0] = 500.0;

      for (let i = 1; i < N; i++) {
        const r = 6.0 + Math.pow(Math.random(), 1.3) * 42.0;
        const theta = Math.random() * Math.PI * 2.0;
        
        px[i] = r * Math.cos(theta);
        py[i] = (Math.random() - 0.5) * 0.3;
        pz[i] = r * Math.sin(theta);
        
        const vOrb = Math.sqrt((params.G * mass[0]) / r);
        vx[i] = -vOrb * Math.sin(theta);
        vy[i] = (Math.random() - 0.5) * 0.02;
        vz[i] = vOrb * Math.cos(theta);
        mass[i] = (i <= 5) ? (4.0 + Math.random() * 6.0) : (0.2 + Math.random() * 0.4);
      }
    }
  }

  // N-body Gravity Physics Kernel
  function computeAccelerations(N, G, epsSq) {
    ax.fill(0); ay.fill(0); az.fill(0);

    for (let i = 0; i < N; i++) {
      const xi = px[i], yi = py[i], zi = pz[i];
      let axi = 0.0, ayi = 0.0, azi = 0.0;

      for (let j = i + 1; j < N; j++) {
        const dx = px[j] - xi;
        const dy = py[j] - yi;
        const dz = pz[j] - zi;
        
        const distSq = dx * dx + dy * dy + dz * dz + epsSq;
        const invDist = 1.0 / Math.sqrt(distSq);
        const invDist3 = invDist * invDist * invDist;

        const f_i = G * mass[j] * invDist3;
        axi += dx * f_i;
        ayi += dy * f_i;
        azi += dz * f_i;

        const f_j = G * mass[i] * invDist3;
        ax[j] -= dx * f_j;
        ay[j] -= dy * f_j;
        az[j] -= dz * f_j;
      }

      ax[i] += axi;
      ay[i] += ayi;
      az[i] += azi;
    }
  }

  // Symplectic Leapfrog Integration Step
  function stepPhysics() {
    const N = params.N;
    const dt = params.dt;
    const substeps = params.substeps;
    const subDt = dt / substeps;
    const G = params.G;
    const epsSq = params.eps * params.eps;

    for (let s = 0; s < substeps; s++) {
      // 1. Half-step velocity update
      for (let i = 0; i < N; i++) {
        vx[i] += 0.5 * ax[i] * subDt;
        vy[i] += 0.5 * ay[i] * subDt;
        vz[i] += 0.5 * az[i] * subDt;

        px[i] += vx[i] * subDt;
        py[i] += vy[i] * subDt;
        pz[i] += vz[i] * subDt;
      }

      // 2. Compute accelerations at new positions
      computeAccelerations(N, G, epsSq);

      // 3. Final half-step velocity update
      for (let i = 0; i < N; i++) {
        vx[i] += 0.5 * ax[i] * subDt;
        vy[i] += 0.5 * ay[i] * subDt;
        vz[i] += 0.5 * az[i] * subDt;
      }
    }

    // Optional Center-of-Mass Correction (keeps cluster in view)
    if (params.recenterCoM) {
      let totalM = 0, cmX = 0, cmY = 0, cmZ = 0, cmVx = 0, cmVy = 0, cmVz = 0;
      for (let i = 0; i < N; i++) {
        const m = mass[i];
        totalM += m;
        cmX += px[i] * m; cmY += py[i] * m; cmZ += pz[i] * m;
        cmVx += vx[i] * m; cmVy += vy[i] * m; cmVz += vz[i] * m;
      }
      cmX /= totalM; cmY /= totalM; cmZ /= totalM;
      cmVx /= totalM; cmVy /= totalM; cmVz /= totalM;

      for (let i = 0; i < N; i++) {
        px[i] -= cmX; py[i] -= cmY; pz[i] -= cmZ;
        vx[i] -= cmVx; vy[i] -= cmVy; vz[i] -= cmVz;
      }
    }

    // Compute Speed & Energy metrics
    let maxV = 0, sumV = 0, ke = 0, pe = 0;
    for (let i = 0; i < N; i++) {
      const v2 = vx[i] * vx[i] + vy[i] * vy[i] + vz[i] * vz[i];
      const v = Math.sqrt(v2);
      speed[i] = v;
      if (v > maxV) maxV = v;
      sumV += v;
      ke += 0.5 * mass[i] * v2;
    }

    // Potential Energy U = - sum (G m_i m_j / sqrt(r^2 + eps^2))
    for (let i = 0; i < N; i++) {
      for (let j = i + 1; j < N; j++) {
        const dx = px[j] - px[i];
        const dy = py[j] - py[i];
        const dz = pz[j] - pz[i];
        const dist = Math.sqrt(dx * dx + dy * dy + dz * dz + epsSq);
        pe -= (G * mass[i] * mass[j]) / dist;
      }
    }

    return { maxV, avgV: sumV / N, ke, pe, te: ke + pe };
  }

  // Camera handling
  const projMatrix = mat4Create();
  const viewMatrix = mat4Create();

  function updateCamera() {
    const eyeX = camera.target[0] + camera.radius * Math.sin(camera.theta) * Math.cos(camera.phi);
    const eyeY = camera.target[1] + camera.radius * Math.sin(camera.phi);
    const eyeZ = camera.target[2] + camera.radius * Math.cos(camera.theta) * Math.cos(camera.phi);

    mat4Perspective(projMatrix, Math.PI / 4, canvas.width / canvas.height, 0.1, 1000.0);
    mat4LookAt(viewMatrix, [eyeX, eyeY, eyeZ], camera.target, [0, 1, 0]);
  }

  // Resize handler
  function onResize() {
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const width = window.innerWidth;
    const height = window.innerHeight;
    canvas.width = Math.floor(width * dpr);
    canvas.height = Math.floor(height * dpr);
    gl.viewport(0, 0, canvas.width, canvas.height);
    updateCamera();
  }
  window.addEventListener('resize', onResize);

  // Initial setup
  initPreset(params.preset);
  computeAccelerations(params.N, params.G, params.eps * params.eps);
  onResize();

  // Mouse / Touch Interaction Controls
  let isDragging = false;
  let dragButton = 0;
  let lastMouseX = 0, lastMouseY = 0;
  let touchDist = 0;

  canvas.addEventListener('mousedown', (e) => {
    isDragging = true;
    dragButton = e.button;
    lastMouseX = e.clientX;
    lastMouseY = e.clientY;
  });

  window.addEventListener('mouseup', () => { isDragging = false; });
  
  window.addEventListener('mousemove', (e) => {
    if (!isDragging) return;
    const dx = e.clientX - lastMouseX;
    const dy = e.clientY - lastMouseY;
    lastMouseX = e.clientX;
    lastMouseY = e.clientY;

    if (dragButton === 0 && !e.shiftKey) {
      // Orbit
      camera.theta -= dx * 0.005;
      camera.phi += dy * 0.005;
      const limit = Math.PI / 2 - 0.02;
      camera.phi = Math.max(-limit, Math.min(limit, camera.phi));
    } else {
      // Pan
      const panSpeed = camera.radius * 0.0012;
      const cosT = Math.cos(camera.theta), sinT = Math.sin(camera.theta);
      camera.target[0] -= (dx * cosT - dy * sinT * Math.sin(camera.phi)) * panSpeed;
      camera.target[1] += dy * Math.cos(camera.phi) * panSpeed;
      camera.target[2] -= (-dx * sinT - dy * cosT * Math.sin(camera.phi)) * panSpeed;
    }
  });

  canvas.addEventListener('wheel', (e) => {
    e.preventDefault();
    camera.radius *= (e.deltaY > 0 ? 1.08 : 0.92);
    camera.radius = Math.max(5.0, Math.min(400.0, camera.radius));
  }, { passive: false });

  // Touch controls for mobile
  canvas.addEventListener('touchstart', (e) => {
    if (e.touches.length === 1) {
      isDragging = true;
      dragButton = 0;
      lastMouseX = e.touches[0].clientX;
      lastMouseY = e.touches[0].clientY;
    } else if (e.touches.length === 2) {
      const dx = e.touches[0].clientX - e.touches[1].clientX;
      const dy = e.touches[0].clientY - e.touches[1].clientY;
      touchDist = Math.hypot(dx, dy);
    }
  }, { passive: true });

  canvas.addEventListener('touchmove', (e) => {
    if (e.touches.length === 1 && isDragging) {
      const dx = e.touches[0].clientX - lastMouseX;
      const dy = e.touches[0].clientY - lastMouseY;
      lastMouseX = e.touches[0].clientX;
      lastMouseY = e.touches[0].clientY;
      camera.theta -= dx * 0.005;
      camera.phi += dy * 0.005;
      const limit = Math.PI / 2 - 0.02;
      camera.phi = Math.max(-limit, Math.min(limit, camera.phi));
    } else if (e.touches.length === 2) {
      const dx = e.touches[0].clientX - e.touches[1].clientX;
      const dy = e.touches[0].clientY - e.touches[1].clientY;
      const dist = Math.hypot(dx, dy);
      if (touchDist > 0) {
        camera.radius *= (touchDist / dist);
        camera.radius = Math.max(5.0, Math.min(400.0, camera.radius));
      }
      touchDist = dist;
    }
  }, { passive: true });

  canvas.addEventListener('touchend', () => { isDragging = false; touchDist = 0; });

  // UI Event Bindings
  const sliderCount = document.getElementById('slider-count');
  const sliderG = document.getElementById('slider-g');
  const sliderEps = document.getElementById('slider-eps');
  const sliderDt = document.getElementById('slider-dt');
  const sliderPsize = document.getElementById('slider-psize');
  const presetSelect = document.getElementById('preset-select');
  const colorSelect = document.getElementById('color-select');
  const btnPlay = document.getElementById('btn-play');
  const btnReset = document.getElementById('btn-reset');
  const btnRecenter = document.getElementById('btn-recenter');
  const btnCoM = document.getElementById('btn-com');
  const legendBar = document.getElementById('legend-bar');

  const gradients = [
    'linear-gradient(to right, #001a80, #00b3ff, #33e633, #ffcc00, #ff1a1a)',
    'linear-gradient(to right, #1a33cc, #e61a99, #33ffe6, #ffffff)',
    'linear-gradient(to right, #4d001a, #e63300, #ffb300, #ffffef)',
    'linear-gradient(to right, #330066, #0080ff, #ccffff, #ffffff)'
  ];

  sliderCount.addEventListener('input', (e) => {
    params.N = parseInt(e.target.value);
    document.getElementById('val-count').textContent = params.N;
    document.getElementById('badge-n').textContent = `N=${params.N}`;

  sliderG.addEventListener('input', (e) => {
    params.G = parseFloat(e.target.value);
    document.getElementById('val-g').textContent = params.G.toFixed(2);
  });

  sliderEps.addEventListener('input', (e) => {
    params.eps = parseFloat(e.target.value);
    document.getElementById('val-eps').textContent = params.eps.toFixed(2);
  });

  sliderDt.addEventListener('input', (e) => {
    params.dt = parseFloat(e.target.value);
    document.getElementById('val-dt').textContent = params.dt.toFixed(3);
  });

  sliderPsize.addEventListener('input', (e) => {
    params.pointSize = parseFloat(e.target.value);
    document.getElementById('val-psize').textContent = params.pointSize;
  });

  presetSelect.addEventListener('change', (e) => {
    params.preset = e.target.value;
    initPreset(params.preset);

    params.colorMode = parseInt(e.target.value);

  });

  btnPlay.addEventListener('click', () => {
    params.paused = !params.paused;
    btnPlay.textContent = params.paused ? 'Play' : 'Pause';
    btnPlay.classList.toggle('btn-primary', !params.paused);
  });

  btnReset.addEventListener('click', () => {
    initPreset(params.preset);
  });

  btnRecenter.addEventListener('click', () => {
    camera.target = [0, 0, 0];
    camera.radius = 75.0;
  });

  btnCoM.addEventListener('click', () => {
    params.recenterCoM = !params.recenterCoM;
    btnCoM.textContent = `CoM Frame: ${params.recenterCoM ? 'ON' : 'OFF'}`;
  });

  // UI Collapse toggle
  const uiContainer = document.getElementById('ui-container');
  const headerBtn = document.getElementById('ui-header-btn');
  const toggleBtn = document.getElementById('toggle-ui');
  headerBtn.addEventListener('click', () => {
    uiContainer.classList.toggle('collapsed');
    toggleBtn.textContent = uiContainer.classList.contains('collapsed') ? '+' : '−';
  });

  // Render Loop & FPS calculation
  let lastFrameTime = performance.now();
  let frameCount = 0;
  let fps = 60;
  let lastStatsUpdate = 0;

  function render(time) {
    animFrameId = requestAnimationFrame(render);

    // Calculate FPS
    frameCount++;
    const delta = time - lastFrameTime;
    if (delta >= 500) {
      fps = Math.round((frameCount * 1000) / delta);
      frameCount = 0;
      lastFrameTime = time;
    }

    // Step physics if not paused
    let stats = { maxV: 0, ke: 0, pe: 0, te: 0 };
    if (!params.paused) {
      stats = stepPhysics();
    }

    // Update stats UI periodically
    if (time - lastStatsUpdate > 100) {
      lastStatsUpdate = time;
      document.getElementById('stat-fps').textContent = fps;
      document.getElementById('stat-vmax').textContent = stats.maxV.toFixed(1);
      document.getElementById('stat-ke').textContent = stats.ke.toFixed(0);
      document.getElementById('stat-pe').textContent = stats.pe.toFixed(0);
      document.getElementById('stat-te').textContent = stats.te.toFixed(0);
      document.getElementById('legend-max-v').textContent = stats.maxV.toFixed(1);
    }

    // Update Camera
    updateCamera();

    // Prepare WebGL State
    gl.clearColor(0.047, 0.047, 0.063, 1.0);
    gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

    gl.enable(gl.BLEND);
    gl.blendFunc(gl.SRC_ALPHA, gl.ONE); // Additive glowing particles
    gl.disable(gl.DEPTH_TEST);

    gl.useProgram(program);

    // Pack position data
    const N = params.N;
    for (let i = 0; i < N; i++) {
      posBufferData[i * 3]     = px[i];
      posBufferData[i * 3 + 1] = py[i];
      posBufferData[i * 3 + 2] = pz[i];
    }

    // Upload VBOs
    gl.bindBuffer(gl.ARRAY_BUFFER, posVBO);
    gl.bufferData(gl.ARRAY_BUFFER, posBufferData.subarray(0, N * 3), gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(locs.aPosition);
    gl.vertexAttribPointer(locs.aPosition, 3, gl.FLOAT, false, 0, 0);

    gl.bindBuffer(gl.ARRAY_BUFFER, velVBO);
    gl.bufferData(gl.ARRAY_BUFFER, speed.subarray(0, N), gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(locs.aVelocity);
    gl.vertexAttribPointer(locs.aVelocity, 1, gl.FLOAT, false, 0, 0);

    gl.bindBuffer(gl.ARRAY_BUFFER, massVBO);
    gl.bufferData(gl.ARRAY_BUFFER, mass.subarray(0, N), gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(locs.aMass);
    gl.vertexAttribPointer(locs.aMass, 1, gl.FLOAT, false, 0, 0);

    // Set Uniforms
    gl.uniformMatrix4fv(locs.uProjectionMatrix, false, projMatrix);
    gl.uniformMatrix4fv(locs.uViewMatrix, false, viewMatrix);
    gl.uniform1f(locs.uMaxSpeed, Math.max(stats.maxV, 5.0));
    gl.uniform1f(locs.uPointSize, params.pointSize);
    gl.uniform1i(locs.uColorMode, params.colorMode);

    // Draw Particles
    gl.drawArrays(gl.POINTS, 0, N);
  }

  // Start Animation Loop
  animFrameId = requestAnimationFrame(render);

} catch (err) {
  const errDiv = document.getElementById('error-log');
  if (errDiv) {
    errDiv.style.display = 'block';
    errDiv.textContent = 'Application Initialization Error: ' + (err.stack || err.message || err);
  }
  console.error(err);
}
