    html,body{margin:0;padding:0;min-height:100%;box-sizing:border-box;background:#0c0c10;color:#ececf0;font-family:ui-sans-serif,system-ui}
    canvas{width:100%;height:100%;touch-action:pan-x;user-select:none}
    .controls{position:absolute;top:10px;right:10px;background:#000;opacity:0.9;backdrop-filter:blur(4px);border-radius:8px;padding:12px;border:1px solid #333;color:#fff}
    .control{display:flex;gap:8px;margin-bottom:12px}
    label{display:flex;align-items:center;font-size:14px;margin-right:4px}
    input{width:80px}
    #fps-label{position:absolute;bottom:10px;left:10px;color:#999;opacity:0.6}
    #info{position:absolute;bottom:30px;right:10px;color:#666;opacity:0.7;display:none}
    @media (max-height: 480px){
      .controls{top:24px}
      #info{display:block}
    }
    .clock{width:20px;height:20px;background:#00ccff;position:absolute;transition:opacity 0.1s}
    .clock.hidden{opacity:0}

<body>
  <div class="controls">
    <div class="control"><label>Pendulums</label><input type="range" id="N" min="5" max="30" value="15"/></div>
    <div class="control"><label>Gravity</label><input type="range" id="g" min="1" max="20" value="10"/></div>
  </div>
  <div id="fps-label"></div>
  <div id="info">Use sliders to adjust pendulums and gravity. Click to reset.</div>
  <canvas id="scene"></canvas>

<script>
(function() {
  try {
    const canvas = document.getElementById('scene'),
        ctx = canvas.getContext('2d');
    let width, height, pixelRatio, startTime;
    let N = 15, g = 10;
    let requestId = null;
    
    function resizeCanvas() {
      width = Math.floor(window.innerWidth * pixelRatio);
      height = Math.floor(window.innerHeight * pixelRatio);
      canvas.width = width;
      canvas.height = height;
      _redraw();
    }
    
    document.addEventListener('DOMContentLoaded', () => {
      pixelRatio = window.devicePixelRatio || 1;
      resizeCanvas();
      
      document.getElementById('N').addEventListener('input', updateParams);
      document.getElementById('g').addEventListener('input', updateParams);
      document.getElementById('g').addEventListener('change', () => document.getElementById('info').style.display = 'none');
      
      canvas.addEventListener('click', () => {
        N = document.getElementById('N').valueAsNumber;
        g = document.getElementById('g').valueAsNumber;
        startTime = Date.now();
        _redraw();
      });
      
      let frame = 0;
      function animate() {
        requestId = requestAnimationFrame(animate);
        _redraw();
        frame++;
      }
      animate();
      
      canvas.addEventListener('mousemove', e => {
        const clock = Array.from(canvas.querySelectorAll('.clock')).find(c => 
          c.offsetLeft + width/2 === Math.floor(e.clientX * pixelRatio) && 
          c.offsetTop + height/2 === Math.floor(e.clientY * pixelRatio)
        );
        clock?.classList.toggle('hidden', !clock);
      });
    });
    
    function _redraw() {
      ctx.fillRect(0, 0, width, height);
      
      const gravity = g/100, angleScale = width/4, timeScale = 250;
      if (!startTime) startTime = Date.now();
      
      for (let i = 0; i < N; i++) {
        const baseLength = width/3, 
              length = baseLength + i*2, 
              phase = i/(N-1)*(Math.PI/2),
              t = (Date.now() - startTime)/timeScale;
        
        const theta = phase * Math.sin(gravity * t);
        const x = width/2 + angleScale * Math.sin(theta) * Math.min(1, width/(length*theta/angleScale));
        const y = height*0.8 + angleScale * Math.cos(theta) * (length/angleScale);
        
        ctx.strokeStyle = `hsl(220,${(1 - i/N)*90}%,50%)`;
        ctx.beginPath();
        ctx.moveTo(width/2, height*0.8);
        ctx.lineTo(x, y);
        ctx.stroke();
        
        ctx.beginPath();
        ctx.arc(x, y, 5, 0, Math.PI*2);
        ctx.lastPath = ctx.fillStyle = (i === N-1) ? '#00ccff' : `rgba(0,204,255,${0.5-(i/N*0.3)})`;
        ctx.fill();
      }
      
      ctx.fillStyle = '#fff';
      ctx.fillText(`N: ${N}  g: ${g}m/s²`, 20, height*0.85);
      
      const now = Date.now(), fps = 1000 / (now - (startTime ?? now)) / pixelRatio;
      document.getElementById('fps-label').textContent = `fps: ${Math.floor(fps)}`;
    }
    
    window.addEventListener('resize', resizeCanvas);
    canvas.addEventListener('contextmenu', e => e.preventDefault());
  } catch (e) {
    const alertBox = document.createElement('div');
    alertBox.style = "position:absolute;top:10px;left:10px;color:#fff;background:#ff3333;padding:8px;border-radius:4px";
    alertBox.textContent = `Error: ${e.message}`;
    document.body.appendChild(alertBox);
  }
})();
</script>
