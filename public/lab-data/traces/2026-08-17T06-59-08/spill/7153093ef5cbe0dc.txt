                            ctx.fillText(vE.toFixed(2)+'V', px+5,2);
                        }
                        else {
                            ctx.fillText(vE.toFixed(2)+'A', px+5,2);
                        }
                    }
                    
                    if(drawMode=='both') {
                        ctx.fillStyle='lime';
                        ctx.fillText(vE.toFixed(2)+'V', px+34,2);
                    }
                    
                    if(drawMode=='greenV') ctx.fillStyle='#00ff00';
                    if(drawMode=='greenI') ctx.fillStyle='#0f0';
                    
                    const dv1=+dv[0].toFixed(2);
                    const id=+dv[0]<=0?'Red':'Blue';
                    const rectColor=(drawMode=='greenV')?id:'blue';
                    
                    context.fillStyle=rectColor;
                    context.fillRect(px,1,10,1);
                    context.fillStyle='lime';
                    context.fillRect(0,1,canvas.width,1);
                });
            }
