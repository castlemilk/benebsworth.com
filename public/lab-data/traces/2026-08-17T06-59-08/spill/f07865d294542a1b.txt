    THREE.Vector3.prototype.setScalar = function(s) { 
      this.x *= s; this.y *= s; this.z *= s; return this; 
    };
    THREE.Vector3.prototype.normalize = function() { 
      const len = Math.sqrt(this.x*this.x + this.y*this.y + this.z*this.z);
      if (len === 0) return this;
      this.x /= len; this.y /= len; this.z /= len; return this; 
    };
    THREE.Vector3.prototype.multiplyScalar = function(s) { 
      this.x *= s; this.y *= s; this.z *= s; return this;

      const ax=a.x, ay=a.y, az=a.z, bx=b.x, by=b.y, bz=b.z;

    const objects = [];
    const softeningEnabled = true;

    try {
      const softening = softeningEnabled ? 0.1 : 0;
      const objectsById = {};

      function createParticles(count) {
        const geometries = [];
        const colors = [];
        for (let i = 0; i < count; i++) {
          const radius = 0.2;
          const x = Math.random() * 100 - 50;
          const y = Math.random() * 100 - 50;
          const z = Math.random() * 100 - 50;
          const velocity = new THREE.Vector3();
          const positionObject = new THREE.Vector3(x, y, z);
          const geometry = new THREE.BufferGeometry();
          geometry.attributes.position = new THREE.BufferAttribute(new Float32Array(3), 3);
          geometry.attributes.position.array.set([x, y, z]);
          geometries.push(geometry);

          var particleElement = document.createElement('div');
          particleElement.className = 'velocity-indicator';
          document.body.appendChild(particleElement);
          colors.push(particleElement);

          objects[i] = {
            positionObject: positionObject,
            velocity: velocity,
            desiredPosition: new THREE.Vector3(),
            geometry: geometry,
            velocityElement: particleElement,
          };
          objectsById[`particle${i}`] = i;
        }
        return { geometries, colors };
      }

      function animate() {
        requestAnimationFrame(animate);
        const now = THREE.Math.now();
        for (let i = 0; i < objects.length; i++) {
          const a = objects[i];
          let forceAcc = new THREE.Vector3(0, 0, 0);
          for (let j = 0; j < objects.length; j++) {
            if (i === j) continue;
            const b = objects[j];
            const targetPosition = objectsById[j] < i ? b.positionObject : null;
            if (!targetPosition) continue;
            const deltaPop = a.positionObject.subVectors(b.positionObject, a.positionObject);
            const dist = deltaPop.setLength(1).length();
            const softeningValue = softeningEnabled ? softening : 0;
            const soft = Math.max(softeningValue, dist);
            const forceMag = (1 / (soft * soft)) * THREE.Math.E;
            const force = deltaPop.clone().multiplyScalar(-forceMag);
            forceAcc.addVectors(forceAcc, force);
          }
          const velocity = a.velocity;
          velocity.addVectors(velocity, forceAcc);
          const desiredPositions = new THREE.Vector3(0, 0, 0);
          desiredPositions.addVectors(desiredPositions, velocity);
          a.positionObject.addVectors(a.positionObject, desiredPositions);
          updateGeometry(a.geometry, a.positionObject);
          refreshVelocityIndicator(a);
        }
        renderer.render();
      }

      function updateGeometry(geometry) {
        geometry.attributes.position.array.set([a.positionObject.x, a.positionObject.y, a.positionObject.z]);
      }

      function refreshVelocityIndicator(obj) {
        const vel = obj.velocity;
        const len = vel.setLength(1).length();
        color = new THREE.Color(`hsl(0, 100%, ${Math.min(70, Math.max(30, parseInt(100 * (1 - len)))}%)}`);
        obj.velocityElement.style.backgroundColor = color.getStyle() || `rgb(${Math.floor(color.r * 255)}, ${Math.floor(color.g * 255)}, ${Math.floor(color.b * 255)})`;
      }

      function init(count) {
        camera.position = { x: 0, y: 80, z: 20 };
        objects.length = 0;
        objectsById = {};
        const numParticles = count;
        const initParticles = createParticles(numParticles);
        objects = initParticles.colors;
        for (let i = 0; i < numParticles; i++) {
          objectsById[`particle${i}`] = i;
        }
        stats.element.firstElementChild.textContent = `Particles: ${numParticles}`;
        window.addEventListener('resize', handleResize);
        animate();
      }

      function handleResize() {
        const width = window.innerWidth;
        const height = window.innerHeight;
        renderer.setSize(width, height);
      }

      document.getElementById('particles-count').onchange = function(e) {
        init(parseInt(e.target.value));
      };
      document.getElementById('toggle-softening').onclick = function() {
        softeningEnabled = !softeningEnabled;
        document.getElementById('softening-text').textContent = softeningEnabled ? 'Softening Enabled' : 'Softening Disabled';
        document.getElementById('softening-text').style.color = softeningEnabled ? '#00ffcc' : '#ffcc00';
        init(objects.length);
      };
      renderFrame = init(200);
    } catch (e) {

      errorDiv.style = 'color:red; padding:1rem; margin:1rem;';
      errorDiv.textContent = `Error: ${e.message}`;
      document.body.prepend(errorDiv);
    }
