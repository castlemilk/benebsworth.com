  <script>
    try {
      'use strict';
      const cells = Array.from(document.querySelectorAll('.cell'));
      const players = Array.from(document.querySelectorAll('.player'));
      const announcement = document.getElementById('announcement');
      const hint = document.getElementById('hint');
      const winningLines = [
        [0,1,2], [3,4,5], [6,7,8],
        [0,3,6], [1,4,7], [2,5,8],
        [0,4,8], [2,4,6]
      ];
      let board = Array(9).fill('');
      let turn = 'X';
      let finished = false;

      function label(index, mark) {
        return 'Row ' + (Math.floor(index / 3) + 1) + ', column ' + (index % 3 + 1) + ', ' + (mark || 'empty');
      }

      function updatePlayers() {
        players.forEach(player => player.classList.toggle('active', !finished && player.dataset.mark === turn));
      }

      function play(index) {
        if (finished || board[index]) return;
        board[index] = turn;
        cells[index].textContent = turn;
        cells[index].dataset.mark = turn;
        cells[index].setAttribute('aria-label', label(index, turn));
        cells[index].setAttribute('aria-disabled', 'true');

        const wins = winningLines.filter(line => line.every(position => board[position] === turn));
        if (wins.length) {
          finished = true;
          wins.forEach(line => line.forEach(position => cells[position].classList.add('winner')));
          announcement.textContent = turn + ' wins!';
          hint.textContent = 'Three in a row. Ready for a rematch?';
        } else if (board.every(Boolean)) {
          finished = true;
          announcement.textContent = "It's a draw!";
          hint.textContent = 'Well matched. Give it another go.';
        } else {
          turn = turn === 'X' ? 'O' : 'X';
          announcement.textContent = turn + "'s turn";
          hint.textContent = 'Choose an empty square. Get three in a row.';
        }

        if (finished) cells.forEach(cell => cell.setAttribute('aria-disabled', 'true'));
        updatePlayers();
      }

      function restart() {
        board = Array(9).fill('');
        turn = 'X';
        finished = false;
        cells.forEach((cell, index) => {
          cell.textContent = '';
          delete cell.dataset.mark;
          cell.classList.remove('winner');
          cell.removeAttribute('aria-disabled');
          cell.setAttribute('aria-label', label(index, ''));
        });
        announcement.textContent = "X's turn";
        hint.textContent = 'Choose an empty square. Get three in a row.';
        updatePlayers();
      }

      cells.forEach((cell, index) => cell.addEventListener('click', () => play(index)));
      document.getElementById('restart').addEventListener('click', restart);
    } catch (error) {
      const errorBox = document.getElementById('error');
      errorBox.hidden = false;
      errorBox.textContent = 'Unable to start the game: ' + (error.message || String(error));
    }
  </script>
