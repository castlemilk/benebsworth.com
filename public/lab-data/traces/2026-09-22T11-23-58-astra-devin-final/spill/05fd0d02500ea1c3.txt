  <script>
    try {
      const game = document.querySelector('.game');
      const cells = Array.from(document.querySelectorAll('#board .cell'));
      const status = document.getElementById('status');
      const hint = document.getElementById('hint');
      const lines = [
        [0, 1, 2], [3, 4, 5], [6, 7, 8],
        [0, 3, 6], [1, 4, 7], [2, 5, 8],
        [0, 4, 8], [2, 4, 6]
      ];
      let board = Array(9).fill('');
      let player = 'X';
      let finished = false;

      function label(index, mark) {
        return 'Row ' + (Math.floor(index / 3) + 1) + ', column ' + (index % 3 + 1) + ': ' + (mark || 'empty');
      }

      function move(index) {
        if (finished || board[index] !== '') return;
        board[index] = player;
        cells[index].textContent = player;
        cells[index].dataset.mark = player;
        cells[index].setAttribute('aria-label', label(index, player));
        cells[index].setAttribute('aria-disabled', 'true');

        const winningLines = lines.filter(line => line.every(position => board[position] === player));
        if (winningLines.length) {
          finished = true;
          status.textContent = player + ' wins!';
          hint.textContent = 'Three in a row. Nicely played.';
          winningLines.forEach(line => line.forEach(position => cells[position].classList.add('winning')));
        } else if (board.every(Boolean)) {
          finished = true;
          status.textContent = 'It’s a draw!';
          hint.textContent = 'An even match. Try another round.';
        } else {
          player = player === 'X' ? 'O' : 'X';
          game.dataset.turn = player;
          status.textContent = player + '’s turn';
        }

        if (finished) cells.forEach(cell => cell.setAttribute('aria-disabled', 'true'));
      }

      cells.forEach((cell, index) => cell.addEventListener('click', () => move(index)));
      document.getElementById('restart').addEventListener('click', () => {
        board = Array(9).fill('');
        player = 'X';
        finished = false;
        game.dataset.turn = player;
        cells.forEach((cell, index) => {
          cell.textContent = '';
          cell.removeAttribute('data-mark');
          cell.removeAttribute('aria-disabled');
          cell.classList.remove('winning');
          cell.setAttribute('aria-label', label(index, ''));
        });
        status.textContent = 'X’s turn';
        hint.textContent = 'Choose an empty square.';
      });
    } catch (error) {
      const errorBox = document.getElementById('error');
      errorBox.textContent = 'The game could not start: ' + (error && error.message ? error.message : String(error));
      errorBox.hidden = false;
    }
  </script>
