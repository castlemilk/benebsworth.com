 const a=z>1?-1/(z+q):0, b=z>1?-(z+q):0;
 function at(t) {
  const x=Math.max(0,t)*w;
  if (x===0) return 0;
  if (Math.abs(z-1)<1e-9) return 1-Math.exp(-x)*(1+x);
  if (z<1) return 1-Math.exp(-z*x)*(Math.cos(q*x)+z*x*sinc(q*x));
  return 1+(b*Math.exp(a*x)-a*Math.exp(b*x))/(a-b);
 }
 const peak=z<1?Math.exp(-Math.PI*z/q):0;
 const peakTime=z<1?Math.PI/(w*q):null;
 // Search backward from a proven error-envelope bound, then bisect the last 2% crossing.
 let high;
 if (z<1-1e-9) high=Math.log(1/(.02*q))/(z*w);
 else if (z<=1+1e-9) high=9/w;
 else high=Math.log((Math.abs(b)/(a-b))/.02)/(-a*w);
 let inside=high, outside=high;
 const dt=z<1 ? Math.min(Math.PI/(w*q)/32,high/300) : high/1000;
 for(let k=0;k<30000;k++) {
  outside=Math.max(0,inside-dt);
  if(Math.abs(at(outside)-1)>.02) break;
  inside=outside;
  if(outside===0) break;
 }
 for(let k=0;k<42;k++) {const mid=(outside+inside)/2;if(Math.abs(at(mid)-1)>.02)outside=mid;else inside=mid;}
 const settle=(inside+outside)/2;
 const period=2*Math.PI/w;
 const auto=z<1?Math.max(1.65*period,Math.min(settle*1.3,8*period)):settle*1.28;
 return {R,L,C,w,z,at,peak,peakTime,settle,auto};
}
