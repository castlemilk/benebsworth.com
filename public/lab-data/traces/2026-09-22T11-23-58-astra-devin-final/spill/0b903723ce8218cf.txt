        <pre><code><span class="string">&quot;&quot;&quot;Constant-time comparison and PBKDF2 password hashing.

Public API
----------
constant_time_compare(a, b) -&gt; bool
    Timing-safe equality check for str/bytes.
hash_password(password, *, iterations=..., salt=None) -&gt; str
    Derive a PBKDF2-HMAC-SHA256 hash with a random salt and return it in a
    self-describing storage string.
verify_password(password, stored) -&gt; bool
    Re-derive and compare against a stored hash in constant time.

Storage format:  pbkdf2_sha256$&lt;iterations&gt;$&lt;b64 salt&gt;$&lt;b64 derived key&gt;

Stdlib only: hashlib, hmac, secrets, base64.
&quot;&quot;&quot;</span>

<span class="kw">from</span> __future__ <span class="kw">import</span> annotations

<span class="kw">import</span> base64
<span class="kw">import</span> hashlib
<span class="kw">import</span> hmac
<span class="kw">import</span> secrets

<span class="const">ALGORITHM</span> = <span class="string">&quot;pbkdf2_sha256&quot;</span>
<span class="const">DEFAULT_ITERATIONS</span> = <span class="number">600_000</span>  <span class="comment"># OWASP recommendation for PBKDF2-HMAC-SHA256</span>
<span class="const">SALT_BYTES</span> = <span class="number">16</span>               <span class="comment"># 128-bit random salt</span>
<span class="const">DKLEN</span> = <span class="number">32</span>                    <span class="comment"># 256-bit derived key</span>


<span class="kw">def</span> <span class="func">_to_bytes</span>(value: str | bytes | bytearray, name: str) -&gt; bytes:
    <span class="kw">if</span> <span class="func">isinstance</span>(value, str):
        <span class="kw">return</span> value.<span class="func">encode</span>(<span class="string">&quot;utf-8&quot;</span>)
    <span class="kw">if</span> <span class="func">isinstance</span>(value, (bytes, bytearray)):
        <span class="kw">return</span> <span class="func">bytes</span>(value)
    <span class="kw">raise</span> <span class="func">TypeError</span>(<span class="string">f&quot;{name} must be str or bytes, got {type(value).__name__}&quot;</span>)


<span class="kw">def</span> <span class="func">_b64encode</span>(raw: bytes) -&gt; str:
    <span class="kw">return</span> base64.<span class="func">b64encode</span>(raw).<span class="func">decode</span>(<span class="string">&quot;ascii&quot;</span>)


<span class="kw">def</span> <span class="func">_b64decode</span>(text: str) -&gt; bytes:
    <span class="kw">return</span> base64.<span class="func">b64decode</span>(text.<span class="func">encode</span>(<span class="string">&quot;ascii&quot;</span>), validate=<span class="kw">True</span>)


<span class="kw">def</span> <span class="func">constant_time_compare</span>(a: str | bytes | bytearray, b: str | bytes | bytearray) -&gt; bool:
    <span class="string">&quot;&quot;&quot;Return True iff a == b, in time independent of where they differ.

    Wraps hmac.compare_digest so the comparison does not short-circuit on
    the first differing byte, which would leak a prefix-length signal to a
    timing attacker.
    &quot;&quot;&quot;</span>
    <span class="kw">return</span> hmac.<span class="func">compare_digest</span>(<span class="func">_to_bytes</span>(a, <span class="string">&quot;a&quot;</span>), <span class="func">_to_bytes</span>(b, <span class="string">&quot;b&quot;</span>))


<span class="kw">def</span> <span class="func">hash_password</span>(
    password: ***REDACTED*** | bytes,
    *,
    iterations: int = <span class="const">DEFAULT_ITERATIONS</span>,
    salt: bytes | <span class="kw">None</span> = <span class="kw">None</span>,
) -&gt; str:
    <span class="string">&quot;&quot;&quot;Hash a password with PBKDF2-HMAC-SHA256 and a random salt.

    Returns a self-describing string: algorithm, iteration count, salt, and
    derived key — everything verify_password needs, nothing else.
    &quot;&quot;&quot;</span>
    <span class="kw">if</span> iterations &lt; <span class="number">1</span>:
        <span class="kw">raise</span> <span class="func">ValueError</span>(<span class="string">&quot;iterations must be &gt;= 1&quot;</span>)
    <span class="kw">if</span> salt <span class="kw">is</span> <span class="kw">None</span>:
        salt = secrets.<span class="func">token_bytes</span>(<span class="const">SALT_BYTES</span>)
    dk = hashlib.<span class="func">pbkdf2_hmac</span>(
        <span class="string">&quot;sha256&quot;</span>, <span class="func">_to_bytes</span>(password, <span class="string">&quot;password&quot;</span>), salt, iterations, dklen=<span class="const">DKLEN</span>
    )
    <span class="kw">return</span> <span class="string">f&quot;{ALGORITHM}${iterations}${_b64encode(salt)}${_b64encode(dk)}&quot;</span>


<span class="kw">def</span> <span class="func">verify_password</span>(password: ***REDACTED*** | bytes, stored: str) -&gt; bool:
    <span class="string">&quot;&quot;&quot;Check a password against a stored hash string. Never raises on bad input.&quot;&quot;&quot;</span>
    <span class="kw">try</span>:
        algorithm, iterations_s, salt_b64, hash_b64 = stored.<span class="func">split</span>(<span class="string">&quot;$&quot;</span>)
        <span class="kw">if</span> <span class="kw">not</span> <span class="func">constant_time_compare</span>(algorithm, <span class="const">ALGORITHM</span>):
            <span class="kw">return</span> <span class="kw">False</span>
        iterations = <span class="func">int</span>(iterations_s)
        <span class="kw">if</span> iterations &lt; <span class="number">1</span>:
            <span class="kw">return</span> <span class="kw">False</span>
        salt = <span class="func">_b64decode</span>(salt_b64)
        expected = <span class="func">_b64decode</span>(hash_b64)
    <span class="kw">except</span> (ValueError, TypeError, AttributeError):
        <span class="kw">return</span> <span class="kw">False</span>
    dk = hashlib.<span class="func">pbkdf2_hmac</span>(
        <span class="string">&quot;sha256&quot;</span>, <span class="func">_to_bytes</span>(password, <span class="string">&quot;password&quot;</span>), salt, iterations, dklen=<span class="func">len</span>(expected)
    )
    <span class="kw">return</span> <span class="func">constant_time_compare</span>(dk, expected)
