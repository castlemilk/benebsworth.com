    const kind = critical ? 'critical' : z < 1 ? 'under' : 'over';
    const wd = kind === 'under' ? wn*Math.sqrt((1-z)*(1+z)) : 0;
    const root = kind === 'over' ? Math.sqrt((z-1)*(z+1)) : 0;
    // The slow root is rationalized to avoid subtracting nearly equal numbers.
    const slow = kind === 'over' ? wn/(z+root) : wn;
    const fast = kind === 'over' ? wn*(z+root) : wn;
    function value(t) {
      if (t <= 0) return 0;
      if (kind === 'under') {
        const x = wd*t;
        const sinc = Math.abs(x)<1e-4 ? 1-x*x/6+x*x*x*x/120 : Math.sin(x)/x;
        return -Math.expm1(-alpha*t) + Math.exp(-alpha*t)*(2*Math.sin(x/2)**2-alpha*t*sinc);
      }
      if (kind === 'critical') return -Math.expm1(-wn*t)-wn*t*Math.exp(-wn*t);
      const gap = fast-slow;
      return -Math.expm1(-slow*t) + slow/gap*Math.exp(-slow*t)*Math.expm1(-gap*t);
    }
    const overshoot = kind === 'under' ? Math.exp(-Math.PI*alpha/wd) : 0;
    const peakTime = kind === 'under' ? Math.PI/wd : null;
    let lo=0, hi;
    if (kind === 'under') {
      // After the last extremum outside the 2% band, the next crossing is final.
      const step = Math.PI/wd;
      const lastPeak = Math.floor(Math.log(50)*wd/(Math.PI*alpha));
      lo=lastPeak*step;
      hi=lo+(Math.PI-Math.atan(wd/alpha))/wd;
    } else hi=8/slow;
    for (let i=0;i<64;i++) {
      const mid=(lo+hi)/2;
      if (Math.abs(value(mid)-1)>0.02) lo=mid; else hi=mid;
    }
    const settle=(lo+hi)/2;
    const idealWindow=kind==='under' ? Math.max(8/wn,Math.min(6/alpha,12*2*Math.PI/wd)) : 7/slow;
    return {wn,alpha,z,kind,wd,value,overshoot,peakTime,settle,idealWindow};
  }
