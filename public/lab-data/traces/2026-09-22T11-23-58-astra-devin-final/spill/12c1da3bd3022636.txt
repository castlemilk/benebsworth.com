  // Integrating phase time makes gravity changes continuous in position.
  function rebuild() {
    pendulums = Array.from({length:count}, (_, i) => {
      const cycles = FIRST + i;
      const length = BASE_G * Math.pow(CYCLE / (TAU * cycles), 2);
      return {i, x:-7.2 + i * 14.4 / (count - 1), length, visualLength:length / length0 * 4.05, omega:TAU * cycles / CYCLE, hue:colors(i)};
    });
    byId('countValue').value = count;
    document.querySelector('.scene-caption').textContent = count + ' OSCILLATORS · ONE SHARED CYCLE';
    canvas.setAttribute('aria-label', count + ' pendulums with different lengths swinging at gravity ' + gravity.toFixed(1) + ' meters per second squared.');
  }
  function project(x,y,z) { return {x:centerX + scale * (x * spread + z * .55), y:centerY + scale * (x * tilt - y * .92 + z * .54)}; }

    const theta = AMPLITUDE * Math.cos(p.omega * time);
    return {x:p.x, y:4.55 - p.visualLength * Math.cos(theta), z:p.visualLength * Math.sin(theta)};
  }
  function line(a,b,color,lineWidth=1) {
    ctx.beginPath(); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); ctx.strokeStyle=color; ctx.lineWidth=lineWidth; ctx.stroke();
  }
  function worldLine(a,b,color,lineWidth=1) {line(project(...a),project(...b),color,lineWidth);}

    for(const x of [-8.5,8.5])for(const y of [-.45,5.0])for(const z of [-1.35,1.5])bounds.push({x:x*spread+z*.55,y:x*tilt-y*.92+z*.54});
    const minX=Math.min(...bounds.map(p=>p.x)),maxX=Math.max(...bounds.map(p=>p.x));
    const minY=Math.min(...bounds.map(p=>p.y)),maxY=Math.max(...bounds.map(p=>p.y));
    scale=Math.max(.5,Math.min((width-(width<681?25:80))/(maxX-minX),Math.max(20,sceneBottom-sceneTop)/(maxY-minY)));
    centerX=width/2-(minX+maxX)/2*scale;
    centerY=(sceneTop+sceneBottom)/2-(minY+maxY)/2*scale;
    draw();
  }
  function failure(error) {
    stopped=true;cancelAnimationFrame(raf);
    const box=byId('error');box.style.display='block';box.textContent='The simulation could not continue: '+(error.message||String(error));
  }
  function animate(now) {
    if(stopped)return;
    try {
      if(previous!==null && playing && !document.hidden)phaseTime=(phaseTime+Math.min((now-previous)/1000,.06)*Math.sqrt(gravity/BASE_G))%CYCLE;
      previous=now;draw();updateUI();raf=requestAnimationFrame(animate);
    } catch(error) {failure(error);}
  }
  function setPlaying(value) {
    playing=value;previous=null;
    byId('playText').textContent=playing?'Pause':'Play';
    byId('play').setAttribute('aria-label',playing?'Pause simulation':'Play simulation');
    byId('playIcon').innerHTML=playing?'<path d="M5 3v10M11 3v10"/>':'<path d="m5 3 8 5-8 5Z"/>';
    byId('statusText').textContent=playing?'IN MOTION':'PAUSED';byId('status').classList.toggle('paused',!playing);
  }
  function reset() {phaseTime=0;previous=null;updateUI(true);draw();}
  function styleRange(input) {input.style.setProperty('--fill',((Number(input.value)-Number(input.min))/(Number(input.max)-Number(input.min))*100)+'%');}
  function guard(fn) {return function(...args){try{fn(...args);}catch(error){failure(error);}};}
  countInput.addEventListener('input',guard(()=>{count=Number(countInput.value);styleRange(countInput);rebuild();reset();}));
  gravityInput.addEventListener('input',guard(()=>{gravity=Number(gravityInput.value);styleRange(gravityInput);byId('gravityValue').innerHTML=gravity.toFixed(1)+'<small>m/s²</small>';gravityInput.setAttribute('aria-valuetext',gravity.toFixed(1)+' meters per second squared');canvas.setAttribute('aria-label',count+' pendulums with different lengths swinging at gravity '+gravity.toFixed(1)+' meters per second squared.');updateUI(true);draw();}));
  byId('play').addEventListener('click',guard(()=>setPlaying(!playing)));
  byId('reset').addEventListener('click',guard(reset));
  byId('trails').addEventListener('click',guard(()=>{trails=!trails;byId('trails').setAttribute('aria-pressed',String(trails));draw();}));
  window.addEventListener('keydown',guard(event=>{
    if(event.target instanceof Element && event.target.closest('input,button,textarea,select,[contenteditable]'))return;
    if(event.code==='Space'){event.preventDefault();setPlaying(!playing);}else if(event.key.toLowerCase()==='r')reset();
  }));
  window.addEventListener('resize',guard(resize));
  document.addEventListener('visibilitychange',()=>{previous=null;});
  window.addEventListener('unload',()=>{stopped=true;cancelAnimationFrame(raf);});
  window.addEventListener('pagehide',()=>{cancelAnimationFrame(raf);previous=null;});
  window.addEventListener('pageshow',event=>{if(event.persisted&&!stopped){previous=null;raf=requestAnimationFrame(animate);}});
