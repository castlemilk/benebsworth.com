function animateCount(el){
  var target=parseFloat(el.getAttribute('data-count')), dec=+(el.getAttribute('data-dec')||0),
      suf=el.getAttribute('data-suffix')||'', t0=null, dur=reduced?1:1600;
  function fmt(v){ return dec?v.toFixed(dec):Math.round(v).toLocaleString(); }
  function step(ts){ if(!t0)t0=ts; var p=Math.min((ts-t0)/dur,1), e=1-Math.pow(1-p,3);
    el.textContent=fmt(target*e)+suf; if(p<1) requestAnimationFrame(step); }
  requestAnimationFrame(step);
}

var cv=doc.getElementById('bg'), ctx=cv.getContext('2d'), W=0,H=0,dpr=1,raf=0,t=0;
var parts=[],running=true;
function resize(){
  var r=cv.parentElement.getBoundingClientRect();
  dpr=Math.min(window.devicePixelRatio||1,2);
  W=r.width; H=r.height;
  cv.width=Math.max(1,Math.round(W*dpr)); cv.height=Math.max(1,Math.round(H*dpr));
  ctx.setTransform(dpr,0,0,dpr,0,0);
  var n=Math.max(24,Math.min(60,Math.floor(W*H/16000)));
  parts=[]; for(var i=0;i<n;i++) parts.push({
    x:Math.random()*W, y:Math.random()*H, r:Math.random()*1.6+.5,
    vy:-(Math.random()*.22+.05), vx:(Math.random()-.5)*.1, o:Math.random()*.5+.15});
}
function themeCols(){
  return root.getAttribute('data-theme')==='light'
    ? {g:['rgba(91,75,255,.30)','rgba(13,148,136,.26)','rgba(219,39,119,.24)'],dot:'rgba(60,60,110,.5)',line:'rgba(91,75,255,.35)'}
    : {g:['rgba(124,108,255,.45)','rgba(79,209,197,.35)','rgba(255,110,199,.35)'],dot:'rgba(200,200,230,.55)',line:'rgba(150,140,255,.4)'};
}
function blob(cx,cy,baseR,tt,fill,stroke){
  var N=8,i,pts=[];
  for(i=0;i<N;i++){ var a=i/N*Math.PI*2;
    var r=baseR*(1+.24*Math.sin(tt*.7+i*1.9)+.13*Math.sin(tt*1.35+i*2.7+2));
    pts.push([cx+Math.cos(a)*r,cy+Math.sin(a)*r]); }
  ctx.beginPath();
  var m=[(pts[0][0]+pts[N-1][0])/2,(pts[0][1]+pts[N-1][1])/2];
  ctx.moveTo(m[0],m[1]);
  for(i=0;i<N;i++){ var p=pts[i],q=pts[(i+1)%N];
    ctx.quadraticCurveTo(p[0],p[1],(p[0]+q[0])/2,(p[1]+q[1])/2); }
  ctx.closePath();
  if(fill)ctx.fill();
  if(stroke){ctx.stroke();}
}
function frame(){
  if(!running){raf=0;return;}
  t+=.008;
  var c=themeCols();
  ctx.clearRect(0,0,W,H);
  ctx.globalCompositeOperation=root.getAttribute('data-theme')==='light'?'source-over':'lighter';
  // glow blobs
  var g1=ctx.createRadialGradient(W*.78+Math.sin(t)*40,H*.34+Math.cos(t*.8)*30,0,W*.78,H*.34,W*.42);
  g1.addColorStop(0,c.g[0]);g1.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g1;ctx.fillRect(0,0,W,H);
  var g2=ctx.createRadialGradient(W*.16+Math.cos(t*.6)*50,H*.72,0,W*.16,H*.72,W*.38);
  g2.addColorStop(0,c.g[1]);g2.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g2;ctx.fillRect(0,0,W,H);
  var g3=ctx.createRadialGradient(W*.5+Math.sin(t*.5)*60,H*.15,0,W*.5,H*.15,W*.3);
  g3.addColorStop(0,c.g[2]);g3.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle=g3;ctx.fillRect(0,0,W,H);
  // signature morphing outline blob, right of center
  ctx.globalCompositeOperation='source-over';
  ctx.lineWidth=1.4;ctx.strokeStyle=c.line;
  var lgr=ctx.createLinearGradient(W*.6,0,W,H);
  lgr.addColorStop(0,c.g[0]);lgr.addColorStop(1,c.g[1]);
  ctx.strokeStyle=lgr;
  blob(W*.76,H*.42,Math.min(W,H)*.20,t,null,true);
  ctx.globalAlpha=.5;
  blob(W*.76,H*.42,Math.min(W,H)*.26,t*1.2+4,null,true);
  ctx.globalAlpha=1;
  // particles
  ctx.fillStyle=c.dot;
  for(var i=0;i<parts.length;i++){var p=parts[i];
    p.x+=p.vx;p.y+=p.vy;
    if(p.y<-4){p.y=H+4;p.x=Math.random()*W}
    if(p.x<-4)p.x=W+4; if(p.x>W+4)p.x=-4;
    ctx.globalAlpha=p.o*(0.6+0.4*Math.sin(t*3+i));
    ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,6.2832);ctx.fill();}
  ctx.globalAlpha=1;
  raf=requestAnimationFrame(frame);
}

        // draw once without looping:
        (function(){ var keep=running; running=true; t=0.001; var old=raf;

          frame(); window.requestAnimationFrame=req; raf=0; running=keep;
        })();
      } else { raf=requestAnimationFrame(frame); } } }
