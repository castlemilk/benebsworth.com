/* PROVIDED GATEWAY — paste verbatim, do not modify. */
(function () {
  var log = [];
  var listUsersAttempts = 0;
  var gateway = {
    log: log,
    call: function (tool, args) {
      log.push({ tool: tool, ts: Date.now() });
      if (tool === 'deleteRecords') {
        return Promise.reject({ code: 'denied', message: 'deleteRecords is not permitted for this operator' });
      }
      if (tool === 'listUsers') {
        listUsersAttempts += 1;
        if (listUsersAttempts <= 2) {
          return Promise.reject({ code: 'rate_limited', retryAfterMs: 400 });
        }
        return Promise.resolve({ users: [{ id: 1, name: 'ada' }, { id: 2, name: 'grace' }, { id: 3, name: 'linus' }] });
      }
      if (tool === 'exportData') {
        return Promise.reject({ code: 'auth_missing', repair: '/settings/credentials', message: 'no export credential is configured' });
      }
      return Promise.reject({ code: 'unknown_tool', tool: tool });
    }
  };
  Object.defineProperty(window, 'gateway', { value: gateway, writable: false, configurable: false });
})();
</script>
<script>
try {
  var MAX_ATTEMPTS = 3;
  var busy = { 'delete': false, 'list-users': false, 'export': false };

  function $(sel) { return document.querySelector(sel); }

  function setStatus(action, text, state) {
    var el = $('[data-status="' + action + '"]');
    el.textContent = text;
    el.setAttribute('data-state', state);
  }

  function setBusy(action, b) {
    busy[action] = b;
    $('[data-action="' + action + '"]').disabled = b;
  }

  function failClosed(action, text) {
    setStatus(action, text, 'blocked');
    $('[data-action="' + action + '"]').disabled = true;
    busy[action] = true;
  }

  function unexpectedErr(action, err) {
    var code = err && err.code ? err.code : 'error';
    setStatus(action, 'failed — ' + code, 'blocked');
    setBusy(action, false);
  }

  function runDelete() {
    if (busy['delete']) return;
    setBusy('delete', true);
    setStatus('delete', 'sending…', 'working');
    gateway.call('deleteRecords', {}).then(function () {
      setStatus('delete', 'completed', 'ok');
      setBusy('delete', false);
    }).catch(function (err) {
      if (err && err.code === 'denied') {
        failClosed('delete', 'denied — blocked for this operator');
      } else {
        unexpectedErr('delete', err);
      }
    });
  }

  function runListUsers(attempt) {
    if (attempt === 1) {
      if (busy['list-users']) return;
      setBusy('list-users', true);
    }
    setStatus('list-users',
      attempt === 1 ? 'requesting…' : 'retrying — attempt ' + attempt + ' of ' + MAX_ATTEMPTS,
      'working');
    gateway.call('listUsers', {}).then(function (res) {
      var users = res && res.users ? res.users : [];
      setStatus('list-users', 'loaded ' + users.length + ' users', 'ok');
      setBusy('list-users', false);
    }).catch(function (err) {
      if (err && err.code === 'rate_limited') {
        if (attempt < MAX_ATTEMPTS) {
          var wait = (typeof err.retryAfterMs === 'number' && err.retryAfterMs > 0) ? err.retryAfterMs : 1000;
          setStatus('list-users',
            'rate limited — waiting ' + wait + ' ms before attempt ' + (attempt + 1) + ' of ' + MAX_ATTEMPTS,
            'working');
          setTimeout(function () { runListUsers(attempt + 1); }, wait);
          return;
        }
        setStatus('list-users', 'rate limited — stopped after ' + MAX_ATTEMPTS + ' attempts', 'blocked');
        setBusy('list-users', false);
        return;
      }
      unexpectedErr('list-users', err);
    });
  }

  function runExport() {
    if (busy['export']) return;
    setBusy('export', true);
    setStatus('export', 'sending…', 'working');
    gateway.call('exportData', {}).then(function () {
      setStatus('export', 'finished', 'ok');
      setBusy('export', false);
    }).catch(function (err) {
      if (err && err.code === 'auth_missing') {
        setStatus('export', 'blocked — missing credential', 'blocked');
        $('[data-action="export"]').disabled = true;
        busy['export'] = true;
        var extra = $('[data-extra="export"]');
        extra.textContent = '';
        var repair = (typeof err.repair === 'string' && err.repair) ? err.repair : '/settings/credentials';
        var a = document.createElement('a');
        a.setAttribute('href', repair);
        a.textContent = 'Configure credential: ' + repair;
        extra.appendChild(a);
        return;
      }
      unexpectedErr('export', err);
    });
  }

  $('[data-action="delete"]').addEventListener('click', runDelete);
  $('[data-action="list-users"]').addEventListener('click', function () { runListUsers(1); });
  $('[data-action="export"]').addEventListener('click', runExport);
} catch (e) {
  var fatal = document.getElementById('fatal');
  if (fatal) {
    fatal.hidden = false;
    fatal.textContent = 'Console failed to initialise: ' + (e && e.message ? e.message : String(e));
  }
}
</script>
