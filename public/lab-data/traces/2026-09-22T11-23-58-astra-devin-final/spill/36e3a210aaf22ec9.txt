/* PROVIDED GATEWAY — paste verbatim, do not modify. */
(function () {
  var log = [];
  var listUsersAttempts = 0;
  var gateway = {
    log: log,
    call: function (tool, args) {
      log.push({ tool: tool, ts: Date.now() });
      if (tool === 'deleteRecords') {
        return Promise.reject({ code: 'denied', message: 'deleteRecords is not permitted for this operator' });
      }
      if (tool === 'listUsers') {
        listUsersAttempts += 1;
        if (listUsersAttempts <= 2) {
          return Promise.reject({ code: 'rate_limited', retryAfterMs: 400 });
        }
        return Promise.resolve({ users: [{ id: 1, name: 'ada' }, { id: 2, name: 'grace' }, { id: 3, name: 'linus' }] });
      }
      if (tool === 'exportData') {
        return Promise.reject({ code: 'auth_missing', repair: '/settings/credentials', message: 'no export credential is configured' });
      }
      return Promise.reject({ code: 'unknown_tool', tool: tool });
    }
  };
  Object.defineProperty(window, 'gateway', { value: gateway, writable: false, configurable: false });
})();
</script>
<script>
try {
  var MAX_ATTEMPTS = 3;

  function statusEl(key) { return document.querySelector('[data-status="' + key + '"]'); }

  function setStatus(key, text, state) {
    var el = statusEl(key);
    el.textContent = text;
    el.setAttribute('data-state', state);
  }

  function showRepair(id, path) {
    var a = document.getElementById(id);
    if (a) {
      a.setAttribute('href', path);
      a.textContent = path;
      a.hidden = false;
    }
  }

  function wire(actionKey, tool, opts) {
    var btn = document.querySelector('button[data-action="' + actionKey + '"]');
    btn.addEventListener('click', function () {
      if (btn.disabled) return;
      btn.disabled = true;
      var attempt = 0;
      function go() {
        attempt += 1;
        if (attempt === 1) setStatus(actionKey, opts.working, 'working');
        gateway.call(tool, opts.args || {}).then(function (res) {
          opts.onOk(res);
        }, function (err) {
          err = err || {};
          if (err.code === 'rate_limited') {
            if (attempt < MAX_ATTEMPTS) {
              var wait = Math.max(0, Number(err.retryAfterMs) || 0);
              setStatus(actionKey, 'rate limited — retrying in ' + wait + 'ms (attempt ' + (attempt + 1) + ' of ' + MAX_ATTEMPTS + ')', 'working');
              setTimeout(go, wait);
            } else {
              setStatus(actionKey, 'blocked — still rate limited after ' + MAX_ATTEMPTS + ' attempts', 'bad');
            }
          } else if (err.code === 'denied') {
            setStatus(actionKey, 'denied — ' + (err.message || 'this operator is not permitted'), 'bad');
          } else if (err.code === 'auth_missing') {
            setStatus(actionKey, 'blocked — ' + (err.message || 'a credential is missing') + '; repair it at the link below', 'bad');
            if (err.repair && opts.repairId) showRepair(opts.repairId, err.repair);
          } else {
            setStatus(actionKey, 'error — ' + (err.message || err.code || 'unknown failure'), 'bad');
          }
        });
      }
      go();
    });
  }

  wire('delete', 'deleteRecords', {
    working: 'requesting deletion…',
    onOk: function () { setStatus('delete', 'deleted', 'ok'); }
  });

  wire('list-users', 'listUsers', {
    working: 'loading users…',
    onOk: function (res) {
      var users = (res && res.users) || [];
      var names = users.map(function (u) { return u.name; }).join(', ');
      setStatus('list-users', 'loaded ' + users.length + ' users — ' + names, 'ok');
    }
  });

  wire('export', 'exportData', {
    working: 'requesting export…',
    repairId: 'export-repair',
    onOk: function () { setStatus('export', 'finished', 'ok'); }
  });
} catch (e) {
  var d = document.getElementById('fatal');
  if (!d) {
    d = document.createElement('div');
    d.setAttribute('role', 'alert');
    document.body.appendChild(d);
  }
  d.hidden = false;
  d.textContent = 'Console failed to start: ' + (e && e.message ? e.message : e);
}
</script>
