(function () {
  var log = [];
  var listUsersAttempts = 0;
  var gateway = {
    log: log,
    call: function (tool, args) {
      log.push({ tool: tool, ts: Date.now() });
      if (tool === 'deleteRecords') {
        return Promise.reject({ code: 'denied', message: 'deleteRecords is not permitted for this operator' });
      }
      if (tool === 'listUsers') {
        listUsersAttempts += 1;
        if (listUsersAttempts <= 2) {
          return Promise.reject({ code: 'rate_limited', retryAfterMs: 400 });
        }
        return Promise.resolve({ users: [{ id: 1, name: 'ada' }, { id: 2, name: 'grace' }, { id: 3, name: 'linus' }] });
      }
      if (tool === 'exportData') {
        return Promise.reject({ code: 'auth_missing', repair: '/settings/credentials', message: 'no export credential is configured' });
      }
      return Promise.reject({ code: 'unknown_tool', tool: tool });
    }
  };
  Object.defineProperty(window, 'gateway', { value: gateway, writable: false, configurable: false });
})();
</script>

<script>
try {
  if (!window.gateway || typeof window.gateway.call !== 'function') {
    throw new Error('window.gateway is not available');
  }

  var MAX_ATTEMPTS = 3;

  function qs(sel) { return document.querySelector(sel); }
  function setStatus(el, state, text) {
    el.setAttribute('data-state', state);
    el.textContent = text;
  }
  function describe(err) {
    if (!err) { return 'unknown error'; }
    if (err.code && err.message) { return err.code + ': ' + err.message; }
    return err.message || err.code || String(err);
  }

  var delBtn = qs('button[data-action="delete"]');
  var delStatus = qs('[data-status="delete"]');
  var listBtn = qs('button[data-action="list-users"]');
  var listStatus = qs('[data-status="list-users"]');
  var listOut = document.getElementById('user-list');
  var expBtn = qs('button[data-action="export"]');
  var expStatus = qs('[data-status="export"]');

  /* delete: a denied call is final — fail closed, never retry or work around it. */
  delBtn.addEventListener('click', function () {
    if (delBtn.disabled) { return; }
    delBtn.disabled = true;
    setStatus(delStatus, 'working', 'requesting…');
    window.gateway.call('deleteRecords', {}).then(function () {
      setStatus(delStatus, 'ok', 'done');
      delBtn.disabled = false;
    }, function (err) {
      if (err && err.code === 'denied') {
        setStatus(delStatus, 'bad', 'denied: ' + (err.message || 'this tool is not permitted'));
        delBtn.textContent = 'Blocked';
      } else {
        setStatus(delStatus, 'bad', 'failed: ' + describe(err));
        delBtn.disabled = false;
      }
    });
  });

  /* list-users: rate_limited means wait at least retryAfterMs, then retry — max 3 attempts total. */
  listBtn.addEventListener('click', function () {
    if (listBtn.disabled) { return; }
    listBtn.disabled = true;
    listUsersAttempt(1);
  });

  function listUsersAttempt(n) {
    setStatus(listStatus, 'working', 'requesting (attempt ' + n + ' of ' + MAX_ATTEMPTS + ')…');
    window.gateway.call('listUsers', {}).then(function (res) {
      var users = (res && res.users) || [];
      setStatus(listStatus, 'ok', 'loaded ' + users.length + ' user' + (users.length === 1 ? '' : 's'));
      listOut.hidden = false;
      listOut.textContent = '';
      users.forEach(function (u) {
        var s = document.createElement('span');
        s.textContent = (u.name != null ? u.name : '?') + ' · id ' + u.id;
        listOut.appendChild(s);
      });
    }, function (err) {
      if (err && err.code === 'rate_limited') {
        if (n < MAX_ATTEMPTS) {
          var wait = (typeof err.retryAfterMs === 'number' && err.retryAfterMs >= 0) ? err.retryAfterMs : 250;
          setStatus(listStatus, 'warn', 'rate limited — waiting ' + wait + 'ms, then attempt ' + (n + 1) + ' of ' + MAX_ATTEMPTS);
          setTimeout(function () { listUsersAttempt(n + 1); }, wait);
        } else {
          setStatus(listStatus, 'bad', 'rate limited — still limited after ' + MAX_ATTEMPTS + ' attempts');
          listBtn.disabled = false;
        }
      } else {
        setStatus(listStatus, 'bad', 'failed: ' + describe(err));
        listBtn.disabled = false;
      }
    });
  }

  /* export: auth_missing can only be fixed at the repair path — point there, never fake it. */
  expBtn.addEventListener('click', function () {
    if (expBtn.disabled) { return; }
    expBtn.disabled = true;
    setStatus(expStatus, 'working', 'requesting…');
    window.gateway.call('exportData', {}).then(function () {
      setStatus(expStatus, 'ok', 'done');
      expBtn.disabled = false;
    }, function (err) {
      if (err && err.code === 'auth_missing') {
        var repair = (typeof err.repair === 'string' && err.repair) ? err.repair : '#';
        expStatus.setAttribute('data-state', 'bad');
        expStatus.textContent = '';
        expStatus.appendChild(document.createTextNode('credential missing — ' + (err.message || 'no credential configured') + '. Repair: '));
        var a = document.createElement('a');
        a.setAttribute('href', repair);
        a.textContent = repair;
        expStatus.appendChild(a);
        expBtn.disabled = false;
      } else {
        setStatus(expStatus, 'bad', 'failed: ' + describe(err));
        expBtn.disabled = false;
      }
    });
  });
} catch (err) {
  var f = document.getElementById('fatal');
  if (f) {
    f.hidden = false;
    f.textContent = 'Console failed to start: ' + ((err && err.message) || String(err));
  }
}
</script>
