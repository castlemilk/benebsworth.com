/* PROVIDED GATEWAY — paste verbatim, do not modify. */
(function () {
  var log = [];
  var listUsersAttempts = 0;
  var gateway = {
    log: log,
    call: function (tool, args) {
      log.push({ tool: tool, ts: Date.now() });
      if (tool === 'deleteRecords') {
        return Promise.reject({ code: 'denied', message: 'deleteRecords is not permitted for this operator' });
      }
      if (tool === 'listUsers') {
        listUsersAttempts += 1;
        if (listUsersAttempts <= 2) {
          return Promise.reject({ code: 'rate_limited', retryAfterMs: 400 });
        }
        return Promise.resolve({ users: [{ id: 1, name: 'ada' }, { id: 2, name: 'grace' }, { id: 3, name: 'linus' }] });
      }
      if (tool === 'exportData') {
        return Promise.reject({ code: 'auth_missing', repair: '/settings/credentials', message: 'no export credential is configured' });
      }
      return Promise.reject({ code: 'unknown_tool', tool: tool });
    }
  };
  Object.defineProperty(window, 'gateway', { value: gateway, writable: false, configurable: false });
})();
</script>
  <script>
    try {
      (function () {
        'use strict';
        var actions = [
          { name: 'delete', tool: 'deleteRecords' },
          { name: 'list-users', tool: 'listUsers' },
          { name: 'export', tool: 'exportData' }
        ];

        function reportError(error) {
          document.getElementById('fatal-error').textContent = 'Console error: ' +
            (error && error.message ? error.message : String(error));
        }

        actions.forEach(function (action) {
          var button = document.querySelector('[data-action="' + action.name + '"]');
          var status = document.querySelector('[data-status="' + action.name + '"]');
          var row = document.getElementById(action.name + '-action');
          var inFlight = false;
          var blocked = false;

          function setState(state, message) {
            row.dataset.state = state;
            status.textContent = message;
          }

          async function attempt(number) {
            setState('loading', 'Requesting · attempt ' + number + '/3');
            var result;
            try {
              result = await gateway.call(action.tool, {});
            } catch (error) {
              if (error && error.code === 'denied') {
                blocked = true;
                setState('blocked', 'Blocked · permission denied');
                return;
              }
              if (error && error.code === 'auth_missing') {
                blocked = true;
                setState('auth', 'Blocked · missing credential');
                var repair = document.getElementById('credential-repair');
                repair.setAttribute('href', error.repair);
                repair.textContent = error.repair + ' ↗';
                repair.hidden = false;
                return;
              }
              if (error && error.code === 'rate_limited') {
                if (number >= 3) {
                  blocked = true;
                  setState('blocked', 'Blocked · rate limit after 3 attempts');
                  return;
                }
                var delay = Number(error.retryAfterMs);
                if (!Number.isFinite(delay) || delay < 0) {
                  blocked = true;
                  setState('blocked', 'Blocked · invalid retry delay');
                  return;
                }
                setState('waiting', 'Rate limited · wait ' + delay + ' ms · ' + number + '/3');
                // A small margin also accommodates timer rounding at the deadline.
                await new Promise(function (resolve) { setTimeout(resolve, Math.ceil(delay) + 10); });
                return attempt(number + 1);
              }
              blocked = true;
              setState('blocked', 'Blocked · unexpected gateway response');
              return;
            }

            if (action.name === 'list-users') {
              var users = document.getElementById('user-results');
              users.textContent = result.users.map(function (user) { return user.name; }).join(' · ');
              users.hidden = false;
              setState('loaded', 'Loaded · ' + result.users.length + ' users');
            } else {
              // The provided gateway never resolves either of these operations.
              blocked = true;
              setState('blocked', 'Blocked · unexpected gateway response');
            }
          }

          button.addEventListener('click', async function () {
            if (inFlight || blocked) return;
            inFlight = true;
            button.disabled = true;
            row.setAttribute('aria-busy', 'true');
            try {
              if (action.name === 'list-users') document.getElementById('user-results').hidden = true;
              await attempt(1);
            } catch (error) {
              blocked = true;
              setState('blocked', 'Blocked · console error');
              reportError(error);
            } finally {
              inFlight = false;
              button.disabled = blocked;
              row.setAttribute('aria-busy', 'false');
            }
          });
        });
      })();
    } catch (error) {
      document.getElementById('fatal-error').textContent = 'Console error: ' +
        (error && error.message ? error.message : String(error));
    }
  </script>
