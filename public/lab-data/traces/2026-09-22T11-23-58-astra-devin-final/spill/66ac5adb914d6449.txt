var GATES=[0.25,0.5,0.75];
function lapLogic(c,now){
  if(c.finished)return;
  var base=c.lap*L;
  if(c.gate<3&&c.prog>base+GATES[c.gate]*L)c.gate++;
  if(c.gate>=3&&c.prog>base+L){
    c.lap++;

    }
  }

}

  var sAbs=Math.abs(c.speed);
  var maxSteer=0.6-0.42*clamp(sAbs/620,0,1);
  var sIn=typeof inp.steer==='number'?inp.steer:((inp.right?1:0)-(inp.left?1:0));
  c.steer+=(sIn*maxSteer-c.steer)*Math.min(1,dt*9);
  var yaw=c.speed/42*Math.tan(c.steer);
  var latAcc=Math.abs(c.speed*yaw);
  var maxLat=off?560:1150;
  c.slide=false;
  if(latAcc>maxLat&&sAbs>10){yaw*=maxLat/latAcc;c.slide=true;c.speed-=c.speed*0.85*dt;}
  c.heading+=yaw*dt;
  if(curb&&sAbs>140){c.heading+=(rand()-0.5)*0.03;c.speed-=c.speed*0.25*dt;}
  if(off&&sAbs>60){c.speed-=c.speed*0.4*dt;}

  c.x+=Math.cos(c.heading)*c.speed*dt;
  c.y+=Math.sin(c.heading)*c.speed*dt;
  c.distance+=Math.max(0,c.speed)*dt;

  locate(c);
  if(c.lat>620){ // soft world edge: ease back toward the circuit
    var a2=pts[c.idx];var dx=a2.x-c.x,dy=a2.y-c.y,dl=Math.hypot(dx,dy)||1;
    var pull=(c.lat-620)*1.4*dt;c.x+=dx/dl*pull;c.y+=dy/dl*pull;locate(c);
  }
  lapLogic(c,now);

  if(c.isPlayer){
    if(c.dprog/dt<-60&&sAbs>80)c.wrongT=0.9;else c.wrongT=Math.max(0,c.wrongT-dt);
    if(c.slide||(input.down&&c.speed>220))addSkids(c);
  }
}
function aiInput(c){
  var ahead=Math.round((60+sAbs(c)*0.45)/segAvg);
  var ti=(c.idx+ahead)%NP,q=pts[ti];
  var wob=Math.sin(c.prog*0.001+c.lane)*10;
  var tx=q.x+q.nx*(c.lane*0.8+wob),ty=q.y+q.ny*(c.lane*0.8+wob);
  var want=Math.atan2(ty-c.y,tx-c.x);
  var steer=clamp(wrap(want-c.heading)*2.4,-1,1);
  var km=0.0001;
  for(var o=2;o<ahead+26;o+=3){var kk=Math.abs(PW(c.idx+o).k);if(kk>km)km=kk;}
  var vt=Math.min(c.vmax,Math.sqrt(1150/km));if(vt<150)vt=150;
  var d2p=(tx-c.x),d2q=(ty-c.y),near=Math.hypot(d2p,d2q);
  return {up:c.speed<vt-8,down:c.speed>vt+40,steer:steer};
}
function sAbs(c){return Math.abs(c.speed);}
function addSkids(c){
  var hl=Math.cos(c.heading),hy=Math.sin(c.heading),nx=-hy,ny=hl;
  for(var s=-1;s<=1;s+=2){
    skids.push({x1:c.x-hl*16+nx*s*9,y1:c.y-hy*16+ny*s*9,x2:c.x-hl*13+nx*s*9,y2:c.y-hy*13+ny*s*9,a:0.55});
  }
  if(skids.length>900)skids.splice(0,skids.length-900);
}
function collide(){
  for(var i=0;i<cars.length;i++)for(var j=i+1;j<cars.length;j++){
    var a=cars[i],b=cars[j],dx=b.x-a.x,dy=b.y-a.y,d=Math.hypot(dx,dy);
    if(d<R*2&&d>0.01){
      var nx=dx/d,ny=dy/d,ov=(R*2-d)/2;
      a.x-=nx*ov;a.y-=ny*ov;b.x+=nx*ov;b.y+=ny*ov;
      var avx=Math.cos(a.heading)*a.speed,avy=Math.sin(a.heading)*a.speed;
      var bvx=Math.cos(b.heading)*b.speed,bvy=Math.sin(b.heading)*b.speed;
      var rel=(bvx-avx)*nx+(bvy-avy)*ny;
      if(rel<0){
        var imp=-rel*0.55;
        var nva={x:avx-nx*imp,y:avy-ny*imp},nvb={x:bvx+nx*imp,y:bvy+ny*imp};
        a.speed=nva.x*Math.cos(a.heading)+nva.y*Math.sin(a.heading);
        b.speed=nvb.x*Math.cos(b.heading)+nvb.y*Math.sin(b.heading);
        a.speed*=0.94;b.speed*=0.94;
        a.heading+=(rand()-0.5)*0.04;b.heading+=(rand()-0.5)*0.04;
        if(a.isPlayer||b.isPlayer){addSkids(a.isPlayer?a:b);}
      }
      locate(a);locate(b);
    }
  }
}

function minimap(){
  var mw=Math.min(150,vw*0.32),mh=mw*(WH/WW),mx=vw-mw-12,my=48;
  var sc=Math.min((mw-14)/WW,(mh-14)/WH);
  var ox=mx+7+(WW*sc<mw-14?(mw-14-WW*sc)/2:0)-minX*sc;
  var oy=my+7+(WH*sc<mh-14?(mh-14-WH*sc)/2:0)-minY*sc;
  ctx.fillStyle='rgba(8,9,14,0.55)';
  ctx.beginPath();
  if(ctx.roundRect)ctx.roundRect(mx,my,mw,mh,10);else ctx.rect(mx,my,mw,mh);
  ctx.fill();
  ctx.strokeStyle='rgba(255,255,255,0.14)';ctx.lineWidth=1;ctx.stroke();
