  <p id="status" aria-live="polite"></p>
  <div id="board" role="grid" aria-label="Tic-tac-toe board"></div>
  <button id="restart" type="button">Restart</button>
</main>
<script>
try {
  var board = document.getElementById('board');
  var statusEl = document.getElementById('status');
  var restartBtn = document.getElementById('restart');

  var LINES = [
    [0, 1, 2], [3, 4, 5], [6, 7, 8],
    [0, 3, 6], [1, 4, 7], [2, 5, 8],
    [0, 4, 8], [2, 4, 6]
  ];

  var cells = [];
  var state = [];
  var current = 'X';
  var over = false;

  for (var i = 0; i < 9; i++) {
    var btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'cell';
    btn.setAttribute('role', 'gridcell');
    btn.setAttribute('aria-label', 'Cell ' + (i + 1));
    (function (idx) {
      btn.addEventListener('click', function () { onMove(idx); });
    })(i);
    board.appendChild(btn);
    cells.push(btn);
  }

  function setStatus(html) {
    statusEl.innerHTML = html;
  }

  function markSpan(p) {
    return '<span class="' + p.toLowerCase() + '">' + p + '</span>';
  }

  function findWin(p) {
    for (var k = 0; k < LINES.length; k++) {
      var line = LINES[k];
      if (state[line[0]] === p && state[line[1]] === p && state[line[2]] === p) {
        return line;
      }
    }
    return null;
  }

  function onMove(idx) {
    if (over || state[idx]) return;
    state[idx] = current;
    var cell = cells[idx];
    cell.textContent = current;
    cell.classList.add(current === 'X' ? 'mark-x' : 'mark-o');
    cell.disabled = true;

    var winLine = findWin(current);
    if (winLine) {
      over = true;
      for (var k = 0; k < winLine.length; k++) {
        cells[winLine[k]].classList.add('win-line');
      }
      for (var j = 0; j < cells.length; j++) cells[j].disabled = true;
      setStatus(markSpan(current) + ' <span class="win">wins!</span>');
      return;
    }
    if (state.every(function (s) { return s; })) {
      over = true;
      for (var j2 = 0; j2 < cells.length; j2++) cells[j2].disabled = true;
      setStatus('It\u2019s a draw!');
      return;
    }
    current = current === 'X' ? 'O' : 'X';
    setStatus(markSpan(current) + '\u2019s turn');
  }

  function restart() {
    state = [];
    current = 'X';
    over = false;
    for (var j = 0; j < cells.length; j++) {
      cells[j].textContent = '';
      cells[j].disabled = false;
      cells[j].classList.remove('mark-x', 'mark-o', 'win-line');
    }
    setStatus(markSpan('X') + '\u2019s turn');
  }

  restartBtn.addEventListener('click', restart);
  restart();
} catch (err) {
  var alert = document.createElement('div');
  alert.setAttribute('role', 'alert');
  alert.textContent = 'Failed to start the game: ' + (err && err.message ? err.message : String(err));
  document.body.appendChild(alert);
}
</script>
