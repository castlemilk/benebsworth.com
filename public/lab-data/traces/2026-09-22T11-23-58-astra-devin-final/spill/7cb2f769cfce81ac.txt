  <script>
  (function () {
    "use strict";
    try {
      var statusEl = document.getElementById("status");
      var restartBtn = document.getElementById("restart");
      var cells = Array.prototype.slice.call(
        document.querySelectorAll("#board .cell")
      );

      var WIN_LINES = [
        [0, 1, 2], [3, 4, 5], [6, 7, 8],
        [0, 3, 6], [1, 4, 7], [2, 5, 8],
        [0, 4, 8], [2, 4, 6]
      ];

      var board = ["", "", "", "", "", "", "", "", ""];
      var currentPlayer = "X";
      var gameOver = false;

      function setStatus(text, cls) {
        statusEl.textContent = text;
        statusEl.className = "status" + (cls ? " " + cls : "");
      }

      function findWin() {
        for (var i = 0; i < WIN_LINES.length; i++) {
          var a = WIN_LINES[i][0], b = WIN_LINES[i][1], c = WIN_LINES[i][2];
          if (board[a] && board[a] === board[b] && board[a] === board[c]) {
            return WIN_LINES[i];
          }
        }
        return null;
      }

      function onCellClick(index) {
        if (gameOver || board[index] !== "") return;

        board[index] = currentPlayer;
        var cell = cells[index];
        cell.textContent = currentPlayer;
        cell.classList.add(currentPlayer === "X" ? "mark-x" : "mark-o");
        cell.disabled = true;

        var line = findWin();
        if (line) {
          gameOver = true;
          for (var i = 0; i < line.length; i++) {
            cells[line[i]].classList.add("winner");
          }
          for (var j = 0; j < cells.length; j++) {
            cells[j].disabled = true;
          }
          setStatus(currentPlayer + " wins", "win");
          return;
        }

        if (board.indexOf("") === -1) {
          gameOver = true;
          setStatus("It’s a draw", "draw");
          return;
        }

        currentPlayer = currentPlayer === "X" ? "O" : "X";
        setStatus(currentPlayer + "’s turn", "");
      }

      function restart() {
        board = ["", "", "", "", "", "", "", "", ""];
        currentPlayer = "X";
        gameOver = false;
        for (var i = 0; i < cells.length; i++) {
          cells[i].textContent = "";
          cells[i].classList.remove("mark-x", "mark-o", "winner");
          cells[i].disabled = false;
        }
        setStatus("X’s turn", "");
      }

      cells.forEach(function (cell, index) {
        cell.addEventListener("click", function () {
          onCellClick(index);
        });
      });
      restartBtn.addEventListener("click", restart);
    } catch (err) {
      var fatal = document.getElementById("fatal");
      if (fatal) {
        fatal.hidden = false;
        fatal.textContent = "Something went wrong: " + (err && err.message ? err.message : String(err));
      }
    }
  })();
  </script>
