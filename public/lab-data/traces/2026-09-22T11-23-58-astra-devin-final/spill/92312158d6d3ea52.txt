    var wx2=b.x+lx*ca+lz*sa,wy2=b.y+ly,wz2=b.z-lx*sa+lz*ca;
    wvx[i]=wx2;wvy[i]=wy2;wvz[i]=wz2;
    var rx2=wx2-CAM[0],ry2=wy2-CAM[1],rz2=wz2-CAM[2];

    cvy[i]=rx2*UP[0]+ry2*UP[1]+rz2*UP[2];

  }
  for(var f=0;f<6;f++){
    var n=CN[f];
    var nx=n[0]*ca+n[2]*sa,ny=n[1],nz=-n[0]*sa+n[2]*ca;
    var f0=CF[f][0];

    var inten=AMB+DIF*Math.max(0,nx*L[0]+ny*L[1]+nz*L[2])+b.e;
    if(inten>1.35)inten=1.35;
    faces.push({d:depth*0.25,p:pts,r:col[0]*inten,g:col[1]*inten,b:col[2]*inten,a:b.a});
  }
}

function projectDir(dx,dy,dz){
  // project a point far along dir from camera (for sun/flare)
  var px=CAM[0]+dx*400,py=CAM[1]+dy*400,pz=CAM[2]+dz*400;
  var rx2=px-CAM[0],ry2=py-CAM[1],rz2=pz-CAM[2];
  var cx2=rx2*RT[0]+ry2*RT[1]+rz2*RT[2];
  var cy2=rx2*UP[0]+ry2*UP[1]+rz2*UP[2];
  var cz2=rx2*FW[0]+ry2*FW[1]+rz2*FW[2];
  if(cz2<1)return null;
  return[W*0.5+FOCAL*cx2/cz2,H*0.5-FOCAL*cy2/cz2];
}
