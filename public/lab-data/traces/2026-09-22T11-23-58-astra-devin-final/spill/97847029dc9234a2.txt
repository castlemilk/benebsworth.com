<script>
try{
const TAU=Math.PI*2, G0=9.81, T=30, K=27, AMP=0.36;
const canvas=document.getElementById('c'), ctx=canvas.getContext('2d');
const $=id=>document.getElementById(id);
let W=0,H=0,dpr=1,pivotY=0,lenMax=0,mx=36;
let N=24,g=G0,tEff=0,paused=false,trails=true,rafId=0,last=-1;
let om=[],px=[],dl=[],hue=[],spacing=40;

function build(){
  om=[];hue=[];
  for(let i=0;i<N;i++){
    om.push(TAU*(K+i)/T);           // pendulum i does K+i swings per cycle T at 1 g
    hue.push(190+150*(N>1?i/(N-1):0));
  }
  layout();
}
function layout(){
  pivotY=Math.max(54,H*0.10);
  lenMax=Math.max(80,H-pivotY-Math.max(30,H*0.08));
  mx=Math.max(36,W*0.07);
  spacing=N>1?(W-2*mx)/(N-1):W;
  px=[];dl=[];
  const L=om.map(w=>G0/(w*w));      // true length in metres: L = g/w^2
  const lmin=Math.min.apply(null,L),lmax=Math.max.apply(null,L);
  for(let i=0;i<N;i++){
    px.push(N>1?mx+i*spacing:W/2);
    const u=lmax>lmin?(L[i]-lmin)/(lmax-lmin):1;
    dl.push(lenMax*(0.80+0.20*u));  // compressed length fan for display
  }
}
function resize(){
  dpr=Math.min(window.devicePixelRatio||1,2);
  W=window.innerWidth;H=window.innerHeight;
  canvas.width=Math.round(W*dpr);canvas.height=Math.round(H*dpr);
  ctx.setTransform(dpr,0,0,dpr,0,0);
  layout();
  render();
}
function render(){
  ctx.fillStyle=trails?'rgba(11,11,16,0.16)':'#0b0b10';
  ctx.fillRect(0,0,W,H);
  // support bar
  ctx.strokeStyle='rgba(236,236,240,0.16)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(px[0]-22,pivotY);ctx.lineTo(px[N-1]+22,pivotY);ctx.stroke();
  // bob positions
  const pts=new Array(N);
  for(let i=0;i<N;i++){
    const th=AMP*Math.cos(om[i]*tEff);
    pts[i]=[px[i]+dl[i]*Math.sin(th),pivotY+dl[i]*Math.cos(th)];
  }
  // strings
  ctx.strokeStyle='rgba(236,236,240,0.18)';ctx.lineWidth=1;
  ctx.beginPath();
  for(let i=0;i<N;i++){ctx.moveTo(px[i],pivotY);ctx.lineTo(pts[i][0],pts[i][1]);}
  ctx.stroke();
  // emergent wave polyline through bob centres
  ctx.strokeStyle='rgba(255,255,255,0.35)';ctx.lineWidth=1.4;
  ctx.beginPath();
  for(let i=0;i<N;i++){i?ctx.lineTo(pts[i][0],pts[i][1]):ctx.moveTo(pts[i][0],pts[i][1]);}
  ctx.stroke();
  // pivots
  ctx.fillStyle='rgba(236,236,240,0.45)';
  ctx.beginPath();
  for(let i=0;i<N;i++){ctx.moveTo(px[i]+2,pivotY);ctx.arc(px[i],pivotY,2,0,TAU);}
  ctx.fill();
  // bobs (glow + core)
  const r=Math.max(3,Math.min(11,spacing*0.34));
  for(let i=0;i<N;i++){
    const x=pts[i][0],y=pts[i][1],h=hue[i];
    ctx.fillStyle='hsla('+h+',90%,65%,0.22)';
    ctx.beginPath();ctx.arc(x,y,r*2.3,0,TAU);ctx.fill();
    ctx.fillStyle='hsl('+h+',90%,68%)';
    ctx.beginPath();ctx.arc(x,y,r,0,TAU);ctx.fill();
  }
}
function frame(ts){
  rafId=requestAnimationFrame(frame);
  if(last<0)last=ts;
  const dt=Math.min(0.05,(ts-last)/1000);last=ts;
  const s=Math.sqrt(g/G0);          // w_i scales by sqrt(g): same rig, different planet
  if(!paused)tEff+=s*dt;
  render();
  const p=(tEff%T)/T;
  $('syncbar').style.width=(p*100).toFixed(1)+'%';
  $('syncTxt').textContent='resync in '+(((1-p)*T)/s).toFixed(1)+' s';
}
function planet(v){
  if(v<2)return'Moon';if(v<4.5)return'Mars';if(v<11)return'Earth';
  if(v<17)return'Super-Earth';return'Jupiter';
}
$('n').addEventListener('input',e=>{N=+e.target.value;$('nVal').textContent=N;build();render();});
$('g').addEventListener('input',e=>{g=+e.target.value;$('gVal').textContent=g.toFixed(2)+' m/s² · '+planet(g);});
$('pause').addEventListener('click',e=>{paused=!paused;e.target.textContent=paused?'Resume':'Pause';});
$('reset').addEventListener('click',()=>{tEff=0;});
$('trails').addEventListener('change',e=>{trails=e.target.checked;});
window.addEventListener('resize',resize);
window.addEventListener('unload',()=>cancelAnimationFrame(rafId));
build();
resize();
rafId=requestAnimationFrame(frame);
}catch(e){
  const d=document.getElementById('err');
  d.hidden=false;
  d.textContent='Error: '+(e&&e.message?e.message:e);
}
</script>
