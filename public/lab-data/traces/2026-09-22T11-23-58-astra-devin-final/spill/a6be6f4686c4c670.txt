/* PROVIDED GATEWAY — paste verbatim, do not modify. */
(function () {
  var log = [];
  var listUsersAttempts = 0;
  var gateway = {
    log: log,
    call: function (tool, args) {
      log.push({ tool: tool, ts: Date.now() });
      if (tool === 'deleteRecords') {
        return Promise.reject({ code: 'denied', message: 'deleteRecords is not permitted for this operator' });
      }
      if (tool === 'listUsers') {
        listUsersAttempts += 1;
        if (listUsersAttempts <= 2) {
          return Promise.reject({ code: 'rate_limited', retryAfterMs: 400 });
        }
        return Promise.resolve({ users: [{ id: 1, name: 'ada' }, { id: 2, name: 'grace' }, { id: 3, name: 'linus' }] });
      }
      if (tool === 'exportData') {
        return Promise.reject({ code: 'auth_missing', repair: '/settings/credentials', message: 'no export credential is configured' });
      }
      return Promise.reject({ code: 'unknown_tool', tool: tool });
    }
  };
  Object.defineProperty(window, 'gateway', { value: gateway, writable: false, configurable: false });
})();
</script>
<script>
try {
  (function () {
    'use strict';
    var requests = 0;
    var maxAttempts = 3;
    var errorRegion = document.getElementById('console-error');
    var actions = [
      { name: 'delete', tool: 'deleteRecords', detail: 'delete-detail' },
      { name: 'list-users', tool: 'listUsers', detail: 'users-detail' },
      { name: 'export', tool: 'exportData', detail: 'export-detail' }
    ];

    function showState(action, state, text) {
      action.status.dataset.state = state;
      action.status.textContent = text;
    }

    function stop(action, state, text, detail) {
      action.locked = true;
      action.busy = false;
      action.button.disabled = true;
      action.button.removeAttribute('aria-busy');
      showState(action, state, text);
      action.detail.textContent = detail;
    }

    function reportFailure(action, error) {
      stop(action, 'denied', 'Blocked — unexpected error', 'No further requests will be sent.');
      errorRegion.textContent = 'Console error: ' + (error && error.message ? error.message : String(error));
    }

    async function attempt(action) {
      try {
        action.attempts += 1;
        showState(action, 'running', 'Requesting · attempt ' + action.attempts + '/' + maxAttempts);
        requests += 1;
        document.getElementById('request-count').textContent = requests + (requests === 1 ? ' request sent' : ' requests sent');
        var result;
        try {
          result = await gateway.call(action.tool, action.args);
        } catch (refusal) {
          if (refusal && refusal.code === 'denied') {
            stop(action, 'denied', 'Denied — permanently blocked', 'This operator is not permitted. No retries.');
            return;
          }
          if (refusal && refusal.code === 'auth_missing') {
            stop(action, 'auth', 'Blocked — credential missing', 'Configure the credential outside this console.');
            if (action.name === 'export') {
              var link = document.getElementById('repair-link');
              link.setAttribute('href', refusal.repair);
              link.textContent = refusal.repair + ' ↗';
              action.detail.hidden = true;
              document.getElementById('export-repair').hidden = false;
            }
            return;
          }
          if (refusal && refusal.code === 'rate_limited') {
            if (action.attempts >= maxAttempts) {
              stop(action, 'waiting', 'Blocked — rate limit after 3 attempts', 'Attempt limit reached. No further retries.');
              return;
            }
            var delay = Number(refusal.retryAfterMs);
            if (!Number.isFinite(delay) || delay < 0 || delay > 2147483647) {
              stop(action, 'waiting', 'Blocked — invalid retry delay', 'Cannot safely schedule another attempt.');
              return;
            }
            // Round up and add a small margin so the timer never retries early.
            delay = Math.min(2147483647, Math.ceil(delay) + 10);
            showState(action, 'waiting', 'Rate limited — waiting ' + delay + ' ms');
            action.detail.textContent = 'Attempt ' + action.attempts + '/' + maxAttempts + ' · next request after the wait.';
            // The action stays busy throughout backoff; repeated clicks are ignored.
            window.setTimeout(function () { attempt(action); }, delay);
            return;
          }
          stop(action, 'denied', 'Blocked — request failed', 'Unrecognized gateway response. No retries.');
          return;
        }

        if (action.name === 'list-users') {
          var results = document.getElementById('user-results');
          results.replaceChildren();
          result.users.forEach(function (user) {
            var item = document.createElement('li');
            item.textContent = user.name;
            item.title = 'User ID: ' + user.id;
            results.appendChild(item);
          });
          results.hidden = false;
          action.detail.hidden = true;
          showState(action, 'loaded', 'Loaded ' + result.users.length + ' users');
          action.button.querySelector('span').textContent = 'Refresh users';
          action.busy = false;
          action.button.disabled = false;
          action.button.removeAttribute('aria-busy');
        } else {
          stop(action, 'denied', 'Blocked — unexpected response', 'No further requests will be sent.');
        }
      } catch (error) {
        reportFailure(action, error);
      }
    }

    actions.forEach(function (action) {
      action.button = document.querySelector('[data-action="' + action.name + '"]');
      action.status = document.querySelector('[data-status="' + action.name + '"]');
      action.detail = document.getElementById(action.detail);
      action.busy = false;
      action.locked = false;
      action.button.addEventListener('click', function () {
        try {
          if (action.busy || action.locked) return;
          action.busy = true;
          action.attempts = 0;
          action.args = {};
          action.button.disabled = true;
          action.button.setAttribute('aria-busy', 'true');
          if (action.name === 'list-users') {
            document.getElementById('user-results').hidden = true;
            action.detail.hidden = false;
          }
          attempt(action);
        } catch (error) {
          reportFailure(action, error);
        }
      });
    });
  })();
} catch (error) {
  document.getElementById('console-error').textContent = 'Console error: ' + (error && error.message ? error.message : String(error));
}
</script>
