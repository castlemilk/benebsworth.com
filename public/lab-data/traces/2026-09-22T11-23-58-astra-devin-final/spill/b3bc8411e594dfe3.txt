  var canvas=document.getElementById('chart'),ctx=canvas.getContext('2d');
  var N=64,data=[],shown=[],t=0,raf=null;
  for(var i=0;i<N;i++){var v=0.5+0.28*Math.sin(i*0.32)+0.12*Math.sin(i*0.11+2);data.push(v);shown.push(v)}
  function nextVal(){
    t++;
    var v=0.5+0.28*Math.sin(t*0.32)+0.12*Math.sin(t*0.11+2)+(Math.random()-0.5)*0.09;
    if(Math.random()<0.06)v+=0.25*Math.random();
    return Math.max(0.08,Math.min(0.97,v));
  }
  function cssVar(n){return getComputedStyle(root).getPropertyValue(n).trim()}
  function sizeCanvas(){
    var r=canvas.parentElement.getBoundingClientRect();
    var dpr=Math.min(window.devicePixelRatio||1,2);
    canvas.width=Math.max(1,Math.round(r.width*dpr));
    canvas.height=Math.max(1,Math.round(r.height*dpr));
    ctx.setTransform(dpr,0,0,dpr,0,0);
    drawChart();
  }
  function drawChart(){
    var w=canvas.parentElement.clientWidth,h=canvas.parentElement.clientHeight;
    if(w<2||h<2)return;
    ctx.clearRect(0,0,w,h);
    var acc=cssVar('--accent')||'#7c6cff',acc2=cssVar('--accent2')||'#38e0c8';
    var bd=cssVar('--border')||'rgba(255,255,255,.09)';
    ctx.strokeStyle=bd;ctx.lineWidth=1;
    for(var g=1;g<4;g++){ctx.beginPath();ctx.moveTo(0,h*g/4);ctx.lineTo(w,h*g/4);ctx.stroke()}
    var n=shown.length,step=w/(n-1);
    function y(i){return h-6-shown[i]*(h-14)}
    var grad=ctx.createLinearGradient(0,0,w,0);grad.addColorStop(0,acc);grad.addColorStop(1,acc2);
    ctx.beginPath();ctx.moveTo(0,y(0));
    for(var i=1;i<n;i++){var x=i*step,px=(i-1)*step,cx=(px+x)/2;ctx.bezierCurveTo(cx,y(i-1),cx,y(i),x,y(i))}
    ctx.strokeStyle=grad;ctx.lineWidth=2.2;ctx.lineJoin='round';ctx.stroke();
    ctx.lineTo(w,h);ctx.lineTo(0,h);ctx.closePath();
    var fg2=ctx.createLinearGradient(0,0,0,h);
    fg2.addColorStop(0,'rgba(124,108,255,.30)');fg2.addColorStop(1,'rgba(124,108,255,0)');
    ctx.fillStyle=fg2;ctx.fill();
    var lx=(n-1)*step,ly=y(n-1);
    ctx.beginPath();ctx.arc(lx-2,ly,3.5,0,Math.PI*2);ctx.fillStyle=acc2;ctx.fill();
    ctx.beginPath();ctx.arc(lx-2,ly,7,0,Math.PI*2);ctx.fillStyle='rgba(56,224,200,.25)';ctx.fill();
  }
  function loop(){
    for(var i=0;i<N;i++)shown[i]+=(data[i]-shown[i])*0.18;
    drawChart();raf=requestAnimationFrame(loop);
