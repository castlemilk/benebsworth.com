function step(dt){
 var th=(keys.up||touch.gas)?1:0,br=(keys.down||touch.brake)?1:0;
 var st=(keys.right||touch.right?1:0)-(keys.left||touch.left?1:0);
 P.idx=nearestIdx(P.x,P.y,P.idx,false);

 var fx=Math.cos(P.h),fy=Math.sin(P.h),rx=-Math.sin(P.h),ry=Math.cos(P.h);
 var vf=P.vx*fx+P.vy*fy,vl=P.vx*rx+P.vy*ry;
 var acc=ACC,drag=DRAG,grp=GRIP;
 if(P.off){acc*=0.45;drag+=0.6;grp=4.0;if(P.spd>50&&rng()<dt*28)dust();}
 else if(P.onCurb){acc*=0.85;drag+=0.18;grp=6.5;if(P.spd>60)shake=Math.max(shake,1.6);}
 vf+=th*acc*dt;
 if(br){if(vf>2)vf=Math.max(0,vf-BRK*dt);else vf=Math.max(-85,vf-ACC*0.55*dt);}
 vf-=vf*drag*dt;
 if(!th&&!br)vf-=vf*0.35*dt;
 vl*=Math.exp(-grp*dt);
 var sf=clamp(Math.abs(vf)/70,0,1)*(1-0.35*clamp(Math.abs(vf)/VMAX,0,1));
 P.h+=st*YAW*sf*(vf<0?-1:1)*dt;
 fx=Math.cos(P.h);fy=Math.sin(P.h);rx=-Math.sin(P.h);ry=Math.cos(P.h);
 P.vx=fx*vf+rx*vl;P.vy=fy*vf+ry*vl;
 var ox=P.x,oy=P.y;
 P.x+=P.vx*dt;P.y+=P.vy*dt;
 P.spd=Math.hypot(P.vx,P.vy);P.vf=vf;P.vl=vl;
 if(vf>0)P.dist+=vf*dt;
 crossGate(P.nextGate,ox,oy);
 if(!P.armed)crossGate(0,ox,oy);
 var sRaw=P.idx*DS,dd=sRaw-P.sRawP;
 if(dd>L/2)dd-=L;if(dd<-L/2)dd+=L;P.sU+=dd;P.sRawP=sRaw;
 var mv=P.vx*Math.cos(ta[P.idx])+P.vy*Math.sin(ta[P.idx]);
 if(!P.off&&mv<-30)P.wrongT+=dt;else P.wrongT=0;
 if(!P.off&&Math.abs(vl)>55&&P.spd>90){stamp(P.x-fx*10+rx*7,P.y-fy*10+ry*7);stamp(P.x-fx*10-rx*7,P.y-fy*10-ry*7);}
 var pS=((P.sU%L)+L)%L;
 for(var i=0;i<AI.length;i++){var ai=AI[i],ii=Math.floor(ai.s/DS)%N;
  var km=1e-4;for(var q=4;q<=48;q+=4){var cv2=Math.abs(kv[(ii+q)%N]);if(cv2>km)km=cv2;}
  var vd=Math.min(VMAX*ai.sk,Math.sqrt(880/km));
  var gap=((pS-ai.s)%L+L)%L;
  if(gap<70&&Math.abs(ti.lat-ai.lat)<20)vd=Math.min(vd,Math.abs(vf)*0.9+4);
  for(var j2=0;j2<AI.length;j2++){if(j2===i)continue;var aj=AI[j2];var gp2=((aj.s-ai.s)%L+L)%L;if(gp2<70&&Math.abs(aj.lat-ai.lat)<20)vd=Math.min(vd,aj.vf*0.9+4);}
  if(ai.hitT>0){vd*=0.55;ai.hitT-=dt;}
  if(ai.vf<vd)ai.vf=Math.min(vd,ai.vf+215*dt);else ai.vf=Math.max(vd,ai.vf-440*dt);
  ai.s=(ai.s+ai.vf*dt+L)%L;ai.sT+=ai.vf*dt;
  var jj=Math.floor(ai.s/DS)%N;
  ai.lat=ai.lb+Math.sin(nowT*0.55+ai.ph)*3.5;
  var nx=-Math.sin(ta[jj]),ny=Math.cos(ta[jj]);
  ai.x=sx[jj]+nx*ai.lat;ai.y=sy[jj]+ny*ai.lat;ai.h=ta[jj];
  ai.vx=Math.cos(ai.h)*ai.vf;ai.vy=Math.sin(ai.h)*ai.vf;
  var dx=P.x-ai.x,dy=P.y-ai.y,d2=Math.hypot(dx,dy);
  if(d2<28&&d2>0.01){var ux=dx/d2,uy=dy/d2,ov=28-d2;P.x+=ux*ov;P.y+=uy*ov;
   var rv=(P.vx-ai.vx)*ux+(P.vy-ai.vy)*uy;
   if(rv<0){P.vx-=ux*rv*1.6;P.vy-=uy*rv*1.6;}
   ai.hitT=0.7;shake=Math.max(shake,5);sparks((P.x+ai.x)/2,(P.y+ai.y)/2,9);}
 }
 for(i=parts.length-1;i>=0;i--){var pt=parts[i];pt.l-=dt;if(pt.l<=0){parts.splice(i,1);continue;}pt.x+=pt.vx*dt;pt.y+=pt.vy*dt;pt.vx*=0.96;pt.vy*=0.96;}
 var lv=clamp(P.spd*0.3,0,85);
 var tx2=P.x+fx*lv,ty2=P.y+fy*lv,kf=1-Math.exp(-5*dt);
 cam.x+=(tx2-cam.x)*kf;cam.y+=(ty2-cam.y)*kf;
 cam.rot+=wrapAng(P.h-cam.rot)*(1-Math.exp(-6*dt));
 shake*=Math.exp(-5*dt);
 attrs();
}
