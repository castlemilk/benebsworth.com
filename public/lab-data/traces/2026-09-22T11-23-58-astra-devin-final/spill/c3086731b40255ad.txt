<main>
  <h1>Tic-Tac-Toe</h1>
  <p id="status" aria-live="polite">X&rsquo;s turn</p>
  <div id="board" role="grid" aria-label="Tic-tac-toe board"></div>
  <button id="restart" type="button">Restart</button>
</main>
<script>
try {
  var WIN_LINES = [
    [0, 1, 2], [3, 4, 5], [6, 7, 8],
    [0, 3, 6], [1, 4, 7], [2, 5, 8],
    [0, 4, 8], [2, 4, 6]
  ];

  var board = document.getElementById('board');
  var status = document.getElementById('status');
  var restartBtn = document.getElementById('restart');
  var cells = [];
  var marks = [];
  var current = 'X';
  var over = false;

  for (var i = 0; i < 9; i++) {
    var cell = document.createElement('button');
    cell.type = 'button';
    cell.className = 'cell';
    cell.setAttribute('role', 'gridcell');
    cell.setAttribute('aria-label', 'Cell ' + (i + 1));
    (function (idx) {
      cell.addEventListener('click', function () { play(idx); });
    })(i);
    board.appendChild(cell);
    cells.push(cell);
    marks.push('');
  }

  function winner() {
    for (var l = 0; l < WIN_LINES.length; l++) {
      var line = WIN_LINES[l];
      var a = line[0], b = line[1], c = line[2];
      if (marks[a] && marks[a] === marks[b] && marks[b] === marks[c]) {
        return { mark: marks[a], line: line };
      }
    }
    return null;
  }

  function play(idx) {
    if (over || marks[idx]) return;
    marks[idx] = current;
    var cell = cells[idx];
    cell.textContent = current;
    cell.classList.add(current === 'X' ? 'x' : 'o');
    cell.disabled = true;

    var result = winner();
    if (result) {
      over = true;
      status.textContent = result.mark + ' wins';
      status.classList.add('result');
      for (var j = 0; j < result.line.length; j++) {
        cells[result.line[j]].classList.add('win');
      }
      lockBoard();
    } else if (marks.indexOf('') === -1) {
      over = true;
      status.textContent = 'Draw — nobody wins';
      status.classList.add('result');
      lockBoard();
    } else {
      current = current === 'X' ? 'O' : 'X';
      status.textContent = current + '’s turn';
    }
  }

  function lockBoard() {
    for (var j = 0; j < 9; j++) cells[j].disabled = true;
  }

  function restart() {
    marks = [];
    current = 'X';
    over = false;
    for (var j = 0; j < 9; j++) {
      marks.push('');
      cells[j].textContent = '';
      cells[j].className = 'cell';
      cells[j].disabled = false;
    }
    status.textContent = 'X’s turn';
    status.classList.remove('result');
  }

  restartBtn.addEventListener('click', restart);
} catch (err) {
  var alertBox = document.createElement('div');
  alertBox.setAttribute('role', 'alert');
  alertBox.textContent = 'Something went wrong: ' + (err && err.message ? err.message : String(err));
  document.body.appendChild(alertBox);
}
</script>
