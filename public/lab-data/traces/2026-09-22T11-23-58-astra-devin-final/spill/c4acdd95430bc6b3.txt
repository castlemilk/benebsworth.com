  <div id="error" role="alert"></div>
  <script>
    try {
      const cells = Array.from(document.querySelectorAll('.cell'));
      const players = Array.from(document.querySelectorAll('.player'));
      const status = document.getElementById('status');
      const winningLines = [
        [0,1,2], [3,4,5], [6,7,8],
        [0,3,6], [1,4,7], [2,5,8],
        [0,4,8], [2,4,6]
      ];
      let board = Array(9).fill('');
      let currentPlayer = 'X';
      let gameOver = false;

      function reportError(error) {
        document.getElementById('error').textContent = 'Unable to continue: ' + (error instanceof Error ? error.message : String(error));
      }

      function updatePlayers() {
        players.forEach(player => {
          player.classList.toggle('active', !gameOver && player.dataset.player === currentPlayer);
        });
      }

      function play(index) {
        if (gameOver || board[index] !== '') return;
        board[index] = currentPlayer;
        cells[index].textContent = currentPlayer;
        cells[index].dataset.mark = currentPlayer;
        cells[index].setAttribute('aria-label', 'Row ' + (Math.floor(index / 3) + 1) + ', column ' + (index % 3 + 1) + ', ' + currentPlayer);
        cells[index].setAttribute('aria-disabled', 'true');

        const winningLine = winningLines.find(line => line.every(position => board[position] === currentPlayer));
        if (winningLine) {
          gameOver = true;
          winningLine.forEach(position => cells[position].classList.add('winning'));
          status.textContent = currentPlayer + ' wins! Three in a row.';
        } else if (board.every(mark => mark !== '')) {
          gameOver = true;
          status.textContent = "It's a draw! Play another round.";
        } else {
          currentPlayer = currentPlayer === 'X' ? 'O' : 'X';
          status.textContent = currentPlayer + "'s turn — choose a square.";
        }
        if (gameOver) {
          cells.forEach(cell => cell.setAttribute('aria-disabled', 'true'));
          status.classList.add('result');
        }
        updatePlayers();
      }

      cells.forEach((cell, index) => {
        cell.addEventListener('click', () => {
          try { play(index); } catch (error) { reportError(error); }
        });
      });

      document.getElementById('restart').addEventListener('click', () => {
        try {
          board = Array(9).fill('');
          currentPlayer = 'X';
          gameOver = false;
          cells.forEach((cell, index) => {
            cell.textContent = '';
            cell.removeAttribute('data-mark');
            cell.removeAttribute('aria-disabled');
            cell.classList.remove('winning');
            cell.setAttribute('aria-label', 'Row ' + (Math.floor(index / 3) + 1) + ', column ' + (index % 3 + 1) + ', empty');
          });
          status.textContent = "X's turn — choose a square.";
          status.classList.remove('result');
          document.getElementById('error').textContent = '';
          updatePlayers();
        } catch (error) { reportError(error); }
      });
    } catch (error) {
      document.getElementById('error').textContent = 'Unable to start the game: ' + (error instanceof Error ? error.message : String(error));
    }
  </script>
