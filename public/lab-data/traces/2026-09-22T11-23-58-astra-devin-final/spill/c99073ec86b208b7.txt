 if(Math.abs(latNow)>HW-14){tx=tp.x;ty=tp.y;}
 var des=Math.atan2(ty-ai.y,tx-ai.x);
 var err=wrapAng(des-ai.heading);
 ai.heading+=clamp(err*3.2,-2.6,2.6)*dt*clamp(Math.abs(ai.speed)/120+0.25,0,1);
 var mc=maxCurvAhead(ai.s,look+220);
 var vt=Math.min(MAXV*0.88*ai.skill,mc>0.0005?Math.sqrt(1500/mc):MAXV);
 // rubber band vs player
 var gap=(lapsDone*L+player.s)-(ai.laps*L+ai.s);
 vt*=clamp(1+gap/4000,0.85,1.15)*slowK;
 if(ai.speed<vt)ai.speed+=ACCEL*0.8*dt;else ai.speed-=BRAKE*0.6*dt;
 ai.speed=clamp(ai.speed,0,MAXV*0.95);
 ai.x+=Math.cos(ai.heading)*ai.speed*dt;
 ai.y+=Math.sin(ai.heading)*ai.speed*dt;
 if(Math.abs(nearest(ai.x,ai.y).lat)>HW+8)ai.speed*=Math.max(0,1-3*dt);
}
function collide(){
 var all=[player].concat(rivals),i2,j;
 for(i2=0;i2<all.length;i2++)for(j=i2+1;j<all.length;j++){
  var a=all[i2],b=all[j];var dx=b.x-a.x,dy=b.y-a.y;var d=Math.hypot(dx,dy),min=a.r+b.r;
  if(d<min&&d>0.01){var nx2=dx/d,ny2=dy/d,ov=(min-d)/2;
   a.x-=nx2*ov;a.y-=ny2*ov;b.x+=nx2*ov;b.y+=ny2*ov;
   var closing=(Math.cos(a.heading)*a.speed-Math.cos(b.heading)*b.speed)*nx2+(Math.sin(a.heading)*a.speed-Math.sin(b.heading)*b.speed)*ny2;
   if(closing>0){var f2=clamp(1-closing*0.0035,0.55,1);a.speed*=f2;b.speed*=f2;}
   if(closing>70){for(var k=0;k<6;k++)dust.push({x:(a.x+b.x)/2,y:(a.y+b.y)/2,vx:(Math.random()-0.5)*180,vy:(Math.random()-0.5)*180,life:0.4,r:2+Math.random()*3,spark:true});}
  }}
}
