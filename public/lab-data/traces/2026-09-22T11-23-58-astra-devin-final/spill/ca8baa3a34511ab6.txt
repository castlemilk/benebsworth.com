    ui.returnTime.textContent=((CYCLE-phase)/factor).toFixed(1);
    ui.cycleTime.textContent=(phase/factor).toFixed(1).padStart(4,'0');
    ui.cycleLength.textContent=period.toFixed(1)+' s';
    ui.time.value=phase/CYCLE*1000;paintRange(ui.time);
    ui.time.setAttribute('aria-valuetext',(phase/factor).toFixed(1)+' seconds of '+period.toFixed(1)+' seconds');
  }
  function updatePlaying() {
    ui.play.setAttribute('aria-label',running?'Pause simulation':'Play simulation');
    ui.playIcon.innerHTML=running?'<path d="M7 4v12M13 4v12" stroke-width="2.5"/>':'<path d="M6 3.5 16 10 6 16.5Z" fill="currentColor" stroke="none"/>';
    ui.statusText.textContent=running?'RUNNING':'PAUSED';ui.status.classList.toggle('paused',!running);lastTime=null;
  }
  function tick(now) {
    if(stopped)return;
    try {
      if(lastTime!==null && running && !document.hidden) phaseTime=(phaseTime+Math.min((now-lastTime)/1000,.1)*Math.sqrt(gravity/G0))%CYCLE;
      lastTime=now;
      if(running)draw();
      if(now-lastUI>60){updateUI();lastUI=now;}
      raf=requestAnimationFrame(tick);
    } catch(error){fail(error);}
  }
  ui.count.addEventListener('input',guarded(()=>{n=Number(ui.count.value);ui.countValue.value=n;paintRange(ui.count);rebuild();draw();updateUI();}));
  ui.gravity.addEventListener('input',guarded(()=>{gravity=Number(ui.gravity.value);ui.gravityValue.value=gravity.toFixed(2);paintRange(ui.gravity);updateUI();draw();}));
  ui.time.addEventListener('input',guarded(()=>{phaseTime=Number(ui.time.value)/1000*CYCLE;running=false;updatePlaying();paintRange(ui.time);draw();updateUI();}));
  ui.play.addEventListener('click',guarded(()=>{running=!running;updatePlaying();}));
  $('reset').addEventListener('click',guarded(()=>{phaseTime=0;lastTime=null;updateUI();draw();}));
  ui.wave.addEventListener('click',guarded(()=>{showWave=!showWave;ui.wave.setAttribute('aria-pressed',String(showWave));draw();}));
  canvas.addEventListener('pointermove',guarded(event=>{
    pointer={x:event.clientX,y:event.clientY};let closest=-1,distance=23;
    for(const item of rendered){const d=Math.hypot(item.point.x-pointer.x,item.point.y-pointer.y);if(d<distance){distance=d;closest=item.p.i;}}
    if(closest!==selected){selected=closest;canvas.style.cursor=selected<0?'default':'crosshair';draw();}
  }));
  canvas.addEventListener('pointerleave',guarded(()=>{selected=-1;pointer=null;draw();}));
  window.addEventListener('keydown',guarded(event=>{
    if(event.target instanceof HTMLInputElement || event.target instanceof HTMLButtonElement || event.ctrlKey || event.metaKey || event.altKey)return;
    if(event.code==='Space'){event.preventDefault();running=!running;updatePlaying();}
    if(event.key.toLowerCase()==='r'){phaseTime=0;lastTime=null;draw();updateUI();}
  }));
  window.addEventListener('resize',guarded(layout));
  document.addEventListener('visibilitychange',()=>{lastTime=null;});
  window.addEventListener('unload',()=>{stopped=true;cancelAnimationFrame(raf);});
  window.addEventListener('pagehide',()=>{cancelAnimationFrame(raf);lastTime=null;});
  window.addEventListener('pageshow',event=>{if(event.persisted&&!stopped){lastTime=null;raf=requestAnimationFrame(tick);}});
  rebuild();updatePlaying();updateUI();layout();raf=requestAnimationFrame(tick);
} catch(error) {
  document.getElementById('error').textContent='Unable to start the simulation: '+(error && error.message ? error.message : String(error));
}
