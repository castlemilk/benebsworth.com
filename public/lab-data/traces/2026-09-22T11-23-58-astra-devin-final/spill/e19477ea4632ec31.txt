function clamp(v,a,b){return v<a?a:(v>b?b:v)}
function rr(c,x,y,w,h,r){
  r=Math.min(r,w/2,h/2);
  c.beginPath();
  c.moveTo(x+r,y);
  c.arcTo(x+w,y,x+w,y+h,r);
  c.arcTo(x+w,y+h,x,y+h,r);
  c.arcTo(x,y+h,x,y,r);
  c.arcTo(x,y,x+w,y,r);
  c.closePath();
}
function rng(seed){return function(n){var s=Math.sin(n*127.1+seed*311.7)*43758.5453;return s-Math.floor(s)}}
function overlaps(a,b){return a.x<b.x+b.w&&a.x+a.w>b.x&&a.y<b.y+b.h&&a.y+a.h>b.y}
function circRect(cx,cy,r,b){var nx=clamp(cx,b.x,b.x+b.w),ny=clamp(cy,b.y,b.y+b.h);var dx=cx-nx,dy=cy-ny;return dx*dx+dy*dy<r*r}

/* ---------- particles ---------- */
function burst(x,y,n,col,spd){
  for(var i=0;i<n;i++){
    var a=Math.random()*6.28,s=spd*(0.4+Math.random()*0.8);
    parts.push({x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,life:0.5+Math.random()*0.4,max:0.9,col:col,r:1.5+Math.random()*2.5});
  }
}

/* ---------- update ---------- */
function update(dt){
  var p=player;
  var left=kb.l||tc.l,right=kb.r||tc.r,jump=kb.j||tc.j;
  if((left||right||jump)&&!started)started=true;
  if(!won){
    var ax=0;
    if(left){ax-=1;p.face=-1}
    if(right){ax+=1;p.face=1}
    var acc=p.grounded?2700:1900;
    if(ax!==0){p.vx=clamp(p.vx+ax*acc*dt,-RUN,RUN);p.runPh+=dt*(p.grounded?14:0)}
    else{
      var f=(p.grounded?3400:700)*dt;
      p.vx-=Math.sign(p.vx)*Math.min(Math.abs(p.vx),f);
    }
    /* jump: edge-triggered + buffer + variable height */
    if(jump&&!prevJump)p.jbuf=0.12;
    if(!jump&&prevJump&&p.vy<-250)p.vy=-250;
    if(p.jbuf===undefined)p.jbuf=0;
    p.jbuf-=dt;
    if(p.grounded)p.coyote=0.1;else p.coyote=(p.coyote||0)-dt;
    if(p.jbuf>0&&p.coyote>0){
      p.vy=-JUMP_V;p.grounded=false;p.coyote=0;p.jbuf=0;
      burst(p.x+p.w/2,p.y+p.h,6,'#8a93b8',120);
    }
    p.vy=Math.min(p.vy+GRAV*dt,1450);
    /* horizontal */
    p.x+=p.vx*dt;
    for(var i=0;i<plats.length;i++){
      var pl=plats[i];
      if(overlaps(p,pl)){
        if(p.vx>0)p.x=pl.x-p.w;else if(p.vx<0)p.x=pl.x+pl.w;
        p.vx=0;
      }
    }
    p.x=clamp(p.x,0,LEVEL_W-p.w);
    /* vertical */
    var wasG=p.grounded;p.grounded=false;
    p.y+=p.vy*dt;
    for(i=0;i<plats.length;i++){
      pl=plats[i];
      if(overlaps(p,pl)){
        if(p.vy>0){
          p.y=pl.y-p.h;p.vy=0;p.grounded=true;
          if(!wasG)burst(p.x+p.w/2,p.y+p.h,5,'#6b7492',90);
        }else if(p.vy<0){p.y=pl.y+pl.h;p.vy=0}
      }
    }
    /* coins */
    for(i=0;i<coins.length;i++){
      var c=coins[i];
      if(!c.taken&&circRect(c.x,c.y,12,p)){c.taken=true;coinCount++;burst(c.x,c.y,10,'#ffd34d',160)}
    }
    /* flag */
    if(overlaps(p,{x:FLAG.x-8,y:FLAG.y-96,w:56,h:96})){
      won=true;
      winStats.textContent='Time '+time.toFixed(1)+'s · Coins '+coinCount+'/'+coins.length;
      winEl.style.display='flex';
    }
    if(p.y>KILL_Y){
      p.x=SPAWN.x;p.y=SPAWN.y;p.vx=0;p.vy=0;
      camX=clamp(p.x-W*0.42,0,Math.max(0,LEVEL_W-W));
      camY=clamp(p.y-H*0.6,-80,Math.max(-80,760-H));
    }
    if(started)time+=dt;
  }
  prevJump=jump;
  /* camera */
  var tx=clamp(p.x+p.w/2-W*0.42,0,Math.max(0,LEVEL_W-W));
  var ty=clamp(p.y+p.h/2-H*0.62,-80,Math.max(-80,760-H));
  camX+=(tx-camX)*Math.min(1,dt*6);
  camY+=(ty-camY)*Math.min(1,dt*5);
  /* particles */
  for(var i=parts.length-1;i>=0;i--){
    var pt=parts[i];
    pt.life-=dt;pt.x+=pt.vx*dt;pt.y+=pt.vy*dt;pt.vy+=900*dt;
    if(pt.life<=0)parts.splice(i,1);
  }
}

function drawRidge(f,spacing,baseY,amp,fill,seed){
  var off=camX*f,r=rng(seed);
  var i0=Math.floor((off-700)/spacing),i1=Math.ceil((off+W+700)/spacing);
  ctx.fillStyle=fill;
  for(var i=i0;i<=i1;i++){
    var h=amp*(0.5+0.5*r(i));
    var x=i*spacing+r(i*7)*140-off;
    var w=spacing*1.6;
    ctx.beginPath();
    ctx.moveTo(x-w/2,baseY);
    ctx.lineTo(x,baseY-h);
    ctx.lineTo(x+w/2,baseY);
    ctx.closePath();ctx.fill();
  }
}

function drawCoins(){
  for(var i=0;i<coins.length;i++){
    var c=coins[i];
    if(c.taken||c.x<camX-40||c.x>camX+W+40)continue;
    var bob=Math.sin(t*3+c.ph)*3.5;
    var sc=Math.cos(t*4+c.ph);
    var rx=Math.max(2.2,Math.abs(sc)*9);
    ctx.save();
    ctx.translate(c.x,c.y+bob);
    ctx.fillStyle='#ffb31f';
    ctx.beginPath();ctx.ellipse(0,0,rx,10,0,0,6.2832);ctx.fill();
    ctx.fillStyle='#ffd34d';
    ctx.beginPath();ctx.ellipse(0,-1,rx*0.72,7,0,0,6.2832);ctx.fill();
    ctx.fillStyle='rgba(255,255,255,.8)';
    ctx.beginPath();ctx.ellipse(-rx*0.25,-4,Math.max(0.8,rx*0.18),2.4,0,0,6.2832);ctx.fill();
    ctx.restore();
  }
}

  ctx.quadraticCurveTo(x+22,y-92+w1,x+42,y-85+w2);

  ctx.closePath();ctx.fill();

function drawPlayer(){
  var p=player;
  var air=!p.grounded;
  var k=air?clamp(Math.abs(p.vy)/2600,0,0.15):0;
  var sy=1+k,sx=1-k*0.6;
  ctx.save();
  ctx.translate(p.x+p.w/2,p.y+p.h);
  if(p.grounded){
    ctx.fillStyle='rgba(0,0,0,.28)';
    ctx.beginPath();ctx.ellipse(0,1.5,p.w*0.55,4,0,0,6.2832);ctx.fill();
  }
  ctx.scale(sx,sy);
  var g=ctx.createLinearGradient(0,-p.h,0,0);
  g.addColorStop(0,'#ff9a62');g.addColorStop(1,'#ff5f3d');
  ctx.fillStyle=g;
  rr(ctx,-p.w/2,-p.h,p.w,p.h,9);ctx.fill();
  /* feet */
  var sw=p.grounded&&Math.abs(p.vx)>30?Math.sin(p.runPh)*4:0;
  ctx.fillStyle='#3a2a4a';
  rr(ctx,-p.w/2+1+sw,-5,11,5,2.5);ctx.fill();
  rr(ctx,p.w/2-12-sw,-5,11,5,2.5);ctx.fill();
  /* eyes */
  var ex=p.face*4;
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(-5+ex,-p.h+13,3.6,0,6.2832);ctx.arc(6+ex,-p.h+13,3.6,0,6.2832);ctx.fill();
  ctx.fillStyle='#1b1430';
  ctx.beginPath();ctx.arc(-5+ex+p.face*1.4,-p.h+13,1.8,0,6.2832);ctx.arc(6+ex+p.face*1.4,-p.h+13,1.8,0,6.2832);ctx.fill();
  ctx.restore();
}
function drawParts(){
  for(var i=0;i<parts.length;i++){
    var pt=parts[i];
    ctx.globalAlpha=clamp(pt.life/pt.max,0,1);
    ctx.fillStyle=pt.col;
    ctx.beginPath();ctx.arc(pt.x,pt.y,pt.r,0,6.2832);ctx.fill();
  }
  ctx.globalAlpha=1;
}
function render(){
  drawSky();
  drawStars();
  drawMoon();
  drawRidge(0.22,460,H*0.78,220,'#131a36',11);
  drawClouds();
  drawHills();
  ctx.save();
  ctx.translate(-camX,-camY);
  drawFlag();
  drawPlats();
  drawCoins();
  drawParts();
  drawPlayer();
  ctx.restore();
  coinsEl.textContent=coinCount+'/'+coins.length;
  timeEl.textContent=time.toFixed(1)+'s';
}

/* ---------- main loop ---------- */
function loop(now){
  var dt=Math.min((now-last)/1000,0.033);
  last=now;t+=dt;
  update(dt);
  render();
  raf=requestAnimationFrame(loop);
}
