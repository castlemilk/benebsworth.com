  function update(source){

    const critical=Math.abs(z-1)<1e-8,under=z<1&&!critical;
    const settling=settlingX(z),overshoot=under?100*Math.exp(-Math.PI*z/Math.sqrt(1-z*z)):0;
    const horizon=z===0?8*Math.PI:under?Math.max(9,Math.min(36*Math.PI,settling*1.45)):settling*1.45;
    const end=niceCeiling(horizon/wn)*Number($('time-scale').value);
    model={wn,z,overshoot,settling:settling/wn,end,ymax:overshoot>45?2.2:1.6,critical,under};
    $('zeta-label').textContent='ζ = '+(z<10?z.toFixed(2):short(z));
    $('regime').textContent=z===0?'Undamped':critical?'Critical':under?'Underdamped':'Overdamped';
    const f=wn/(2*Math.PI);$('frequency').textContent=short(f>=1000?f/1000:f);$('frequency-unit').textContent=f>=1000?'kHz':'Hz';
    $('overshoot').textContent=overshoot.toFixed(1);

    $('insight').textContent=z===0?'No resistance, no decay. Energy keeps oscillating.':critical?'Critical damping: the fastest return without overshoot.':under?'Turn up R to tame the oscillation.':'Too much resistance slows the return. Try Critical.';
    document.querySelectorAll('.preset').forEach(b=>b.setAttribute('aria-pressed',String(Math.abs(z-Number(b.dataset.zeta))<1e-7)));
    canvas.setAttribute('aria-label','Capacitor voltage for a 1 volt step. '+$('regime').textContent+', damping ratio '+z.toFixed(3)+', overshoot '+overshoot.toFixed(1)+' percent. Use arrow keys to inspect.');
