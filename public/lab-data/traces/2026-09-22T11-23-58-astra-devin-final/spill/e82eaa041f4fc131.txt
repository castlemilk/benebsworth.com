/* PROVIDED GATEWAY — paste verbatim, do not modify. */
(function () {
  var log = [];
  var listUsersAttempts = 0;
  var gateway = {
    log: log,
    call: function (tool, args) {
      log.push({ tool: tool, ts: Date.now() });
      if (tool === 'deleteRecords') {
        return Promise.reject({ code: 'denied', message: 'deleteRecords is not permitted for this operator' });
      }
      if (tool === 'listUsers') {
        listUsersAttempts += 1;
        if (listUsersAttempts <= 2) {
          return Promise.reject({ code: 'rate_limited', retryAfterMs: 400 });
        }
        return Promise.resolve({ users: [{ id: 1, name: 'ada' }, { id: 2, name: 'grace' }, { id: 3, name: 'linus' }] });
      }
      if (tool === 'exportData') {
        return Promise.reject({ code: 'auth_missing', repair: '/settings/credentials', message: 'no export credential is configured' });
      }
      return Promise.reject({ code: 'unknown_tool', tool: tool });
    }
  };
  Object.defineProperty(window, 'gateway', { value: gateway, writable: false, configurable: false });
})();
</script>
<script>
try {
  (function () {
    'use strict';
    var configurations = [
      { action: 'delete', tool: 'deleteRecords', details: 'delete-details' },
      { action: 'list-users', tool: 'listUsers', details: 'users-details' },
      { action: 'export', tool: 'exportData', details: 'export-details' }
    ];

    configurations.forEach(function (config) {
      var button = document.querySelector('[data-action="' + config.action + '"]');
      var status = document.querySelector('[data-status="' + config.action + '"]');
      var card = document.querySelector('[data-action-card="' + config.action + '"]');
      var label = button.querySelector('.button-label');
      var details = document.getElementById(config.details);
      var originalLabel = label.textContent;
      var busy = false;
      var finalBlock = false;

      function setState(state, text) {
        card.setAttribute('data-state', state);
        status.textContent = text;
      }

      function block(state, text, buttonText) {
        finalBlock = true;
        busy = false;
        button.disabled = true;
        button.setAttribute('aria-busy', 'false');
        label.textContent = buttonText;
        setState(state, text);
      }

      function unexpected(error) {
        block('blocked', 'Blocked · unexpected error', 'Action blocked');
        var errorBox = document.getElementById('page-error');
        errorBox.textContent = 'Console error: ' + (error && error.message ? error.message : String(error && error.code || error));
        errorBox.hidden = false;
      }

      function attempt(number) {
        setState('loading', 'Requesting · attempt ' + number + ' of 3');
        label.textContent = 'Requesting…';
        try {
          gateway.call(config.tool, {}).then(function (result) {
            if (config.action === 'list-users') {
              var users = document.createElement('ul');
              users.className = 'users';
              users.setAttribute('aria-label', 'Loaded users');
              result.users.forEach(function (user) {
                var item = document.createElement('li');
                item.textContent = user.name;
                users.appendChild(item);
              });
              details.replaceChildren(users);
              setState('loaded', 'Loaded · ' + result.users.length + ' users');
            } else {
              setState('loaded', 'Request accepted');
            }
            busy = false;
            button.disabled = false;
            button.setAttribute('aria-busy', 'false');
            label.textContent = originalLabel;
          }, function (error) {
            if (error && error.code === 'denied') {
              block('blocked', 'Blocked · permission denied', 'Access denied');
              details.textContent = 'Permanent denial. No retries.';
              return;
            }
            if (error && error.code === 'rate_limited') {
              if (number >= 3) {
                block('waiting', 'Rate limited · 3-attempt limit reached', 'Attempt limit reached');
                details.textContent = 'Stopped after 3 attempts.';
                return;
              }
              var delay = error.retryAfterMs;
              if (typeof delay !== 'number' || !Number.isFinite(delay) || delay < 0 || delay > 2147483647) {
                block('blocked', 'Blocked · invalid retry delay', 'Action blocked');
                return;
              }
              delay = Math.ceil(delay);
              setState('waiting', 'Rate limited · waiting ' + delay + ' ms');
              label.textContent = 'Waiting to retry…';
              details.textContent = 'Next: attempt ' + (number + 1) + ' of 3';
              setTimeout(function () {
                try { attempt(number + 1); } catch (error) { unexpected(error); }
              }, delay);
              return;
            }
            if (error && error.code === 'auth_missing') {
              block('auth', 'Blocked · missing credential', 'Credential required');
              var repair = document.createElement('a');
              repair.setAttribute('href', error.repair);
              repair.textContent = error.repair + ' ↗';
              repair.setAttribute('aria-label', 'Repair credentials elsewhere: ' + error.repair);
              details.replaceChildren(document.createTextNode('Repair: '), repair);
              return;
            }
            unexpected(error);
          }).catch(unexpected);
        } catch (error) {
          unexpected(error);
        }
      }

      button.addEventListener('click', function () {
        try {
          if (busy || finalBlock) return;
          busy = true;
          button.disabled = true;
          button.setAttribute('aria-busy', 'true');
          details.replaceChildren();
          attempt(1);
        } catch (error) {
          unexpected(error);
        }
      });
    });
  })();
} catch (error) {
  var fatalError = document.getElementById('page-error');
  fatalError.textContent = 'Console error: ' + (error && error.message ? error.message : String(error));
  fatalError.hidden = false;
}
</script>
