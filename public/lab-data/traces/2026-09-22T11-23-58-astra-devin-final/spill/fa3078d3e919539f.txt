(function(){
'use strict';
try{
var cv=document.getElementById('game'),ctx=cv.getContext('2d');
var coinEl=document.getElementById('coinCt'),timeEl=document.getElementById('time');
var winEl=document.getElementById('win'),winStats=document.getElementById('winStats');

/* ---------- canvas sizing ---------- */
var vw=0,vh=0,dpr=1;
function resize(){
  dpr=Math.min(window.devicePixelRatio||1,2);
  vw=window.innerWidth;vh=window.innerHeight;
  cv.width=Math.round(vw*dpr);cv.height=Math.round(vh*dpr);
  cv.style.width=vw+'px';cv.style.height=vh+'px';
}
window.addEventListener('resize',resize);resize();

/* ---------- level ---------- */
var T=40,W=82,H=12,worldW=W*T,worldH=H*T,groundRow=10;
var grid=[];
for(var y=0;y<H;y++){grid.push(new Array(W).fill('.'));}
function rect(x,y,w,h,c){for(var j=y;j<y+h;j++)for(var i=x;i<x+w;i++)if(i>=0&&j>=0&&i<W&&j<H)grid[j][i]=c;}
/* ground segments with pits */
rect(0,10,17,2,'#');rect(21,10,14,2,'#');rect(38,10,9,2,'#');rect(52,10,30,2,'#');
/* floating platforms */
rect(6,7,4,1,'=');rect(25,7,5,1,'=');rect(55,7,4,1,'=');rect(74,7,4,1,'=');
rect(14,5,4,1,'=');rect(44,5,4,1,'=');rect(66,5,4,1,'=');

function solid(tx,ty){
  if(tx<0||tx>=W)return true;
  if(ty<0||ty>=H)return false;
  return grid[ty][tx]==='#'||grid[ty][tx]=== '=';
}

var COIN_POS=[[2,9],[9,9],[7,6],[8,6],[15,4],[16,4],[19,9],[23,9],[27,6],[28,6],[31,9],[36,9],[41,9],[45,4],[46,4],[49,9],[56,6],[57,9],[61,9],[67,4],[68,4],[73,9],[75,6],[77,9],[80,9]];
var FLAG_X=79.5*T,FLAG_BASE=10*T,FLAG_TOP=FLAG_BASE-118;
var SPAWN={x:2*T,y:9*T};

/* ---------- state ---------- */
var coins=[],parts=[],cam={x:0,y:0},won=false,runT=0,gTime=0,landFlash=0;
var p={x:0,y:0,w:26,h:36,vx:0,vy:0,face:1,grounded:false,coyote:0,jbuf:0,sq:1,dead:0};

function reset(){
  coins=COIN_POS.map(function(c){return{x:c[0]*T+T/2,y:c[1]*T+T/2,taken:false,ph:Math.random()*6.28};});
  parts=[];won=false;runT=0;
  p.x=SPAWN.x;p.y=SPAWN.y;p.vx=0;p.vy=0;p.face=1;p.grounded=false;p.coyote=0;p.jbuf=0;p.sq=1;p.dead=0;
  cam.x=Math.max(0,Math.min(p.x-vw*.42,worldW-vw));
  winEl.style.display='none';
  updateCoinHud();
}
function updateCoinHud(){
  var got=0;coins.forEach(function(c){if(c.taken)got++;});
  coinEl.innerHTML='&#9679; '+got+'/'+coins.length;
}

/* ---------- input ---------- */
var keys={},prevJump=false;
addEventListener('keydown',function(e){
  if(['ArrowLeft','ArrowRight','ArrowUp','ArrowDown','Space'].indexOf(e.code)>=0)e.preventDefault();
  keys[e.code]=true;
  if(e.code==='KeyR')reset();
});
addEventListener('keyup',function(e){keys[e.code]=false;});
function bindBtn(id,on,off){
  var el=document.getElementById(id);
  var dn=function(e){e.preventDefault();on();};
  var up=function(e){e.preventDefault();off();};
  el.addEventListener('pointerdown',dn);
  el.addEventListener('pointerup',up);
  el.addEventListener('pointercancel',up);
  el.addEventListener('pointerleave',up);
  el.addEventListener('contextmenu',function(e){e.preventDefault();});
}
var tL=false,tR=false,tJ=false;
bindBtn('tL',function(){tL=true;},function(){tL=false;});
bindBtn('tR',function(){tR=true;},function(){tR=false;});
bindBtn('tJ',function(){tJ=true;},function(){tJ=false;});
document.getElementById('reset').addEventListener('click',reset);
document.getElementById('again').addEventListener('click',reset);

/* ---------- particles ---------- */
function burst(x,y,n,col,spd,up){
  for(var i=0;i<n;i++){
    var a=Math.random()*6.283,s=spd*(0.35+Math.random()*0.85);
    parts.push({x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-up,life:.5+Math.random()*.5,t:0,col:col,sz:2+Math.random()*3,g:900});
  }
}
function confetti(){
  var cols=['#ffd34d','#4dd0e1','#ff7eb6','#8dff7e','#c792ea'];
  for(var i=0;i<80;i++)parts.push({x:FLAG_X,y:FLAG_TOP+20,vx:(Math.random()-.5)*420,vy:-Math.random()*520-60,life:1.2+Math.random()*1.4,t:0,col:cols[i%5],sz:3+Math.random()*4,g:620});
}

/* ---------- physics ---------- */
var GRAV=2300,RUN=270,ACC=2400,AIR_ACC=1600,JUMP=740;
function update(dt){
  gTime+=dt;
  if(!won&&p.dead<=0)runT+=dt;
  timeEl.textContent=runT.toFixed(1)+'s';

  var left=keys['ArrowLeft']||keys['KeyA']||tL;
  var right=keys['ArrowRight']||keys['KeyD']||tR;
  var jump=keys['ArrowUp']||keys['KeyW']||keys['Space']||tJ;
  var jp=jump&&!prevJump;prevJump=jump;

  if(won){left=right=false;jp=false;}

  if(p.dead>0){p.dead-=dt;if(p.dead<=0){p.x=SPAWN.x;p.y=SPAWN.y;p.vx=0;p.vy=0;}}

  var ax=0;
  if(left)ax-=1;if(right)ax+=1;
  if(ax!==0)p.face=ax;
  var acc=p.grounded?ACC:AIR_ACC;
  if(ax!==0){
    p.vx+=ax*acc*dt;
    if(p.vx>RUN)p.vx=RUN;if(p.vx<-RUN)p.vx=-RUN;
  }else{
    var f=(p.grounded?2600:500)*dt;
    if(Math.abs(p.vx)<=f)p.vx=0;else p.vx-=Math.sign(p.vx)*f;
  }
  p.vy+=GRAV*dt;if(p.vy>980)p.vy=980;

  if(jp)p.jbuf=.12;else p.jbuf=Math.max(0,p.jbuf-dt);
  if(p.grounded)p.coyote=.1;else p.coyote=Math.max(0,p.coyote-dt);
  if(p.jbuf>0&&p.coyote>0){
    p.vy=-JUMP;p.grounded=false;p.coyote=0;p.jbuf=0;p.sq=1.25;
    burst(p.x+p.w/2,p.y+p.h,5,'#8b90a3',70,30);
  }
  if(!jump&&p.vy<-260)p.vy=-260; /* variable jump height */

  /* X axis */
  p.x+=p.vx*dt;
  var t0=Math.floor(p.y/T),b0=Math.floor((p.y+p.h-0.01)/T);
  var l0,r0,tx,ty;
  if(p.vx>0){
    r0=Math.floor((p.x+p.w)/T);l0=Math.floor(p.x/T);
    for(ty=t0;ty<=b0;ty++)if(solid(r0,ty)){p.x=r0*T-p.w;p.vx=0;break;}
  }else if(p.vx<0){
    l0=Math.floor(p.x/T);
    for(ty=t0;ty<=b0;ty++)if(solid(l0,ty)){p.x=(l0+1)*T;p.vx=0;break;}
  }
  /* Y axis */
  p.y+=p.vy*dt;
  var wasG=p.grounded;p.grounded=false;
  var l1=Math.floor(p.x/T),r1=Math.floor((p.x+p.w-0.01)/T);
  if(p.vy>0){
    var bb=Math.floor((p.y+p.h)/T);
    for(tx=l1;tx<=r1;tx++)if(solid(tx,bb)){
      p.y=bb*T-p.h;
      if(p.vy>420){burst(p.x+p.w/2,p.y+p.h,6,'#7a6a55',90,20);p.sq=.72;}
      p.vy=0;p.grounded=true;break;
    }
  }else if(p.vy<0){
    var tt=Math.floor(p.y/T);
    for(tx=l1;tx<=r1;tx++)if(solid(tx,tt)){p.y=(tt+1)*T;p.vy=0;break;}
  }
  p.sq+=(1-p.sq)*Math.min(1,dt*10);

  /* fell in a pit */
  if(p.y>worldH+60&&p.dead<=0){p.dead=.55;burst(p.x+p.w/2,worldH-10,14,'#ff7eb6',160,80);}

  /* coins */
  var cx=p.x+p.w/2,cy=p.y+p.h/2;
  coins.forEach(function(c){
    if(!c.taken&&Math.abs(c.x-cx)<26&&Math.abs(c.y-cy)<30){
      c.taken=true;burst(c.x,c.y,9,'#ffd34d',140,60);updateCoinHud();
    }
  });

  /* flag */
  if(!won&&p.dead<=0&&cx>FLAG_X-16&&cx<FLAG_X+34&&cy>FLAG_TOP&&cy<FLAG_BASE+8){
    won=true;confetti();
    var got=0;coins.forEach(function(c){if(c.taken)got++;});
    winStats.textContent='Time '+runT.toFixed(1)+'s · Coins '+got+'/'+coins.length+(got===coins.length?' · Perfect run!':'');
    setTimeout(function(){winEl.style.display='flex';},700);
  }

  /* particles */
  for(var i=parts.length-1;i>=0;i--){
    var q=parts[i];q.t+=dt;
    if(q.t>=q.life){parts.splice(i,1);continue;}
    q.vy+=q.g*dt;q.x+=q.vx*dt;q.y+=q.vy*dt;
  }

  /* camera */
  var tx2=Math.max(0,Math.min(cx-vw*.42,Math.max(0,worldW-vw)));
  cam.x+=(tx2-cam.x)*Math.min(1,dt*7);
  if(worldH<=vh)cam.y=worldH-vh;
  else{var ty2=Math.max(0,Math.min(cy-vh*.55,worldH-vh));cam.y+=(ty2-cam.y)*Math.min(1,dt*7);}
  landFlash=Math.max(0,landFlash-dt);
}

/* ---------- deterministic hash for scenery ---------- */
function h1(i){var x=Math.sin(i*127.1+311.7)*43758.5453;return x-Math.floor(x);}

/* ---------- render ---------- */
function render(){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  /* sky */
  var g=ctx.createLinearGradient(0,0,0,vh);
  g.addColorStop(0,'#0b0e1d');g.addColorStop(.55,'#1d2340');g.addColorStop(1,'#3a2b52');
  ctx.fillStyle=g;ctx.fillRect(0,0,vw,vh);
  /* stars */
  ctx.fillStyle='#dfe6ff';
  for(var i=0;i<90;i++){
    var sx=((h1(i)*3000-cam.x*.05)%(vw+40)+vw+40)%(vw+40)-20;
    var sy=h1(i+500)*vh*.55;
    var tw=.35+.65*Math.abs(Math.sin(gTime*1.4+i));
    ctx.globalAlpha=tw*.8;ctx.fillRect(sx,sy,1.6,1.6);
  }
  ctx.globalAlpha=1;
  /* moon */
  var mx=vw*.78-cam.x*.03,my=vh*.2;
  var mg=ctx.createRadialGradient(mx,my,6,mx,my,90);
  mg.addColorStop(0,'rgba(240,225,190,.9)');mg.addColorStop(.25,'rgba(240,225,190,.28)');mg.addColorStop(1,'rgba(240,225,190,0)');
  ctx.fillStyle=mg;ctx.beginPath();ctx.arc(mx,my,90,0,6.29);ctx.fill();
  ctx.fillStyle='#f3e8c8';ctx.beginPath();ctx.arc(mx,my,22,0,6.29);ctx.fill();

  var groundY=groundRow*T-cam.y;

  /* far mountains (parallax .18) */
  drawRidge(.18,340,150,groundY+30,'#1a2038');
  /* clouds (parallax .3 + drift) */
  ctx.fillStyle='rgba(190,200,230,.10)';
  for(i=0;i<10;i++){
    var cxx=((i*360+h1(i+77)*200-cam.x*.3+gTime*8)%(vw+420)+vw+420)%(vw+420)-210;
    var cyy=40+h1(i+31)*vh*.32;
    ctx.beginPath();
    ctx.ellipse(cxx,cyy,60,14,0,0,6.29);
    ctx.ellipse(cxx+34,cyy+4,42,11,0,0,6.29);
    ctx.ellipse(cxx-36,cyy+5,38,10,0,0,6.29);
    ctx.fill();
  }
  /* nearer hills (parallax .45) */
  drawRidge(.45,240,80,groundY+16,'#232c4a');
  /* tree silhouettes (parallax .68) */
  ctx.fillStyle='#141c2c';
  var sp=230;
  var i0=Math.floor((cam.x*.68-80)/sp),i1=Math.ceil((cam.x*.68+vw+80)/sp);
  for(i=i0;i<=i1;i++){
    var tx3=i*sp+h1(i+9)*90-cam.x*.68;
    var th=46+h1(i+4)*44,by=groundY+8;
    ctx.beginPath();
    ctx.moveTo(tx3,by-th);ctx.lineTo(tx3-16,by);ctx.lineTo(tx3+16,by);ctx.closePath();ctx.fill();
    ctx.fillRect(tx3-3,by-6,6,10);
  }

  /* world */
  ctx.save();ctx.translate(-cam.x,-cam.y);
  var x0=Math.max(0,Math.floor(cam.x/T)-1),x1=Math.min(W-1,Math.ceil((cam.x+vw)/T)+1);
  var y0=Math.max(0,Math.floor(cam.y/T)-1),y1=Math.min(H-1,Math.ceil((cam.y+vh)/T)+1);
  for(ty=y0;ty<=y1;ty++)for(tx=x0;tx<=x1;tx++){
    var c=grid[ty][tx];
    if(c==='#'){
      ctx.fillStyle='#4a3527';ctx.fillRect(tx*T,ty*T,T,T);
      ctx.fillStyle='#3a2a1f';ctx.fillRect(tx*T+4,ty*T+14,8,6);ctx.fillRect(tx*T+22,ty*T+26,10,5);
      if(!solid(tx,ty-1)){ctx.fillStyle='#43a35c';ctx.fillRect(tx*T,ty*T,T,10);ctx.fillStyle='#57c473';ctx.fillRect(tx*T,ty*T,T,4);}
    }else if(c==='='){
      ctx.fillStyle='#5d6f9e';
      ctx.beginPath();
      ctx.moveTo(tx*T,ty*T+4);ctx.quadraticCurveTo(tx*T,ty*T,tx*T+4,ty*T);
      ctx.lineTo(tx*T+T-4,ty*T);ctx.quadraticCurveTo(tx*T+T,ty*T,tx*T+T,ty*T+4);
      ctx.lineTo(tx*T+T,ty*T+14);ctx.lineTo(tx*T,ty*T+14);ctx.closePath();ctx.fill();
      ctx.fillStyle='#8ba0d8';ctx.fillRect(tx*T,ty*T,T,4);
    }
  }

  /* flag */
  ctx.strokeStyle='#c8cede';ctx.lineWidth=4;ctx.lineCap='round';
  ctx.beginPath();ctx.moveTo(FLAG_X,FLAG_BASE);ctx.lineTo(FLAG_X,FLAG_TOP);ctx.stroke();
  ctx.fillStyle='#e8b93e';ctx.beginPath();ctx.arc(FLAG_X,FLAG_TOP-4,5,0,6.29);ctx.fill();
  var fw=won?Math.sin(gTime*10)*2:Math.sin(gTime*5)*3;
  ctx.fillStyle='#ff5d73';
  ctx.beginPath();
  ctx.moveTo(FLAG_X+2,FLAG_TOP+2);
  ctx.quadraticCurveTo(FLAG_X+30,FLAG_TOP+8+fw,FLAG_X+48,FLAG_TOP+14+fw*2);
  ctx.quadraticCurveTo(FLAG_X+28,FLAG_TOP+24+fw,FLAG_X+2,FLAG_TOP+30);
  ctx.closePath();ctx.fill();

  /* coins */
  coins.forEach(function(cn){
    if(cn.taken)return;
    var s=Math.abs(Math.cos(gTime*3.4+cn.ph));
    var bob=Math.sin(gTime*2.2+cn.ph)*3;
    ctx.save();ctx.translate(cn.x,cn.y+bob);
    var cg=ctx.createRadialGradient(0,-2,1,0,0,11);
    cg.addColorStop(0,'#fff0b0');cg.addColorStop(.5,'#ffd34d');cg.addColorStop(1,'#c8921e');
    ctx.fillStyle=cg;
    ctx.beginPath();ctx.ellipse(0,0,9*Math.max(.18,s),9,0,0,6.29);ctx.fill();
    ctx.strokeStyle='#8a6410';ctx.lineWidth=1.5;ctx.stroke();
    ctx.restore();
  });

  /* player */
  if(p.dead<=0){
    ctx.save();
    ctx.translate(p.x+p.w/2,p.y+p.h);
    ctx.scale(2-p.sq,p.sq);
    ctx.translate(-p.w/2,-p.h);
    var pg=ctx.createLinearGradient(0,0,0,p.h);
    pg.addColorStop(0,'#6fe3f4');pg.addColorStop(1,'#28a8c9');
    ctx.fillStyle=pg;
    roundRect(0,0,p.w,p.h,7);ctx.fill();
    ctx.strokeStyle='#0e5f78';ctx.lineWidth=2;ctx.stroke();
    /* eyes */
    var ex=p.face*4;
    ctx.fillStyle='#fff';
    ctx.beginPath();ctx.arc(p.w/2-5+ex,12,4.5,0,6.29);ctx.arc(p.w/2+5+ex,12,4.5,0,6.29);ctx.fill();
    ctx.fillStyle='#12212e';
    ctx.beginPath();ctx.arc(p.w/2-5+ex*1.5,12,2,0,6.29);ctx.arc(p.w/2+5+ex*1.5,12,2,0,6.29);ctx.fill();
    /* feet */
    ctx.fillStyle='#0e5f78';
    var st=p.grounded&&Math.abs(p.vx)>30?Math.sin(gTime*14)*3:0;
    ctx.fillRect(3+st,p.h-4,8,4);ctx.fillRect(p.w-11-st,p.h-4,8,4);
    ctx.restore();
  }

  /* particles */
  parts.forEach(function(q){
    ctx.globalAlpha=Math.max(0,1-q.t/q.life);
    ctx.fillStyle=q.col;ctx.fillRect(q.x-q.sz/2,q.y-q.sz/2,q.sz,q.sz);
  });
  ctx.globalAlpha=1;
  ctx.restore();

  /* vignette */
  var vg=ctx.createRadialGradient(vw/2,vh/2,Math.min(vw,vh)*.35,vw/2,vh/2,Math.max(vw,vh)*.75);
  vg.addColorStop(0,'rgba(0,0,0,0)');vg.addColorStop(1,'rgba(0,0,0,.35)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,vw,vh);
}
function roundRect(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}
function drawRidge(f,spacing,hMax,baseY,col){
  ctx.fillStyle=col;
  var i0=Math.floor((cam.x*f-spacing)/spacing),i1=Math.ceil((cam.x*f+vw+spacing)/spacing);
  for(var i=i0;i<=i1;i++){
    var px=i*spacing+h1(i+13)*spacing*.55-cam.x*f;

  }
}
